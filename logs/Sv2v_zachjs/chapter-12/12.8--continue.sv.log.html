module jump_tb;

      
	initial begin : sv2v_autoblock_1

      
		reg [0:1] _sv2v_jump;

      
		_sv2v_jump = 2'b00;

      
		begin : sv2v_autoblock_2

      
			reg signed [31:0] i;

      
			begin : sv2v_autoblock_3

      
				reg signed [31:0] _sv2v_value_on_break;

      
				for (i = 0; i < 256; i = i + 1)

      
					if (_sv2v_jump < 2'b10) begin

      
						_sv2v_jump = 2'b00;

      
						if (i < 255)

      
							_sv2v_jump = 2'b01;

      
						if (_sv2v_jump == 2'b00)

      
							$display(":assert:(%d == 255)", i);

      
						_sv2v_value_on_break = i;

      
					end

      
				if (!(_sv2v_jump < 2'b10))

      
					i = _sv2v_value_on_break;

      
				if (_sv2v_jump != 2'b11)

      
					_sv2v_jump = 2'b00;

      
			end

      
		end

      
	end
