module bsg_fifo_tracker (

      
	clk_i,

      
	reset_i,

      
	enq_i,

      
	deq_i,

      
	wptr_r_o,

      
	rptr_r_o,

      
	rptr_n_o,

      
	full_o,

      
	empty_o

      
);

      
	parameter els_p = 0;

      
	parameter ptr_width_lp = ((els_p == 1) || (els_p == 0) ? 1 : $clog2(els_p));

      
	input clk_i;

      
	input reset_i;

      
	input enq_i;

      
	input deq_i;

      
	output wire [ptr_width_lp - 1:0] wptr_r_o;

      
	output wire [ptr_width_lp - 1:0] rptr_r_o;

      
	output wire [ptr_width_lp - 1:0] rptr_n_o;

      
	output wire full_o;

      
	output wire empty_o;

      
	wire [ptr_width_lp - 1:0] rptr_r;

      
	wire [ptr_width_lp - 1:0] rptr_n;

      
	wire [ptr_width_lp - 1:0] wptr_r;

      
	assign wptr_r_o = wptr_r;

      
	assign rptr_r_o = rptr_r;

      
	assign rptr_n_o = rptr_n;

      
	reg enq_r;

      
	reg deq_r;

      
	wire empty;

      
	wire full;

      
	wire equal_ptrs;

      
	bsg_circular_ptr #(

      
		.slots_p(els_p),

      
		.max_add_p(1)

      
	) rptr(

      
		.clk(clk_i),

      
		.reset_i(reset_i),

      
		.add_i(deq_i),

      
		.o(rptr_r),

      
		.n_o(rptr_n)

      
	);

      
	bsg_circular_ptr #(

      
		.slots_p(els_p),

      
		.max_add_p(1)

      
	) wptr(

      
		.clk(clk_i),

      
		.reset_i(reset_i),

      
		.add_i(enq_i),

      
		.o(wptr_r),

      
		.n_o()

      
	);

      
	always @(posedge clk_i)

      
		if (reset_i) begin

      
			enq_r <= 1'b0;

      
			deq_r <= 1'b1;

      
		end

      
		else if (enq_i | deq_i) begin

      
			enq_r <= enq_i;

      
			deq_r <= deq_i;

      
		end

      
	assign equal_ptrs = rptr_r == wptr_r;

      
	assign empty = equal_ptrs & deq_r;

      
	assign full = equal_ptrs & enq_r;

      
	assign full_o = full;

      
	assign empty_o = empty;

      
endmodule
