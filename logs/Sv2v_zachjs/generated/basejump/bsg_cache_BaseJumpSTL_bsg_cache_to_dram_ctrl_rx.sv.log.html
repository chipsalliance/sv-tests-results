	input clk_i;

      
	input reset_i;

      
	output wire [dma_data_width_p - 1:0] dma_data_o;

      
	output wire dma_data_v_o;

      
	input dma_data_ready_and_i;

      
	input app_rd_data_valid_i;

	wire fifo_yumi_li;
