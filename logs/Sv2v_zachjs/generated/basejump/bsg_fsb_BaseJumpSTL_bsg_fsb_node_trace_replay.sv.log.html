module bsg_fsb_node_trace_replay (

      
	clk_i,

      
	reset_i,

      
	en_i,

      
	v_i,

      
	data_i,

      
	ready_o,

      
	v_o,

      
	data_o,

      
	yumi_i,

      
	rom_addr_o,

      
	rom_data_i,

      
	done_o,

      
	error_o

      
);

      
	parameter ring_width_p = 80;

      
	parameter rom_addr_width_p = 6;

      
	parameter counter_width_p = (ring_width_p < 16 ? ring_width_p : 16);

      
	parameter uptime_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input en_i;

      
	input v_i;

      
	input [ring_width_p - 1:0] data_i;

      
	output reg ready_o;

      
	output reg v_o;

      
	output wire [ring_width_p - 1:0] data_o;

      
	input yumi_i;

      
	output wire [rom_addr_width_p - 1:0] rom_addr_o;

      
	input [ring_width_p + 3:0] rom_data_i;

      
	output wire done_o;

      
	output wire error_o;

      
	initial $display("## WARNING: bsg_fsb_node_trace_replay will be DEPRECATED soon; please discontinue use (%m).");

      
	reg [counter_width_p - 1:0] cycle_ctr_r;

      
	reg [counter_width_p - 1:0] cycle_ctr_n;

      
	reg [rom_addr_width_p - 1:0] addr_r;

      
	wire [rom_addr_width_p - 1:0] addr_n;

      
	reg done_r;

      
	reg done_n;

      
	reg error_r;

      
	reg error_n;

      
	assign rom_addr_o = addr_r;

      
	assign data_o = rom_data_i[0+:ring_width_p];

      
	assign done_o = done_r;

      
	assign error_o = error_r;

      
	always @(posedge clk_i)

      
		if (reset_i) begin

      
			addr_r <= 0;

      
			done_r <= 0;

      
			error_r <= 0;

      
			cycle_ctr_r <= 16'b0000000000000001;

      
		end

      
		else begin

      
			addr_r <= addr_n;

      
			done_r <= done_n;

      
			error_r <= error_n;

      
			cycle_ctr_r <= cycle_ctr_n;

      
		end

      
	wire [3:0] op;

      
	assign op = rom_data_i[ring_width_p+:4];

      
	reg instr_completed;

      
	assign addr_n = (instr_completed ? addr_r + 1'b1 : addr_r);

      
	always @(*) begin

      
		v_o = 1'b0;

      
		ready_o = done_r;

      
		done_n = done_r;

      
		if ((!done_r & en_i) & ~reset_i)

      
			case (op)

      
				4'd1: v_o = 1'b1;

      
				4'd2: ready_o = 1'b1;

      
				4'd3: done_n = 1'b1;

      
				default:

      
					;

      
			endcase

      
	end

      
	always @(*) begin

      
		instr_completed = 1'b0;

      
		error_n = error_r;

      
		cycle_ctr_n = cycle_ctr_r;

      
		if ((!done_r & en_i) & ~reset_i)

      
			case (op)

      
				4'd0: instr_completed = 1'b1;

      
				4'd1:

      
					if (yumi_i)

      
						instr_completed = 1'b1;

      
				4'd2:

      
					if (v_i) begin

      
						instr_completed = 1'b1;

      
						if (error_r == 0)

      
							error_n = data_i != data_o;

      
					end

      
				4'd3: instr_completed = 1'b1;

      
				4'd4: instr_completed = 1'b1;

      
				4'd5: begin

      
					cycle_ctr_n = cycle_ctr_r - 1'b1;

      
					instr_completed = ~(|cycle_ctr_r);

      
				end

      
				4'd6: begin

      
					cycle_ctr_n = rom_data_i[counter_width_p - 1:0];

      
					instr_completed = 1;

      
				end

      
				default:

      
					;

      
			endcase

      
	end

      
	always @(negedge clk_i)

      
		if ((instr_completed & ~reset_i) & ~done_r) begin

      
			case (op)

      
				4'd1: begin : sv2v_autoblock_1

      
					string localtime;

      
					if (uptime_p) begin : sv2v_autoblock_2

      
						reg signed [31:0] fd;

      
						fd = $fopen("/proc/uptime", "r");

      
						$fscanf(fd, "%s", localtime);

      
						$fclose(fd);

      
					end

      
					else

      
						localtime = "";

      
					$display("### bsg_fsb_node_trace_replay SEND %d'b%b (%m), time=%t uptime=%s", ring_width_p, data_o, $time, localtime);

      
				end

      
				4'd2:

      
					if (data_i !== data_o) begin

      
						$display("############################################################################");

      
						$display("### bsg_fsb_node_trace_replay RECEIVE unmatched (%m) ");

      
						$display("###    ");

      
						$display("### FAIL (trace mismatch) = %h", data_i);

      
						$display("###              expected = %h\n", data_o);

      
						$display("###              diff     = %h\n", data_o ^ data_i);

      
						$display("############################################################################");

      
						$finish;

      
					end

      
					else

      
						$display("### bsg_fsb_node_trace_replay RECEIVE matched %h (%m)", data_o);

      
				4'd3: begin

      
					$display("############################################################################");

      
					$display("###### bsg_fsb_node_trace_replay DONE done_o=1 (trace finished addr=%x) (%m)", rom_addr_o);

      
					$display("############################################################################");

      
				end

      
				4'd4: begin

      
					$display("############################################################################");

      
					$display("###### bsg_fsb_node_trace_replay FINISH (trace finished; CALLING $finish) (%m)");

      
					$display("############################################################################");

      
					$finish;

      
				end

      
				4'd5:

      
					$display("### bsg_fsb_node_trace_replay CYCLE DEC cycle_ctr_r = %x (%m)", cycle_ctr_r);

      
				4'd6:

      
					$display("### bsg_fsb_node_trace_replay CYCLE INIT = %x (%m)", cycle_ctr_n);

      
				default:

      
					;

      
			endcase

      
			case (op)

      
				4'd0, 4'd1, 4'd2, 4'd3, 4'd4, 4'd5, 4'd6:

      
					;

      
				default:

      
					$display("### bsg_fsb_node_trace_replay UNKNOWN op %x (%m)\n", op);

      
			endcase

      
		end

      
endmodule
