module bsg_lru_pseudo_tree_backup (

      
	disabled_ways_i,

      
	modify_mask_o,

      
	modify_data_o

      
);

      
	parameter ways_p = 0;

      
	parameter lg_ways_lp = (ways_p == 1 ? 1 : $clog2(ways_p));

      
	input [ways_p - 1:0] disabled_ways_i;

      
	output wire [(ways_p < 2 ? 0 : ways_p - 2):0] modify_mask_o;

      
	output wire [(ways_p < 2 ? 0 : ways_p - 2):0] modify_data_o;

      
	generate

      
		if (ways_p == 1) begin : no_lru

      
			assign modify_mask_o = 1'b1;

      
			assign modify_data_o = 1'b0;

      
		end

      
		else begin : lru

      
			genvar i;

      
			for (i = 0; i < lg_ways_lp; i = i + 1) begin : genblk1

      
				wire [(2 ** (i + 1)) - 1:0] and_reduce;

      
				genvar j;

      
				for (j = 0; j < (2 ** (i + 1)); j = j + 1) begin : genblk1

      
					assign and_reduce[j] = &disabled_ways_i[(ways_p / (2 ** (i + 1))) * j+:ways_p / (2 ** (i + 1))];

      
				end

      
				genvar k;

      
				for (k = 0; k < ((2 ** (i + 1)) / 2); k = k + 1) begin : genblk2

      
					assign modify_data_o[((2 ** i) - 1) + k] = and_reduce[2 * k];

      
					assign modify_mask_o[((2 ** i) - 1) + k] = |and_reduce[2 * k+:2];

      
				end

      
			end

      
		end

      
	endgenerate

      
endmodule
