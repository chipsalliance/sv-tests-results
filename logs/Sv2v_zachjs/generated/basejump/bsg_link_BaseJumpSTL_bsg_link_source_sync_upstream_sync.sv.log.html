module bsg_link_source_sync_upstream_sync (

      
	io_clk_i,

      
	io_link_reset_i,

      
	async_token_reset_i,

      
	io_v_i,

      
	io_data_i,

      
	io_ready_and_o,

      
	io_v_o,

      
	io_data_o,

      
	token_clk_i

      
);

      
	parameter width_p = 0;

      
	parameter lg_fifo_depth_p = 3;

      
	parameter lg_credit_to_token_decimation_p = 0;

      
	parameter bypass_twofer_fifo_p = 0;

      
	input io_clk_i;

      
	input io_link_reset_i;

      
	input async_token_reset_i;

      
	input io_v_i;

      
	input [width_p - 1:0] io_data_i;

      
	output wire io_ready_and_o;

      
	output wire io_v_o;

      
	output wire [width_p - 1:0] io_data_o;

      
	input token_clk_i;

      
	wire io_fifo_v;

      
	wire io_fifo_ready;

      
	wire [width_p - 1:0] io_fifo_data;

      
	generate

      
		if (bypass_twofer_fifo_p == 0) begin : twofer

      
			bsg_two_fifo #(.width_p(width_p)) twofer_fifo(

      
				.clk_i(io_clk_i),

      
				.reset_i(io_link_reset_i),

      
				.ready_param_o(io_ready_and_o),

      
				.data_i(io_data_i),

      
				.v_i(io_v_i),

      
				.v_o(io_fifo_v),

      
				.data_o(io_fifo_data),

      
				.yumi_i(io_fifo_v & io_fifo_ready)

      
			);

      
		end

      
		else begin : no_twofer

      
			assign io_fifo_v = io_v_i;

      
			assign io_fifo_data = io_data_i;

      
			assign io_ready_and_o = (io_link_reset_i ? 1'b1 : io_fifo_ready);

      
		end

      
	endgenerate

      
	wire io_v_n;

      
	assign io_v_o = (io_link_reset_i ? 1'b0 : io_v_n);

      
	assign io_data_o = (io_link_reset_i | ~io_v_n ? {width_p {1'sb0}} : io_fifo_data);

      
	wire [lg_credit_to_token_decimation_p + 0:0] io_token_alternator_r;

      
	bsg_counter_clear_up #(

      
		.max_val_p({lg_credit_to_token_decimation_p + 1 {1'b1}}),

      
		.init_val_p(0),

      
		.disable_overflow_warning_p(1)

      
	) token_alt(

      
		.clk_i(io_clk_i),

      
		.reset_i(io_link_reset_i),

      
		.clear_i(1'b0),

      
		.up_i(io_v_n),

      
		.count_o(io_token_alternator_r)

      
	);

      
	wire io_on_negedge_token = io_token_alternator_r[lg_credit_to_token_decimation_p];

      
	wire io_negedge_credits_avail;

      
	wire io_posedge_credits_avail;

      
	wire io_credit_avail = (io_on_negedge_token ? io_negedge_credits_avail : io_posedge_credits_avail);

      
	assign io_v_n = io_credit_avail & io_fifo_v;

      
	assign io_fifo_ready = (io_link_reset_i ? 1'b1 : io_credit_avail);

      
	wire io_negedge_credits_deque = io_v_n & io_on_negedge_token;

      
	wire io_posedge_credits_deque = io_v_n & ~io_on_negedge_token;

      
	bsg_async_credit_counter #(

      
		.max_tokens_p(2 ** ((lg_fifo_depth_p - 1) - lg_credit_to_token_decimation_p)),

      
		.lg_credit_to_token_decimation_p(lg_credit_to_token_decimation_p),

      
		.count_negedge_p(1'b0),

      
		.extra_margin_p(0),

      
		.start_full_p(1),

      
		.use_async_w_reset_p(1'b1)

      
	) pos_credit_ctr(

      
		.w_clk_i(token_clk_i),

      
		.w_inc_token_i(1'b1),

      
		.w_reset_i(async_token_reset_i),

      
		.r_clk_i(io_clk_i),

      
		.r_reset_i(io_link_reset_i),

      
		.r_dec_credit_i(io_posedge_credits_deque),

      
		.r_infinite_credits_i(1'b0),

      
		.r_credits_avail_o(io_posedge_credits_avail)

      
	);

      
	bsg_async_credit_counter #(

      
		.max_tokens_p(2 ** ((lg_fifo_depth_p - 1) - lg_credit_to_token_decimation_p)),

      
		.lg_credit_to_token_decimation_p(lg_credit_to_token_decimation_p),

      
		.count_negedge_p(1'b1),

      
		.extra_margin_p(0),

      
		.start_full_p(1),

      
		.use_async_w_reset_p(1'b1)

      
	) neg_credit_ctr(

      
		.w_clk_i(token_clk_i),

      
		.w_inc_token_i(1'b1),

      
		.w_reset_i(async_token_reset_i),

      
		.r_clk_i(io_clk_i),

      
		.r_reset_i(io_link_reset_i),

      
		.r_dec_credit_i(io_negedge_credits_deque),

      
		.r_infinite_credits_i(1'b0),

      
		.r_credits_avail_o(io_negedge_credits_avail)

      
	);

      
endmodule
