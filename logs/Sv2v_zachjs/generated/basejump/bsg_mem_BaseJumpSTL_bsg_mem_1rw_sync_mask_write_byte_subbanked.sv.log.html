module bsg_mem_1rw_sync_mask_write_byte_subbanked (

      
	clk_i,

      
	reset_i,

      
	v_i,

      
	w_i,

      
	w_mask_i,

      
	data_i,

      
	addr_i,

      
	data_o

      
);

      
	parameter width_p = 0;

      
	parameter els_p = 0;

      
	parameter latch_last_read_p = 0;

      
	parameter num_subbank_p = 0;

      
	localparam subbank_width_lp = width_p / num_subbank_p;

      
	localparam mask_width_lp = subbank_width_lp >> 3;

      
	localparam lg_els_lp = (els_p == 1 ? 1 : $clog2(els_p));

      
	input clk_i;

      
	input reset_i;

      
	input [num_subbank_p - 1:0] v_i;

      
	input w_i;

      
	input [(num_subbank_p * mask_width_lp) - 1:0] w_mask_i;

      
	input [(num_subbank_p * subbank_width_lp) - 1:0] data_i;

      
	input [lg_els_lp - 1:0] addr_i;

      
	output wire [(num_subbank_p * subbank_width_lp) - 1:0] data_o;

      
	wire [(num_subbank_p * subbank_width_lp) - 1:0] data_lo;

      
	wire [(num_subbank_p * mask_width_lp) - 1:0] w_mask_lo;

      
	genvar i;

      
	generate

      
		for (i = 0; i < num_subbank_p; i = i + 1) begin : genblk1

      
			genvar j;

      
			for (j = 0; j < mask_width_lp; j = j + 1) begin : genblk1

      
				assign w_mask_lo[(i * mask_width_lp) + j] = w_mask_i[(i * mask_width_lp) + j] & v_i[i];

      
			end

      
		end

      
	endgenerate

      
	bsg_mem_1rw_sync_mask_write_byte #(

      
		.data_width_p(width_p),

      
		.els_p(els_p),

      
		.latch_last_read_p(0)

      
	) bank(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.v_i(|v_i),

      
		.w_i(w_i),

      
		.addr_i(addr_i),

      
		.data_i(data_i),

      
		.write_mask_i(w_mask_lo),

      
		.data_o(data_lo)

      
	);

      
	generate

      
		if (latch_last_read_p) begin : llr

      
			wire [num_subbank_p - 1:0] read_en;

      
			wire [num_subbank_p - 1:0] read_en_r;

      
			genvar i;

      
			for (i = 0; i < num_subbank_p; i = i + 1) begin : bk1

      
				assign read_en[i] = v_i[i] & ~w_i;

      
				bsg_dff #(.width_p(1)) read_en_dff(

      
					.clk_i(clk_i),

      
					.data_i(read_en[i]),

      
					.data_o(read_en_r[i])

      
				);

      
				bsg_dff_en_bypass #(.width_p(subbank_width_lp)) dff_bypass(

      
					.clk_i(clk_i),

      
					.en_i(read_en_r[i]),

      
					.data_i(data_lo[i * subbank_width_lp+:subbank_width_lp]),

      
					.data_o(data_o[i * subbank_width_lp+:subbank_width_lp])

      
				);

      
			end

      
		end

      
		else begin : no_llr

      
			assign data_o = data_lo;

      
		end

      
		if (!(((1 << $clog2(width_p)) == width_p) && ((1 << $clog2(els_p)) == els_p))) begin : genblk3

      
			$error("width_p and els_p should be power of 2");

      
		end

      
		if ((width_p % num_subbank_p) != 0) begin : genblk4

      
			$error("The remainder for the factor -> (width_p/num_subbank_p) should be equal to 0");

      
		end

      
		if ((subbank_width_lp % 8) != 0) begin : genblk5

      
			$error("For byte-mask SRAM, subbank_width_lp should be a multiple of 8");

      
		end

      
	endgenerate

      
endmodule
