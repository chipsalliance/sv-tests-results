	parameter width_p = 0;

      
	parameter fan_out_p = 2;

      
	input clk_i;

      
	input reset_i;

      
	output wire ready_o;

      
	input v_i;

      
	input [width_p - 1:0] data_i;

      
	output wire [fan_out_p - 1:0] v_o;

      
	output wire [(fan_out_p * width_p) - 1:0] data_o;

      
	input [fan_out_p - 1:0] ready_i;

      
	reg [fan_out_p - 1:0] sent_r;

      
	reg [fan_out_p - 1:0] sent_n;

      
	genvar _gv_i_1;

      
	wire [fan_out_p - 1:0] v_o_tmp;

      
	wire [width_p - 1:0] data_o_tmp;

      
	wire fifo_v;

      
	wire fifo_yumi;

      
	bsg_two_fifo #(.width_p(width_p)) fifo(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.data_i(data_i),

      
		.data_o(data_o_tmp),

      
		.v_o(fifo_v),

      
		.yumi_i(fifo_yumi),

      
		.ready_o(ready_o),

      
		.v_i(v_i)

      
	);

		for (_gv_i_1 = 0; _gv_i_1 < fan_out_p; _gv_i_1 = _gv_i_1 + 1) begin : genblk1

			assign data_o[i * width_p+:width_p] = data_o_tmp;

      
			assign v_o[i] = v_o_tmp[i];

			end

      
		end

      
	endgenerate
