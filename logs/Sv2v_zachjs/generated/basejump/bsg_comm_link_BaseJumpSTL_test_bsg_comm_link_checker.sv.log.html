module test_bsg_comm_link_checker (

      
	clk,

      
	valid_in,

      
	ready_in,

      
	yumi_out,

      
	data_in,

      
	data_out,

      
	async_reset,

      
	slave_reset_tline,

      
	io_valid_tline,

      
	io_data_tline,

      
	core_ctr,

      
	io_ctr,

      
	done_o

      
);

      
	parameter channel_width_p = 0;

      
	parameter num_channels_p = 0;

      
	parameter ring_bytes_p = 0;

      
	parameter check_bytes_p = 0;

      
	parameter verbose_p = 0;

      
	parameter iterations_p = 0;

      
	parameter core_0_period_p = 0;

      
	parameter core_1_period_p = 0;

      
	parameter io_master_0_period_p = 0;

      
	parameter io_master_1_period_p = 0;

      
	parameter chip_num_p = 0;

      
	parameter node_num_p = 0;

      
	parameter cycle_counter_width_p = 0;

      
	parameter skip_checks_p = 0;

      
	input clk;

      
	input valid_in;

      
	input ready_in;

      
	input yumi_out;

      
	input [(ring_bytes_p * channel_width_p) - 1:0] data_in;

      
	input [(ring_bytes_p * channel_width_p) - 1:0] data_out;

      
	input async_reset;

      
	input slave_reset_tline;

      
	input [num_channels_p - 1:0] io_valid_tline;

      
	input [(num_channels_p * channel_width_p) - 1:0] io_data_tline;

      
	input [(2 * cycle_counter_width_p) - 1:0] core_ctr;

      
	input [(2 * cycle_counter_width_p) - 1:0] io_ctr;

      
	output wire done_o;

      
	localparam channel_verbose_p = 1'b0;

      
	reg [5:0] top_bits = 0;

      
	reg [31:0] words_received_r;

      
	wire [(check_bytes_p * channel_width_p) - 1:0] data_in_check;

      
	genvar _gv_j_1;

      
	always @(negedge clk)

      
		if (async_reset)

      
			words_received_r <= 0;

      
		else

      
			words_received_r <= words_received_r + (valid_in & ready_in);

      
	reg done_r;

      
	assign done_o = done_r;

      
	test_bsg_data_gen #(

      
		.channel_width_p(channel_width_p),

      
		.num_channels_p(check_bytes_p)

      
	) tbdg_receive(

      
		.clk_i(clk),

      
		.reset_i(async_reset),

      
		.yumi_i(ready_in & valid_in),

      
		.o(data_in_check)

      
	);

      
	always @(negedge clk) begin

      
		if (valid_in & ready_in) begin

      
			if (verbose_p)

      
				$display("## SR=%1d", slave_reset_tline, core_ctr[0+:cycle_counter_width_p], io_ctr[0+:cycle_counter_width_p], core_ctr[cycle_counter_width_p+:cycle_counter_width_p], io_ctr[cycle_counter_width_p+:cycle_counter_width_p], " ## chip %1d node %1d recv %-d, %x", chip_num_p, node_num_p, words_received_r, data_in);

      
			if (!skip_checks_p)

      
				;

      
			if (((words_received_r >= (iterations_p << (channel_width_p - $clog2(num_channels_p)))) & (chip_num_p == 0)) & ~done_r) begin

      
				done_r <= 1'b1;

      
				$display("## DONE node = %-d words = %-d CHANNEL_BITWIDTH = %-d", node_num_p, words_received_r, channel_width_p, " RING_BYTES = %-d;", ring_bytes_p, " NUM_CHAN = %-d;", num_channels_p, " C0 = %-d;", core_0_period_p, " I0 = %-d; I1 = %-d;", io_master_0_period_p, io_master_1_period_p, " C1 = %-d;", core_1_period_p, , " (Cycles Per Word) ", real'(core_ctr[0+:cycle_counter_width_p]) / real'(words_received_r), " ", real'(io_ctr[0+:cycle_counter_width_p]) / real'(words_received_r), " ", real'(io_ctr[cycle_counter_width_p+:cycle_counter_width_p]) / real'(words_received_r), " ", real'(core_ctr[cycle_counter_width_p+:cycle_counter_width_p]) / real'(words_received_r));

      
			end

      
		end

      
		if (yumi_out) begin

      
			if (verbose_p)

      
				$display("## SR=%1d", slave_reset_tline, core_ctr[0+:cycle_counter_width_p], io_ctr[0+:cycle_counter_width_p], core_ctr[cycle_counter_width_p+:cycle_counter_width_p], io_ctr[cycle_counter_width_p+:cycle_counter_width_p], " ## chip %1d node %1d sent %x", chip_num_p, node_num_p, data_out);

      
		end

      
		if (async_reset)

      
			done_r <= 1'b0;

      
	end

      
	generate

      
		if (node_num_p == 0) begin : genblk1

      
			for (_gv_j_1 = 0; _gv_j_1 < num_channels_p; _gv_j_1 = _gv_j_1 + 1) begin : genblk1

      
				localparam j = _gv_j_1;

      
				always @(slave_reset_tline or io_valid_tline[j] or io_data_tline[j * channel_width_p+:channel_width_p] or guts.comm_link.ch[j].sso.pos_credit_ctr.r_free_credits_r or guts.comm_link.ch[j].sso.neg_credit_ctr.r_free_credits_r)

      
					if (verbose_p) begin

      
						if (channel_verbose_p)

      
							$display("## SR=%1d", slave_reset_tline, core_ctr[0+:cycle_counter_width_p], io_ctr[0+:cycle_counter_width_p], core_ctr[cycle_counter_width_p+:cycle_counter_width_p], io_ctr[cycle_counter_width_p+:cycle_counter_width_p], " ## chip %1d channel %1d", chip_num_p, j, " (p,n)=(%2d %2d)", guts.comm_link.ch[j].sso.pos_credit_ctr.r_free_credits_r, guts.comm_link.ch[j].sso.neg_credit_ctr.r_free_credits_r, " ## io     xmit %1d,%x", io_valid_tline[j], io_data_tline[j * channel_width_p+:channel_width_p]);

      
					end

      
			end

      
		end

      
	endgenerate

      
endmodule
