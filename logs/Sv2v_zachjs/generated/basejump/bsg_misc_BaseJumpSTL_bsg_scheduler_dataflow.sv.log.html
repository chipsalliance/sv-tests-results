module bsg_scheduler_dataflow_entry (

      
	clk_i,

      
	reset_i,

      
	we_i,

      
	src_tags_v_i,

      
	src_tags_i,

      
	wakeup_tags_v_i,

      
	wakeup_tags_i,

      
	ready_o,

      
	bypass_o

      
);

      
	parameter tag_width_p = 0;

      
	parameter src_tags_p = 0;

      
	parameter wakeup_tags_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input we_i;

      
	input [src_tags_p - 1:0] src_tags_v_i;

      
	input [(src_tags_p * tag_width_p) - 1:0] src_tags_i;

      
	input [wakeup_tags_p - 1:0] wakeup_tags_v_i;

      
	input [(wakeup_tags_p * tag_width_p) - 1:0] wakeup_tags_i;

      
	output wire ready_o;

      
	output wire [(src_tags_p * wakeup_tags_p) - 1:0] bypass_o;

      
	reg [(src_tags_p * tag_width_p) - 1:0] src_tags_r;

      
	reg [(src_tags_p * tag_width_p) - 1:0] src_tags_n;

      
	reg [src_tags_p - 1:0] src_tags_v_r;

      
	reg [src_tags_p - 1:0] src_tags_v_n;

      
	wire entry_v_r;

      
	wire [(src_tags_p * wakeup_tags_p) - 1:0] bypass_n;

      
	always @(posedge clk_i) begin

      
		src_tags_v_r <= src_tags_v_n;

      
		src_tags_r <= src_tags_n;

      
	end

      
	genvar i;

      
	genvar j;

      
	always @(negedge clk_i)

      
		$display("%m: entry we_i=%b src_tags_v_r=%b src_tags_r=%b wakeup_tags_i=%b ready_o=%b", we_i, src_tags_v_r, src_tags_r, wakeup_tags_i, ready_o);

      
	generate

      
		for (i = 0; i < src_tags_p; i = i + 1) begin : rof

      
			for (j = 0; j < wakeup_tags_p; j = j + 1) begin : rof2

      
				assign bypass_n[(i * wakeup_tags_p) + j] = wakeup_tags_v_i[j] & (src_tags_r[i * tag_width_p+:tag_width_p] == wakeup_tags_i[j * tag_width_p+:tag_width_p]);

      
			end

      
			assign bypass_o[i * wakeup_tags_p+:wakeup_tags_p] = bypass_n[i * wakeup_tags_p+:wakeup_tags_p] & {wakeup_tags_p {src_tags_v_r[i]}};

      
			always @(*)

      
				if (we_i)

      
					src_tags_n[i * tag_width_p+:tag_width_p] = src_tags_i[i * tag_width_p+:tag_width_p];

      
				else

      
					src_tags_n[i * tag_width_p+:tag_width_p] = src_tags_r[i * tag_width_p+:tag_width_p];

      
			wire not_matched = ~(|bypass_n[i * wakeup_tags_p+:wakeup_tags_p]);

      
			always @(*)

      
				if (reset_i)

      
					src_tags_v_n[i] = 1'sb0;

      
				else if (we_i)

      
					src_tags_v_n[i] = src_tags_v_i[i];

      
				else

      
					src_tags_v_n[i] = src_tags_v_r[i] & not_matched;

      
		end

      
	endgenerate

      
	assign ready_o = ~(|src_tags_v_r);

      
endmodule

      
module bsg_scheduler_dataflow (

      
	clk_i,

      
	reset_i,

      
	insert_v_i,

      
	insert_src_tags_v_i,

      
	insert_src_tags_i,

      
	insert_src_yumi_o,

      
	insert_allocated_id_one_hot_o,

      
	wakeup_tags_v_i,

      
	wakeup_tags_i,

      
	selected_operation_one_hot_o,

      
	selected_operation_v_o,

      
	selected_bypass_o

      
);

      
	parameter els_p = 0;

      
	parameter src_tags_p = 0;

      
	parameter wakeup_tags_p = 0;

      
	parameter tag_width_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input insert_v_i;

      
	input [src_tags_p - 1:0] insert_src_tags_v_i;

      
	input [(src_tags_p * tag_width_p) - 1:0] insert_src_tags_i;

      
	output wire insert_src_yumi_o;

      
	output wire [els_p - 1:0] insert_allocated_id_one_hot_o;

      
	input [wakeup_tags_p - 1:0] wakeup_tags_v_i;

      
	input [(wakeup_tags_p * tag_width_p) - 1:0] wakeup_tags_i;

      
	output wire [els_p - 1:0] selected_operation_one_hot_o;

      
	output wire selected_operation_v_o;

      
	output wire [(src_tags_p * wakeup_tags_p) - 1:0] selected_bypass_o;

      
	genvar i;

      
	wire [els_p - 1:0] readies_lo;

      
	wire [els_p - 1:0] active_r_lo;

      
	wire [((els_p * src_tags_p) * wakeup_tags_p) - 1:0] selected_bypass_lo;

      
	generate

      
		for (i = 0; i < els_p; i = i + 1) begin : entry

      
			bsg_scheduler_dataflow_entry #(

      
				.tag_width_p(tag_width_p),

      
				.src_tags_p(src_tags_p),

      
				.wakeup_tags_p(wakeup_tags_p)

      
			) entry(

      
				.clk_i(clk_i),

      
				.reset_i(reset_i),

      
				.we_i(insert_allocated_id_one_hot_o[i] & v_i),

      
				.src_tags_v_i(insert_src_tags_v_i),

      
				.src_tags_i(insert_src_tags_i),

      
				.wakeup_tags_v_i(wakeup_tags_v_i),

      
				.wakeup_tags_i(wakeup_tags_i),

      
				.ready_o(readies_lo[i]),

      
				.bypass_o(selected_bypass_lo[wakeup_tags_p * (i * src_tags_p)+:wakeup_tags_p * src_tags_p])

      
			);

      
		end

      
	endgenerate

      
	wire [els_p - 1:0] ready_to_issue = readies_lo & active_r_lo;

      
	always @(negedge clk_i)

      
		$display("reset_i=%b insert_v_i=%b insert_src_tags_v_i=%b insert_src_tags_i=%b insert_src_yumi=%b insert_allocated_id_one_hot_o=%b wakeup_tags_v_i=%b wakeup_tags_i=%b selected_operation_one_hot_o=%b select_operation_v_o=%b readies_lo=%b, active_r_lo=%b ready_to_issue=%b bypass_lo=%b", reset_i, insert_v_i, insert_src_tags_v_i, insert_src_tags_i, insert_src_yumi_o, insert_allocated_id_one_hot_o, wakeup_tags_v_i, wakeup_tags_i, selected_operation_one_hot_o, selected_operation_v_o, readies_lo, active_r_lo, ready_to_issue, bypass_lo);

      
	bsg_priority_encode_one_hot_out #(

      
		.width_p(els_p),

      
		.lo_to_hi_p(1)

      
	) pe0(

      
		.i(ready_to_issue),

      
		.o(selected_operation_one_hot_o),

      
		.v_o(selected_operation_v_o)

      
	);

      
	wire alloc_id_v_lo;

      
	wire src_yumi_o;

      
	bsg_id_pool_dealloc_alloc_one_hot #(.els_p(els_p)) idpool(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.active_ids_r_o(active_r_lo),

      
		.alloc_id_one_hot_o(insert_allocated_id_one_hot_o),

      
		.alloc_id_v_o(alloc_id_v_lo),

      
		.alloc_yumi_i(src_yumi_o),

      
		.dealloc_ids_i(selected_operation_one_hot_o)

      
	);

      
	assign insert_src_yumi_o = alloc_id_v_lo & insert_v_i;

      
	bsg_mux_one_hot #(

      
		.width_p(src_tags_p * wakeup_tags_p),

      
		.els_p(els_p)

      
	) bmoh(

      
		.data_i(selected_bypass_lo),

      
		.sel_one_hot_i(selected_operation_one_hot_o),

      
		.data_o(selected_bypass_o)

      
	);

      
endmodule
