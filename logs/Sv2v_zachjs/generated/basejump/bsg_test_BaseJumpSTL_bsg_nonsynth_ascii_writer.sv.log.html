module bsg_nonsynth_ascii_writer (

      
	clk,

      
	reset_i,

      
	valid_i,

      
	data_i

      
);

      
	parameter width_p = 0;

      
	parameter values_p = 0;

      
	parameter filename_p = 0;

      
	parameter fopen_param_p = "w";

      
	parameter format_p = "%x ";

      
	input clk;

      
	input reset_i;

      
	input valid_i;

      
	input [(width_p * values_p) - 1:0] data_i;

      
	integer file = -1;

      
	integer i;

      
	always @(posedge reset_i)

      
		if (file == -1)

      
			file = $fopen(filename_p, fopen_param_p);

      
	reg [(width_p * values_p) - 1:0] capture_data_r;

      
	reg capture_valid_r;

      
	always @(posedge clk) begin

      
		capture_data_r <= data_i;

      
		capture_valid_r <= valid_i;

      
	end

      
	always @(negedge clk)

      
		if (capture_valid_r && (file != -1)) begin

      
			for (i = 0; i < values_p; i = i + 1)

      
				$fwrite(file, format_p, capture_data_r[i * width_p+:width_p]);

      
			$fwrite(file, "\n");

      
		end

      
endmodule
