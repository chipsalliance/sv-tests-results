module bsg_front_side_bus_hop_out (

      
	clk_i,

      
	reset_i,

      
	v_i,

      
	data_i,

      
	ready_o,

      
	yumi_o,

      
	v_o,

      
	data_o,

      
	ready_i

      
);

      
	parameter width_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input [1:0] v_i;

      
	input [(2 * width_p) - 1:0] data_i;

      
	output wire ready_o;

      
	output wire yumi_o;

      
	output wire v_o;

      
	output wire [width_p - 1:0] data_o;

      
	input ready_i;

      
	wire fifo_v;

      
	wire fifo_ready;

      
	reg v1_blocked_r;

      
	wire source_sel = ~v_i[0] | v1_blocked_r;

      
	wire yumi_o_tmp = (fifo_ready & v_i[1]) & source_sel;

      
	assign yumi_o = yumi_o_tmp;

      
	always @(posedge clk_i) v1_blocked_r <= (reset_i ? 1'b0 : (fifo_ready ? v_i[1] & ~source_sel : v1_blocked_r));

      
	bsg_two_fifo #(.width_p(width_p)) fifo(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.data_i(data_i[source_sel * width_p+:width_p]),

      
		.data_o(data_o),

      
		.v_o(fifo_v),

      
		.yumi_i(fifo_v & ready_i),

      
		.ready_o(fifo_ready),

      
		.v_i(|v_i)

      
	);

      
	assign v_o = fifo_v;

      
	assign ready_o = fifo_ready & ~v1_blocked_r;

      
endmodule
