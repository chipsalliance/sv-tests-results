module bsg_tag_client_unsync (

      
	bsg_tag_i,

      
	data_async_r_o

      
);

      
	parameter width_p = 0;

      
	parameter harden_p = 1;

      
	parameter debug_level_lp = 0;

      
	input wire [3:0] bsg_tag_i;

      
	output wire [width_p - 1:0] data_async_r_o;

      
	reg op_r;

      
	reg param_r;

      
	always @(posedge bsg_tag_i[3]) begin

      
		op_r <= bsg_tag_i[2];

      
		param_r <= bsg_tag_i[1];

      
	end

      
	wire shift_op = op_r;

      
	wire no_op = ~op_r & ~param_r;

      
	wire [width_p - 1:0] tag_data_r;

      
	wire [width_p - 1:0] tag_data_n;

      
	wire [width_p - 1:0] tag_data_shift;

      
	generate

      
		if (width_p > 1) begin : fi

      
			assign tag_data_shift = {param_r, tag_data_r[width_p - 1:1]};

      
		end

      
		else begin : fi

      
			assign tag_data_shift = param_r;

      
		end

      
	endgenerate

      
	bsg_mux2_gatestack #(

      
		.width_p(width_p),

      
		.harden_p(harden_p)

      
	) tag_data_mux(

      
		.i0(tag_data_r),

      
		.i1(tag_data_shift),

      
		.i2({width_p {shift_op}}),

      
		.o(tag_data_n)

      
	);

      
	bsg_dff #(

      
		.width_p(width_p),

      
		.harden_p(harden_p)

      
	) tag_data_reg(

      
		.clk_i(bsg_tag_i[3]),

      
		.data_i(tag_data_n),

      
		.data_o(tag_data_r)

      
	);

      
	generate

      
		if (debug_level_lp > 1) begin : debug

      
			wire reset_op = ~op_r & param_r;

      
			always @(negedge bsg_tag_i[3]) begin

      
				if (reset_op & ~(~bsg_tag_i[2] & bsg_tag_i[1]))

      
					$display("## bsg_tag_client RESET DEASSERTED time %t (%m)", $time);

      
				if (~reset_op & (~bsg_tag_i[2] & bsg_tag_i[1]))

      
					$display("## bsg_tag_client RESET ASSERTED time   %t  (%m)", $time);

      
				if (shift_op)

      
					$display("## bsg_tag_client (send) SHIFTING  %b (%m)", tag_data_r);

      
			end

      
		end

      
	endgenerate

      
	assign data_async_r_o = tag_data_r;

      
endmodule
