module bsg_dly_line (

      
	async_reset_i,

      
	bsg_tag_i,

      
	bsg_tag_trigger_i,

      
	clk_i,

      
	clk_o

      
);

      
	parameter num_adgs_p = 1;

      
	input async_reset_i;

      
	input wire [3:0] bsg_tag_i;

      
	input wire [3:0] bsg_tag_trigger_i;

      
	input clk_i;

      
	output reg clk_o;

      
	localparam osc_base_delay_lp = 1000;

      
	localparam osc_granularity_lp = 100;

      
	wire [num_adgs_p + 3:0] fb_tag_r;

      
	wire fb_we_r;

      
	bsg_tag_client #(

      
		.width_p(((num_adgs_p + 3) >= 0 ? num_adgs_p + 4 : 1 - (num_adgs_p + 3))),

      
		.harden_p(1)

      
	) btc(

      
		.bsg_tag_i(bsg_tag_i),

      
		.recv_clk_i(clk_o),

      
		.recv_new_r_o(fb_we_r),

      
		.recv_data_r_o(fb_tag_r)

      
	);

      
	wire [1:0] cdt = fb_tag_r[3-:2];

      
	wire [1:0] fdt = fb_tag_r[1-:2];

      
	wire [num_adgs_p - 1:0] adg_ctrl = fb_tag_r[num_adgs_p + 3-:((num_adgs_p + 3) >= 4 ? num_adgs_p + 0 : 5 - (num_adgs_p + 3))];

      
	reg [(4 + num_adgs_p) - 1:0] ctrl_rrr;

      
	always @(clk_o or async_reset_i)

      
		if (async_reset_i)

      
			ctrl_rrr <= 1'sb0;

      
		else if (fb_we_r)

      
			ctrl_rrr <= {adg_ctrl, cdt, fdt};

      
	always begin

      
		#(osc_base_delay_lp)

      
			;

      
		if (ctrl_rrr !== {4 + num_adgs_p {1'sbx}})

      
			#(((1 << (4 + num_adgs_p)) - ctrl_rrr) * osc_granularity_lp) clk_o <= clk_i | async_reset_i;

      
	end

      
endmodule
