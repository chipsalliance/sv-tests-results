				ldst_tick <= 1;

      
		end

      
		else if (((cstate == 2'd3) && (ldst_tick != 0)) && push)

      
			ldst_tick <= ldst_tick - 1;

      
	function automatic signed [3:0] sv2v_cast_4_signed;

      
		input reg signed [3:0] inp;

      
		sv2v_cast_4_signed = inp;

      
	endfunction

      
	always @(*) begin

      
		push = 1'b0;

      
		cmd_sfifo_wdata = {24 {1'b1}};

      
		case (cstate)

      
			2'd1: begin

      
				push = cmd_sfifo_ready;

      
				case (init_tick)

      
					'd4: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0010;

      
						cmd_sfifo_wdata[15-:16] = 16'h0400;

      
					end

      
					'd3: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0001;

      
						cmd_sfifo_wdata[15-:16] = 16'h0000;

      
					end

      
					'd2: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0001;

      
						cmd_sfifo_wdata[15-:16] = 16'h0000;

      
					end

      
					'd1: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0000;

      
						cmd_sfifo_wdata[15-:16] = {8'h00, dmc_p_i[38-:4], sv2v_cast_4_signed($clog2(dfi_burst_length_lp << 1))};

      
						cmd_sfifo_wdata[18-:3] = 4'h0;

      
					end

      
					'd0: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0000;

      
						cmd_sfifo_wdata[15-:16] = 16'h0000;

      
						cmd_sfifo_wdata[18-:3] = 4'h2;

      
					end

      
					default: cmd_sfifo_wdata[23-:4] = 4'b1xxx;

      
				endcase

      
			end

      
			2'd2: begin

      
				push = cmd_sfifo_ready;

      
				case (refr_tick)

      
					'd1: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0010;

      
						cmd_sfifo_wdata[15-:16] = 16'h0400;

      
					end

      
					'd0: cmd_sfifo_wdata[23-:4] = 4'b0001;

      
				endcase

      
			end

      
			2'd3: begin

      
				push = cmd_sfifo_ready;

      
				case (ldst_tick)

      
					'd2: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0010;

      
						cmd_sfifo_wdata[18-:3] = bank_addr;

      
						cmd_sfifo_wdata[15-:16] = open_row[bank_addr];

      
					end

      
					'd1: begin

      
						cmd_sfifo_wdata[23-:4] = 4'b0011;

      
						cmd_sfifo_wdata[18-:3] = bank_addr;

      
						cmd_sfifo_wdata[15-:16] = row_addr;

      
					end

      
					'd0: begin

      
						cmd_sfifo_wdata[18-:3] = bank_addr;

      
						cmd_sfifo_wdata[15-:16] = {col_addr[14:10], ap, col_addr[9:0]};

      
						if (cmd_afifo_rdata[ui_addr_width_p + 0])

      
							cmd_sfifo_wdata[23-:4] = 4'b0101;

      
						else

      
							cmd_sfifo_wdata[23-:4] = 4'b0100;

      
					end

      
				endcase

      
			end

      
		endcase

      
	end

      
	always @(*) begin

      
		nstate = 2'd0;

      
		case (cstate)

      
			2'd0:

      
				if (!init_done)

      
					nstate = 2'd1;

      
				else if (refr_req)

      
					nstate = 2'd2;

      
				else if (cmd_afifo_rvalid)

      
					nstate = 2'd3;

      
				else

      
					nstate = cstate;

      
			2'd1:

      
				if ((init_tick == 0) && push)

      
					nstate = 2'd0;

      
				else

      
					nstate = cstate;

      
			2'd2:

      
				if ((refr_tick == 0) && push)

      
					nstate = 2'd0;

      
				else

      
					nstate = cstate;

      
			2'd3:

      
				if ((ldst_tick == 0) && push)

      
					nstate = 2'd0;

      
				else

      
					nstate = cstate;

      
			default: nstate = 2'd0;

      
		endcase

      
	end

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			cstate <= 2'd0;

      
		else

      
			cstate <= nstate;

      
	assign cmd_sfifo_winc = push;

      
	assign cmd_sfifo_rinc = shoot;

      
	bsg_fifo_1r1w_small #(

      
		.width_p(24),

      
		.els_p(cmd_sfifo_depth_p),

      
		.ready_THEN_valid_p(1)

      
	) cmd_sfifo(

      
		.clk_i(dfi_clk_i),

      
		.reset_i(dfi_clk_sync_rst_i),

      
		.v_i(cmd_sfifo_winc),

      
		.ready_o(cmd_sfifo_ready),

      
		.data_i(cmd_sfifo_wdata),

      
		.v_o(cmd_sfifo_valid),

      
		.data_o(cmd_sfifo_rdata),

      
		.yumi_i(cmd_sfifo_rinc)

      
	);

      
	always @(*)

      
		if (cmd_sfifo_valid)

      
			case (c_cmd)

      
				4'b0000: shoot = cmd_tick >= dmc_p_i[78-:4];

      
				4'b0001: shoot = cmd_tick >= dmc_p_i[74-:4];

      
				4'b0010: shoot = (n_cmd == 4'b0011 ? (cmd_tick >= dmc_p_i[66-:4]) && (cmd_act_tick >= dmc_p_i[62-:4]) : cmd_tick >= dmc_p_i[66-:4]);

      
				4'b0011:

      
					case (n_cmd)

      
						4'b0010: shoot = cmd_tick >= dmc_p_i[62-:4];

      
						4'b0011: shoot = cmd_tick >= dmc_p_i[58-:4];

      
						4'b0100: shoot = ((cmd_tick >= dmc_p_i[54-:4]) & (cmd_rd_tick >= ((dmc_p_i[38-:4] + dfi_burst_length_lp) - 1))) & &tx_sipo_valid_lo;

      
						4'b0101: shoot = (cmd_tick >= dmc_p_i[54-:4]) & (cmd_wr_tick >= dmc_p_i[46-:4]);

      
						default: shoot = 1'b1;

      
					endcase

      
				4'b0100:

      
					case (n_cmd)

      
						4'b0010: shoot = (cmd_tick >= dmc_p_i[50-:4]) & (cmd_act_tick >= dmc_p_i[62-:4]);

      
						4'b0100: shoot = (cmd_tick >= (dfi_burst_length_lp - 1)) & &tx_sipo_valid_lo;

      
						4'b0101: shoot = cmd_tick >= dmc_p_i[46-:4];

      
						4'b0011: shoot = (cmd_act_tick >= dmc_p_i[70-:4]) & (cmd_tick >= (dmc_p_i[50-:4] + dmc_p_i[66-:4]));

      
						default: shoot = 1'b1;

      
					endcase

      
				4'b0101:

      
					case (n_cmd)

      
						4'b0010: shoot = (cmd_tick >= dmc_p_i[42-:4]) & (cmd_act_tick >= dmc_p_i[62-:4]);

      
						4'b0100: shoot = (cmd_tick >= ((dmc_p_i[38-:4] + dfi_burst_length_lp) - 1)) & &tx_sipo_valid_lo;

      
						4'b0101: shoot = cmd_tick >= (dfi_burst_length_lp - 1);

      
						4'b0011: shoot = (cmd_act_tick >= dmc_p_i[70-:4]) & (cmd_tick >= (dmc_p_i[42-:4] + dmc_p_i[66-:4]));

      
						default: shoot = 1'b1;

      
					endcase

      
				default: shoot = 1'b1;

      
			endcase

      
		else

      
			shoot = 1'b0;

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			cmd_tick <= 0;

      
		else if (shoot)

      
			cmd_tick <= 0;

      
		else if (cmd_tick != 8'h0f)

      
			cmd_tick <= cmd_tick + 1;

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			cmd_act_tick <= 0;

      
		else if (shoot && (n_cmd == 4'b0011))

      
			cmd_act_tick <= 0;

      
		else if (cmd_act_tick != 8'h0f)

      
			cmd_act_tick <= cmd_act_tick + 1;

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			cmd_wr_tick <= 0;

      
		else if (shoot && (n_cmd == 4'b0100))

      
			cmd_wr_tick <= 0;

      
		else if (cmd_wr_tick != 8'h0f)

      
			cmd_wr_tick <= cmd_wr_tick + 1;

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			cmd_rd_tick <= 0;

      
		else if (shoot && (n_cmd == 4'b0101))

      
			cmd_rd_tick <= 0;

      
		else if (cmd_rd_tick != 8'h0f)

      
			cmd_rd_tick <= cmd_rd_tick + 1;

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			c_cmd <= 4'b0111;

      
		else if (shoot)

      
			c_cmd <= n_cmd;

      
	assign n_cmd = cmd_sfifo_rdata[23-:4];

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i) begin

      
			cwd_tick <= 0;

      
			cwd_valid <= 0;

      
		end

      
		else if (shoot && (cmd_sfifo_rdata[23:20] == 4'b0100)) begin

      
			cwd_tick <= dmc_p_i[38-:4] - 2;

      
			cwd_valid <= 1;

      
		end

      
		else if (cwd_valid) begin

      
			cwd_tick <= cwd_tick - 1;

      
			if (cwd_tick == 0)

      
				cwd_valid <= 0;

      
		end

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i) begin

      
			wburst_tick <= 0;

      
			wburst_valid <= 0;

      
		end

      
		else if (shoot && (cmd_sfifo_rdata[23:20] == 4'b0100)) begin

      
			wburst_tick <= dfi_burst_length_lp - 1;

      
			wburst_valid <= 1;

      
		end

      
		else if (wburst_valid) begin

      
			wburst_tick <= wburst_tick - 1;

      
			if (wburst_tick == 0)

      
				wburst_valid <= 0;

      
		end

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i) begin

      
			cas_tick <= 0;

      
			cas_valid <= 0;

      
		end

      
		else if (shoot && (cmd_sfifo_rdata[23:20] == 4'b0101)) begin

      
			cas_tick <= dmc_p_i[38-:4] - 1;

      
			cas_valid <= 1;

      
		end

      
		else if (cas_valid) begin

      
			cas_tick <= cas_tick - 1;

      
			if (cas_tick == 0)

      
				cas_valid <= 0;

      
		end

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i) begin

      
			rburst_tick <= 0;

      
			dfi_rddata_en_o <= 0;

      
		end

      
		else if (cas_valid && (cas_tick == 0)) begin

      
			rburst_tick <= dfi_burst_length_lp - 1;

      
			dfi_rddata_en_o <= 1;

      
		end

      
		else if (dfi_rddata_en_o) begin

      
			rburst_tick <= rburst_tick - 1;

      
			if (rburst_tick == 0)

      
				dfi_rddata_en_o <= 0;

      
		end

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i) begin

      
			dfi_bank_o <= 3'b000;

      
			dfi_address_o <= 16'h0000;

      
			{dfi_cs_n_o, dfi_ras_n_o, dfi_cas_n_o, dfi_we_n_o} <= 4'b1111;

      
			dfi_reset_n_o <= 1'b1;

      
			dfi_odt_o <= 1'b1;

      
		end

      
		else if (shoot) begin

      
			dfi_bank_o <= cmd_sfifo_rdata[18:16];

      
			dfi_address_o <= cmd_sfifo_rdata[15:0];

      
			{dfi_cs_n_o, dfi_ras_n_o, dfi_cas_n_o, dfi_we_n_o} <= n_cmd;

      
		end

      
		else begin

      
			{dfi_cs_n_o, dfi_ras_n_o, dfi_cas_n_o, dfi_we_n_o} <= 4'b1111;

      
			dfi_reset_n_o <= 1'b1;

      
			dfi_odt_o <= 1'b1;

      
		end

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			dfi_cke_o <= 1'b0;

      
		else

      
			dfi_cke_o <= 1'b1;

      
	always @(posedge dfi_clk_i)

      
		if (dfi_clk_sync_rst_i)

      
			open_bank <= 0;

      
		else if (cmd_sfifo_winc)

      
			case (cmd_sfifo_wdata[23:20])

      
				4'b0011: begin

      
					open_bank[cmd_sfifo_wdata[18:16]] <= 1'b1;

      
					open_row[cmd_sfifo_wdata[18:16]] <= cmd_sfifo_wdata[15:0];

      
				end

      
				4'b0100, 4'b0101: open_bank[cmd_sfifo_wdata[18:16]] <= ~ap;

      
				4'b0010:

      
					if (cmd_sfifo_wdata[10])

      
						open_bank <= 0;

      
					else

      
						open_bank[cmd_sfifo_wdata[18:16]] <= 1'b0;

      
			endcase

      
	generate

      
		for (k = 0; k < ui_burst_length_lp; k = k + 1) begin : tx_flatten

      
			assign tx_data[k * ui_data_width_p+:ui_data_width_p] = tx_sipo_data_lo[k * (ui_mask_width_lp + ui_data_width_p)+:ui_data_width_p];

      
			assign tx_mask[k * ui_mask_width_lp+:ui_mask_width_lp] = tx_sipo_data_lo[(k * (ui_mask_width_lp + ui_data_width_p)) + ui_data_width_p+:ui_mask_width_lp];

      
		end

      
		for (k = 0; k < dfi_burst_length_lp; k = k + 1) begin : tx_make

      
			always @(posedge dfi_clk_i) begin

      
				tx_data_piso_data_li[k * dfi_data_width_p+:dfi_data_width_p] <= tx_data[k * dfi_data_width_p+:dfi_data_width_p];

      
				tx_mask_piso_data_li[k * dfi_mask_width_lp+:dfi_mask_width_lp] <= tx_mask[k * dfi_mask_width_lp+:dfi_mask_width_lp];

      
			end

      
		end

      
	endgenerate

      
	assign tx_data_piso_valid_li = wburst_valid;

      
	assign tx_data_piso_yumi_li = tx_data_piso_valid_lo;

      
	assign tx_mask_piso_valid_li = wburst_valid;

      
	assign tx_mask_piso_yumi_li = tx_mask_piso_valid_lo;

      
	bsg_parallel_in_serial_out #(

      
		.width_p(dfi_data_width_p),

      
		.els_p(dfi_burst_length_lp)

      
	) tx_data_piso(

      
		.clk_i(dfi_clk_i),

      
		.reset_i(dfi_clk_sync_rst_i),

      
		.valid_i(tx_data_piso_valid_li),

      
		.data_i(tx_data_piso_data_li),

      
		.ready_and_o(tx_data_piso_ready_lo),

      
		.valid_o(tx_data_piso_valid_lo),

      
		.data_o(tx_data_piso_data_lo),

      
		.yumi_i(tx_data_piso_yumi_li)

      
	);

      
	bsg_parallel_in_serial_out #(

      
		.width_p(dfi_mask_width_lp),

      
		.els_p(dfi_burst_length_lp)

      
	) tx_mask_piso(

      
		.clk_i(dfi_clk_i),

      
		.reset_i(dfi_clk_sync_rst_i),

      
		.valid_i(tx_mask_piso_valid_li),

      
		.data_i(tx_mask_piso_data_li),

      
		.ready_and_o(tx_mask_piso_ready_lo),

      
		.valid_o(tx_mask_piso_valid_lo),

      
		.data_o(tx_mask_piso_data_lo),

      
		.yumi_i(tx_mask_piso_yumi_li)

      
	);

      
	assign dfi_wrdata_o = tx_data_piso_data_lo;

      
	assign dfi_wrdata_en_o = tx_data_piso_valid_lo;

      
	assign dfi_wrdata_mask_o = tx_mask_piso_data_lo;

      
	assign rddata_afifo_wclk = dfi_clk_i;

      
	assign rddata_afifo_wrst = dfi_clk_sync_rst_i;

      
	assign rddata_afifo_winc = dfi_rddata_valid_i;

      
	assign rddata_afifo_wdata = dfi_rddata_i;

      
	assign rddata_afifo_rclk = ui_clk_i;

      
	assign rddata_afifo_rrst = ui_clk_sync_rst_i;

      
	assign rddata_afifo_rinc = rx_sipo_ready_lo && rddata_afifo_rvalid;

      
	bsg_async_fifo #(

      
		.width_p(dfi_data_width_p),

      
		.lg_size_p($clog2(cmd_afifo_depth_p * dfi_burst_length_lp))

      
	) rddata_afifo(

      
		.r_data_o(rddata_afifo_rdata),

      
		.w_full_o(rddata_afifo_wfull),

      
		.r_valid_o(rddata_afifo_rvalid),

      
		.w_data_i(rddata_afifo_wdata),

      
		.w_enq_i(rddata_afifo_winc),

      
		.w_clk_i(rddata_afifo_wclk),

      
		.w_reset_i(rddata_afifo_wrst),

      
		.r_deq_i(rddata_afifo_rinc),

      
		.r_clk_i(rddata_afifo_rclk),

      
		.r_reset_i(rddata_afifo_rrst)

      
	);

      
	assign rx_sipo_valid_li = rddata_afifo_rvalid;

      
	assign rx_sipo_data_li = rddata_afifo_rdata;

      
	function automatic signed [(($clog2(dfi_burst_length_lp) + 0) >= 0 ? $clog2(dfi_burst_length_lp) + 1 : 1 - ($clog2(dfi_burst_length_lp) + 0)) - 1:0] sv2v_cast_9529F_signed;

      
		input reg signed [(($clog2(dfi_burst_length_lp) + 0) >= 0 ? $clog2(dfi_burst_length_lp) + 1 : 1 - ($clog2(dfi_burst_length_lp) + 0)) - 1:0] inp;

      
		sv2v_cast_9529F_signed = inp;

      
	endfunction

      
	assign rx_sipo_yumi_cnt_li = sv2v_cast_9529F_signed((&rx_sipo_valid_lo ? dfi_burst_length_lp : 0));

      
	bsg_serial_in_parallel_out #(

      
		.width_p(dfi_data_width_p),

      
		.els_p(dfi_burst_length_lp)

      
	) rx_sipo(

      
		.clk_i(ui_clk_i),

      
		.reset_i(ui_clk_sync_rst_i),

      
		.valid_i(rx_sipo_valid_li),

      
		.data_i(rx_sipo_data_li),

      
		.ready_o(rx_sipo_ready_lo),

      
		.valid_o(rx_sipo_valid_lo),

      
		.data_o(rx_sipo_data_lo),

      
		.yumi_cnt_i(rx_sipo_yumi_cnt_li)

      
	);

      
	generate

      
		for (k = 0; k < dfi_burst_length_lp; k = k + 1) begin : rx_flatten

      
			assign rx_data[k * dfi_data_width_p+:dfi_data_width_p] = rx_sipo_data_lo[k * dfi_data_width_p+:dfi_data_width_p];

      
		end

      
		for (k = 0; k < ui_burst_length_lp; k = k + 1) begin : rx_make

      
			assign rx_piso_data_li[k * ui_data_width_p+:ui_data_width_p] = rx_data[k * ui_data_width_p+:ui_data_width_p];

      
		end

      
	endgenerate

      
	assign rx_piso_valid_li = &rx_sipo_valid_lo;

      
	assign rx_piso_yumi_li = app_rd_data_valid_o;

      
	bsg_parallel_in_serial_out #(

      
		.width_p(ui_data_width_p),

      
		.els_p(ui_burst_length_lp)

      
	) rx_piso(

      
		.clk_i(ui_clk_i),

      
		.reset_i(ui_clk_sync_rst_i),

      
		.valid_i(rx_piso_valid_li),

      
		.data_i(rx_piso_data_li),

      
		.ready_and_o(rx_piso_ready_lo),

      
		.valid_o(rx_piso_valid_lo),

      
		.data_o(rx_piso_data_lo),

      
		.yumi_i(rx_piso_yumi_li)

      
	);

      
	reg [7:0] rd_cnt;

      
	always @(posedge ui_clk_i)

      
		if (ui_clk_sync_rst_i)

      
			rd_cnt <= 0;

      
		else if (rx_piso_yumi_li)

      
			if (rd_cnt == (ui_burst_length_lp - 1))

      
				rd_cnt <= 1'sb0;

      
			else

      
				rd_cnt <= rd_cnt + 1;

      
	assign app_rd_data_valid_o = rx_piso_valid_lo;

      
	assign app_rd_data_o = rx_piso_data_lo;

      
	assign app_rd_data_end_o = rx_piso_valid_lo && (rd_cnt == (ui_burst_length_lp - 1));

      
endmodule
