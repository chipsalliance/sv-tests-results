module bsg_clk_gen_osc_v3 (

      
	async_reset_i,

      
	bsg_tag_i,

      
	bsg_tag_trigger_i,

      
	clk_o

      
);

      
	parameter num_taps_p = 2;

      
	input async_reset_i;

      
	input wire [2:0] bsg_tag_i;

      
	input wire [2:0] bsg_tag_trigger_i;

      
	output reg clk_o;

      
	localparam ctl_width_lp = ((num_taps_p == 1) || (num_taps_p == 0) ? 1 : $clog2(num_taps_p));

      
	wire [ctl_width_lp - 1:0] fb_tag_r;

      
	bsg_tag_client_unsync #(

      
		.width_p(ctl_width_lp),

      
		.harden_p(0)

      
	) btc(

      
		.bsg_tag_i(bsg_tag_i),

      
		.data_async_r_o(fb_tag_r)

      
	);

      
	wire trig_r;

      
	bsg_tag_client_unsync #(

      
		.width_p(1),

      
		.harden_p(0)

      
	) btc_trigger(

      
		.bsg_tag_i(bsg_tag_trigger_i),

      
		.data_async_r_o(trig_r)

      
	);

      
	reg [ctl_width_lp - 1:0] ctrl_rrr;

      
	always @(clk_o or async_reset_i)

      
		if (async_reset_i)

      
			ctrl_rrr <= 1'sb0;

      
		else if (trig_r)

      
			ctrl_rrr <= fb_tag_r;

      
	always #(1000)

      
		if (ctrl_rrr !== {ctl_width_lp {1'sbx}})

      
			#(((1 << ctl_width_lp) - ctrl_rrr) * 100) clk_o <= ~(clk_o | async_reset_i);

      
endmodule
