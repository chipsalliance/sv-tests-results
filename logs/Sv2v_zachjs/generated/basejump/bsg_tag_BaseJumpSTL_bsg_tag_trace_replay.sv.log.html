module bsg_tag_trace_replay (

      
	clk_i,

      
	reset_i,

      
	en_i,

      
	rom_addr_o,

      
	rom_data_i,

      
	valid_i,

      
	data_i,

      
	ready_o,

      
	valid_o,

      
	en_r_o,

      
	tag_data_o,

      
	yumi_i,

      
	done_o,

      
	error_o

      
);

      
	parameter rom_addr_width_p = 0;

      
	parameter rom_data_width_p = 0;

      
	parameter num_masters_p = 0;

      
	parameter num_clients_p = 0;

      
	parameter max_payload_width_p = 0;

      
	parameter uptime_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input en_i;

      
	output wire [rom_addr_width_p - 1:0] rom_addr_o;

      
	input [rom_data_width_p - 1:0] rom_data_i;

      
	input valid_i;

      
	input [max_payload_width_p - 1:0] data_i;

      
	output wire ready_o;

      
	output wire valid_o;

      
	output wire [(1 > num_masters_p ? 1 : num_masters_p) - 1:0] en_r_o;

      
	output wire tag_data_o;

      
	input yumi_i;

      
	output wire done_o;

      
	output wire error_o;

      
	localparam trace_ring_width_lp = rom_data_width_p - 4;

      
	localparam lg_max_payload_width_lp = (((max_payload_width_p + 1) == 1) || ((max_payload_width_p + 1) == 0) ? 1 : $clog2(max_payload_width_p + 1));

      
	localparam bsg_tag_header_width_lp = (((num_clients_p == 1) || (num_clients_p == 0) ? 1 : $clog2(num_clients_p)) + 1) + lg_max_payload_width_lp;

      
	wire tr_valid_lo;

      
	wire [trace_ring_width_lp - 1:0] tr_data_lo;

      
	wire tr_yumi_li;

      
	function automatic [trace_ring_width_lp - 1:0] sv2v_cast_EA130;

      
		input reg [trace_ring_width_lp - 1:0] inp;

      
		sv2v_cast_EA130 = inp;

      
	endfunction

      
	bsg_trace_replay #(

      
		.payload_width_p(trace_ring_width_lp),

      
		.rom_addr_width_p(rom_addr_width_p),

      
		.debug_p(0)

      
	) trace_replay(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.en_i(en_i),

      
		.v_i(valid_i),

      
		.data_i(sv2v_cast_EA130(data_i)),

      
		.ready_o(ready_o),

      
		.v_o(tr_valid_lo),

      
		.data_o(tr_data_lo),

      
		.yumi_i(tr_yumi_li),

      
		.rom_addr_o(rom_addr_o),

      
		.rom_data_i(rom_data_i),

      
		.done_o(done_o),

      
		.error_o(error_o)

      
	);

      
	wire [bsg_tag_header_width_lp - 1:0] header_n = tr_data_lo[max_payload_width_p+:bsg_tag_header_width_lp];

      
	wire [max_payload_width_p - 1:0] payload_n = tr_data_lo[0+:max_payload_width_p];

      
	wire [trace_ring_width_lp + 1:0] data_n = {1'b0, payload_n, header_n, 1'b1};

      
	wire piso_ready_lo;

      
	assign tr_yumi_li = piso_ready_lo & tr_valid_lo;

      
	bsg_parallel_in_serial_out #(

      
		.width_p(1),

      
		.els_p(trace_ring_width_lp + 2)

      
	) trace_piso(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.valid_i(tr_valid_lo),

      
		.data_i(data_n),

      
		.ready_and_o(piso_ready_lo),

      
		.valid_o(valid_o),

      
		.data_o(tag_data_o),

      
		.yumi_i(yumi_i)

      
	);

      
	generate

      
		if (num_masters_p == 0) begin : genblk1

      
			assign en_r_o = 1'bz;

      
		end

      
		else begin : genblk1

      
			bsg_dff_en #(.width_p(num_masters_p)) en_reg(

      
				.clk_i(clk_i),

      
				.en_i(tr_valid_lo & piso_ready_lo),

      
				.data_i(tr_data_lo[max_payload_width_p + bsg_tag_header_width_lp+:num_masters_p]),

      
				.data_o(en_r_o)

      
			);

      
		end

      
	endgenerate

      
endmodule
