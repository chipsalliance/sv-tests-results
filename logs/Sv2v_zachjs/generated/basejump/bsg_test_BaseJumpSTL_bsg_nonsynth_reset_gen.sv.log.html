module bsg_nonsynth_reset_gen (

      
	clk_i,

      
	async_reset_o

      
);

      
	parameter num_clocks_p = 1;

      
	parameter reset_cycles_lo_p = 0;

      
	parameter reset_cycles_hi_p = 0;

      
	input wire [num_clocks_p - 1:0] clk_i;

      
	output wire async_reset_o;

      
	genvar i;

      
	reg [(num_clocks_p * $clog2(reset_cycles_lo_p + 1)) - 1:0] ctr_lo_r;

      
	reg [(num_clocks_p * $clog2(reset_cycles_hi_p + 1)) - 1:0] ctr_hi_r;

      
	wire [num_clocks_p - 1:0] phase_lo_r;

      
	wire [num_clocks_p - 1:0] phase_hi_r;

      
	wire in_phase_1 = &phase_lo_r;

      
	wire in_phase_2 = &phase_hi_r;

      
	generate

      
		for (i = 0; i < num_clocks_p; i = i + 1) begin : rof

      
			assign phase_lo_r[i] = ctr_lo_r[i * $clog2(reset_cycles_lo_p + 1)+:$clog2(reset_cycles_lo_p + 1)] == reset_cycles_lo_p;

      
			assign phase_hi_r[i] = ctr_hi_r[i * $clog2(reset_cycles_hi_p + 1)+:$clog2(reset_cycles_hi_p + 1)] == reset_cycles_hi_p;

      
			always @(negedge clk_i[i])

      
				if (~phase_lo_r[i])

      
					ctr_lo_r[i * $clog2(reset_cycles_lo_p + 1)+:$clog2(reset_cycles_lo_p + 1)] <= ctr_lo_r[i * $clog2(reset_cycles_lo_p + 1)+:$clog2(reset_cycles_lo_p + 1)] + 1;

      
				else if (~phase_hi_r[i])

      
					ctr_hi_r[i * $clog2(reset_cycles_hi_p + 1)+:$clog2(reset_cycles_hi_p + 1)] <= ctr_hi_r[i * $clog2(reset_cycles_hi_p + 1)+:$clog2(reset_cycles_hi_p + 1)] + in_phase_1;

      
		end

      
	endgenerate

      
	assign async_reset_o = in_phase_1 ^ in_phase_2;

      
	always @(negedge async_reset_o) begin

      
		$display("__________ ___________  _______________________________");

      
		$display("\\______   \\\\_   _____/ /   _____/\\_   _____/\\__    ___/");

		$display(" |____|_  //_______  //_______  //_______  /  |____|   ");

      
		$display(" ASYNC  \\/         \\/         \\/         \\/            ");

      
	end

      
	always @(posedge async_reset_o) begin

      
		$display("__________ ___________  _______________________________");

      
		$display("\\______   \\\\_   _____/ /   _____/\\_   _____/\\__    ___/");

      
		$display(" |       _/ |    __)_  \\_____  \\  |    __)_   |    |   ");
