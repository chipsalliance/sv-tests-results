module bsg_two_buncher (

      
	clk_i,

      
	reset_i,

      
	data_i,

      
	v_i,

      
	ready_o,

      
	data_o,

      
	v_o,

      
	ready_i

      
);

      
	parameter width_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input [width_p - 1:0] data_i;

      
	input v_i;

      
	output wire ready_o;

      
	output wire [(width_p * 2) - 1:0] data_o;

      
	output wire [1:0] v_o;

      
	input [1:0] ready_i;

      
	reg [width_p - 1:0] data_r;

      
	wire [width_p - 1:0] data_n;

      
	reg data_v_r;

      
	reg data_v_n;

      
	reg data_en;

      
	always @(posedge clk_i)

      
		;

      
	always @(posedge clk_i)

      
		;

      
	always @(posedge clk_i)

      
		if (reset_i)

      
			data_v_r <= 0;

      
		else

      
			data_v_r <= data_v_n;

      
	always @(posedge clk_i)

      
		if (data_en)

      
			data_r <= data_i;

      
	assign v_o = {data_v_r & v_i, data_v_r | v_i};

      
	assign data_o = {data_i, (data_v_r ? data_r : data_i)};

      
	assign ready_o = (ready_i[0] & v_i) | ~data_v_r;

      
	always @(*) begin

      
		data_v_n = data_v_r;

      
		data_en = 1'b0;

      
		if (~data_v_r) begin

      
			if (v_i) begin

      
				data_v_n = ~ready_i[0];

      
				data_en = ~ready_i[0];

      
			end

      
		end

      
		else if (ready_i[0])

      
			if (v_i) begin

      
				data_v_n = ~ready_i[1];

      
				data_en = ~ready_i[1];

      
			end

      
			else

      
				data_v_n = 1'b0;

      
	end

      
endmodule
