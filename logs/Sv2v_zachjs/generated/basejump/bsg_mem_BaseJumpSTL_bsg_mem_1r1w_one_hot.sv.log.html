module bsg_mem_1r1w_one_hot (

      
	w_clk_i,

      
	w_reset_i,

      
	w_v_i,

      
	w_data_i,

      
	r_v_i,

      
	r_data_o

      
);

      
	parameter width_p = 0;

      
	parameter els_p = 0;

      
	parameter safe_els_lp = (els_p > 1 ? els_p : 1);

      
	input w_clk_i;

      
	input w_reset_i;

      
	input [safe_els_lp - 1:0] w_v_i;

      
	input [width_p - 1:0] w_data_i;

      
	input [safe_els_lp - 1:0] r_v_i;

      
	output wire [width_p - 1:0] r_data_o;

      
	wire [(safe_els_lp * width_p) - 1:0] data_r;

      
	wire unused0 = w_reset_i;

      
	genvar i;

      
	generate

      
		for (i = 0; i < els_p; i = i + 1) begin : mem_array

      
			bsg_dff_en #(.width_p(width_p)) mem_reg(

      
				.clk_i(w_clk_i),

      
				.en_i(w_v_i[i]),

      
				.data_i(w_data_i),

      
				.data_o(data_r[i * width_p+:width_p])

      
			);

      
		end

      
	endgenerate

      
	bsg_mux_one_hot #(

      
		.width_p(width_p),

      
		.els_p(safe_els_lp)

      
	) one_hot_sel(

      
		.data_i(data_r),

      
		.sel_one_hot_i(r_v_i),

      
		.data_o(r_data_o)

      
	);

      
	initial if ((width_p * els_p) >= 64)

      
		$display("## %L: instantiating width_p=%d, els_p=%d (%m)", width_p, els_p);

      
	always @(negedge w_clk_i)

      
		;

      
endmodule
