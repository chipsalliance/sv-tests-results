module bsg_ready_to_credit_flow_converter (

      
	clk_i,

      
	reset_i,

      
	v_i,

      
	ready_o,

      
	v_o,

      
	credit_i

      
);

      
	parameter credit_initial_p = 0;

      
	parameter credit_max_val_p = 0;

      
	parameter decimation_p = 1;

      
	parameter ptr_width_lp = $clog2(credit_max_val_p + 1);

      
	input clk_i;

      
	input reset_i;

      
	input v_i;

      
	output wire ready_o;

      
	output wire v_o;

      
	input credit_i;

      
	localparam step_width_lp = $clog2(decimation_p + 1);

      
	wire [step_width_lp - 1:0] up;

      
	wire [step_width_lp - 1:0] down;

      
	wire [ptr_width_lp - 1:0] credit_cnt;

      
	assign ready_o = credit_cnt != 0;

      
	assign v_o = v_i & ready_o;

      
	function automatic [step_width_lp - 1:0] sv2v_cast_443AF;

      
		input reg [step_width_lp - 1:0] inp;

      
		sv2v_cast_443AF = inp;

      
	endfunction

      
	function automatic signed [step_width_lp - 1:0] sv2v_cast_443AF_signed;

      
		input reg signed [step_width_lp - 1:0] inp;

      
		sv2v_cast_443AF_signed = inp;

      
	endfunction

      
	assign up = (credit_i ? sv2v_cast_443AF($unsigned(decimation_p)) : sv2v_cast_443AF_signed(0));

      
	assign down = {{step_width_lp - 1 {1'b0}}, v_o};

      
	bsg_counter_up_down_variable #(

      
		.max_val_p(credit_max_val_p),

      
		.init_val_p(credit_initial_p),

      
		.max_step_p(decimation_p)

      
	) credit_counter(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.up_i(up),

      
		.down_i(down),

      
		.count_o(credit_cnt)

      
	);

      
endmodule
