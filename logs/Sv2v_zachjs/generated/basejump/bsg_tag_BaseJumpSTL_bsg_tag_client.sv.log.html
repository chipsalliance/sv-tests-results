module bsg_tag_client (

      
	bsg_tag_i,

      
	recv_clk_i,

      
	recv_new_r_o,

      
	recv_data_r_o

      
);

      
	parameter width_p = 0;

      
	parameter harden_p = 1;

      
	input wire [2:0] bsg_tag_i;

      
	input recv_clk_i;

      
	output wire recv_new_r_o;

      
	output wire [width_p - 1:0] recv_data_r_o;

      
	localparam debug_level_lp = 1;

      
	reg op_r;

      
	reg op_r_r;

      
	reg param_r;

      
	always @(posedge bsg_tag_i[2]) begin

      
		op_r <= bsg_tag_i[1];

      
		param_r <= bsg_tag_i[0];

      
		op_r_r <= op_r;

      
	end

      
	wire reset_op = ~op_r & param_r;

      
	wire shift_op = op_r;

      
	wire no_op = ~op_r & ~param_r;

      
	wire send_now = op_r_r & no_op;

      
	wire [width_p - 1:0] tag_data_r;

      
	wire [width_p - 1:0] recv_data_r;

      
	wire [width_p - 1:0] tag_data_n;

      
	wire [width_p - 1:0] tag_data_shift;

      
	wire tag_toggle_r;

      
	generate

      
		if (width_p == 1) begin : sb

      
			assign tag_data_shift = {param_r};

      
		end

      
		else begin : mb

      
			assign tag_data_shift = {param_r, tag_data_r[width_p - 1:1]};

      
		end

      
	endgenerate

      
	bsg_mux2_gatestack #(

      
		.width_p(width_p),

      
		.harden_p(harden_p)

      
	) tag_data_mux(

      
		.i0(tag_data_r),

      
		.i1(tag_data_shift),

      
		.i2({width_p {shift_op}}),

      
		.o(tag_data_n)

      
	);

      
	bsg_dff #(

      
		.width_p(width_p),

      
		.harden_p(harden_p)

      
	) tag_data_reg(

      
		.clk_i(bsg_tag_i[2]),

      
		.data_i(tag_data_n),

      
		.data_o(tag_data_r)

      
	);

      
	reg recv_toggle_r;

      
	wire recv_toggle_n;

      
	bsg_launch_sync_sync #(.width_p(1)) blss(

      
		.iclk_i(bsg_tag_i[2]),

      
		.iclk_reset_i(reset_op),

      
		.iclk_data_i(tag_toggle_r ^ send_now),

      
		.iclk_data_o(tag_toggle_r),

      
		.oclk_i(recv_clk_i),

      
		.oclk_data_o(recv_toggle_n)

      
	);

      
	wire recv_new = recv_toggle_r ^ recv_toggle_n;

      
	reg recv_new_r;

      
	reg recv_new_r_r;

      
	always @(posedge recv_clk_i) begin

      
		recv_toggle_r <= recv_toggle_n;

      
		recv_new_r <= recv_new;

      
		recv_new_r_r <= recv_new_r;

      
	end

      
	bsg_dff_en #(

      
		.width_p(width_p),

      
		.harden_p(harden_p)

      
	) recv(

      
		.clk_i(recv_clk_i),

      
		.en_i(recv_new_r),

      
		.data_i(tag_data_r),

      
		.data_o(recv_data_r)

      
	);

      
	assign recv_new_r_o = recv_new_r_r;

      
	assign recv_data_r_o = recv_data_r;

      
endmodule
