module bsg_counter_max_clear_up (

      
	clk_i,

      
	reset_i,

      
	clear_i,

      
	v_i,

      
	data_i,

      
	data_r_o

      
);

      
	parameter max_val_p = 0;

      
	parameter init_val_p = 1'sbx;

      
	parameter ptr_width_lp = $clog2({1'b0, max_val_p} + 1);

      
	input clk_i;

      
	input reset_i;

      
	input clear_i;

      
	input v_i;

      
	input [ptr_width_lp - 1:0] data_i;

      
	output reg [ptr_width_lp - 1:0] data_r_o;

      
	always @(posedge clk_i)

      
		if (reset_i)

      
			data_r_o <= init_val_p;

      
		else if (clear_i)

      
			data_r_o <= (v_i ? data_i : init_val_p);

      
		else if (v_i && (data_i > data_r_o))

      
			data_r_o <= data_i;

      
endmodule
