	yumi_i

	parameter width_p = 0;

      
	parameter num_in_p = 0;

      
	parameter num_out_p = 0;

      
	parameter in_channel_count_mask_p = 1 << (num_in_p - 1);

      
	input clk;

      
	input reset;

      
	input calibration_done_i;

      
	input [num_in_p - 1:0] valid_i;

	output wire valid_o;

      
	output wire [(num_out_p * width_p) - 1:0] data_o;

      
	input yumi_i;

      
	wire [num_out_p - 1:0] fifo_enq_vec;

      
	wire [num_out_p - 1:0] fifo_not_full_vec;

		.width_p(width_p),

		.in_channel_count_mask_p(in_channel_count_mask_p)

		.clk(clk),

      
		.reset(reset),
