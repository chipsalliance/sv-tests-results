module bsg_source_sync_channel_control_slave (

      
	out_clk_i,

      
	out_reset_i,

      
	out_clk_init_r_o,

      
	out_override_en_o,

      
	out_override_valid_data_o,

      
	in_snoop_valid_data_i,

      
	in_clk_i,

      
	in_trigger_mode_en_o,

      
	in_trigger_mode_alt_en_o,

      
	out_infinite_credits_o,

	core_loopback_en_o,

      
	out_channel_active_o

	output wire out_override_en_o;

      
	output wire [width_p + 0:0] out_override_valid_data_o;

      
	input [width_p + 0:0] in_snoop_valid_data_i;

      
	input in_clk_i;

      
	output wire in_trigger_mode_en_o;

			assign out_snoop_valid_data = {in_snoop_valid_data_i[width_p-:5]};

			assign out_snoop_valid_data = {in_snoop_valid_data_i[width_p-:4], 1'b0};

      
		end

      
		if (width_p <= 4) begin : genblk2

      
			assign out_active_mismatch = 1'b0;

      
		end

      
		else begin : genblk2

      
			assign out_active_mismatch = |(out_active_pattern[width_p - 5:0] ^ in_snoop_valid_data_i[width_p - 5:0]);

      
		end

      
	endgenerate

      
	reg [3:0] out_state_n;

      
	reg [3:0] out_state_r;

      
	always @(posedge out_clk_i)

      
		if (out_reset_i)

      
			out_state_r <= 4'd0;

      
		else

      
			out_state_r <= out_state_n;

      
	always @(*) begin

      
		if (_sv2v_0)

      
			;

      
		out_state_n = out_state_r;

			(* full_case, parallel_case *)

      
			casez (out_snoop_valid_data)

      
				5'b0z111: out_state_n = 4'd7;

      
				5'b01z10: out_state_n = 4'd7;

      
				5'b01011: out_state_n = 4'd7;

      
				5'b01101: out_state_n = 4'd7;

				5'b00101: out_state_n = 4'd2;

      
				5'b00010: out_state_n = 4'd3;

			endcase

	localparam counter_bits_lp = (counter_min_bits_lp > (((width_p + 1) * 2) + 1) ? counter_min_bits_lp : ((width_p + 1) * 2) + 1);

		else

      
			out_rot_r <= out_rot_n;

      
	end

      
	always @(out_state_r)

      
		$display("## Slave %m entering state %s with in_snoop_valid_data_i %b", out_state_r.name, in_snoop_valid_data_i);

      
	assign out_channel_active_o = out_state_r == 4'd5;

      
	wire out_loopback_en = (out_state_r == 4'd3) || (out_state_r == 4'd4);

      
	wire out_trigger_mode_alt_en = out_state_r == 4'd4;

		.iclk_i(out_clk_i),

      
		.iclk_reset_i(1'b0),

      
		.oclk_i(core_clk_i),

		.iclk_reset_i(1'b0),

      
		.oclk_i(in_clk_i),

			end
