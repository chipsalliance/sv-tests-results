module bsg_credit_to_token (

      
	clk_i,

      
	reset_i,

      
	credit_i,

      
	ready_i,

      
	token_o

      
);

      
	parameter decimation_p = 0;

      
	parameter max_val_p = 0;

      
	input clk_i;

      
	input reset_i;

      
	input credit_i;

      
	input ready_i;

      
	output wire token_o;

      
	localparam counter_width_lp = $clog2({1'b0, max_val_p} + 1);

      
	localparam step_width_lp = $clog2({1'b0, decimation_p} + 1);

      
	wire [counter_width_lp - 1:0] count;

      
	wire [step_width_lp - 1:0] up;

      
	wire [step_width_lp - 1:0] down;

      
	wire token_ready;

      
	wire token_almost_ready;

      
	bsg_counter_up_down_variable #(

      
		.max_val_p(max_val_p),

      
		.init_val_p(0),

      
		.max_step_p(decimation_p)

      
	) credit_counter(

      
		.clk_i(clk_i),

      
		.reset_i(reset_i),

      
		.up_i(up),

      
		.down_i(down),

      
		.count_o(count)

      
	);

      
	assign up = {{step_width_lp - 1 {1'b0}}, credit_i};

      
	function automatic [step_width_lp - 1:0] sv2v_cast_59E90;

      
		input reg [step_width_lp - 1:0] inp;

      
		sv2v_cast_59E90 = inp;

      
	endfunction

      
	function automatic signed [step_width_lp - 1:0] sv2v_cast_59E90_signed;

      
		input reg signed [step_width_lp - 1:0] inp;

      
		sv2v_cast_59E90_signed = inp;

      
	endfunction

      
	assign down = (token_o ? sv2v_cast_59E90($unsigned(decimation_p)) : sv2v_cast_59E90_signed(0));

      
	assign token_ready = count >= decimation_p;

      
	assign token_almost_ready = count >= $unsigned(decimation_p - 1);

      
	assign token_o = ready_i & (token_ready | (token_almost_ready & credit_i));

      
endmodule
