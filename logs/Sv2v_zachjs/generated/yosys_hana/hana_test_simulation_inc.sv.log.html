	output reg unequal;

      
	output reg greater;

      
	output reg lesser;

      
	always @(in1 or in2)

      
		if (in1 == in2) begin

			unequal = 0;

      
			greater = 0;

      
			lesser = 0;

      
		end

			equal = 0;

			if (in1 < in2) begin

      
				greater = 0;

      
				lesser = 1;

      
			end

      
			else begin

      
				greater = 1;

      
				lesser = 0;

      
			end

      
		end
