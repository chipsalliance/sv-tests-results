			out <= 0;

      
			carry <= 0;

      
		end

			out <= load;

      
			carry <= 0;

      
		end

      
		else

      
			case (op)

      
				2'b01: {carry, out} <= {carry, out} + 1;

      
				2'b10: {carry, out} <= {carry, out} - 1;

      
				2'b11: out <= ~out;

      
			endcase

		op = 2'b00;

      
		rst = 1;

      
		set = 0;

      
		load = 0;
