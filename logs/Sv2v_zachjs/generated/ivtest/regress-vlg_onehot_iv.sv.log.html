module test_onehot;

      
	reg clk;

      
	reg resetb;

      
	reg [7:0] a;

      
	reg [7:0] b;

      
	wire [7:0] x;

      
	wire [7:0] y;

      
	reg error;

      
	onehot onehot(

      
		.clk(clk),

      
		.resetb(resetb),

      
		.a(a),

      
		.b(b),

      
		.x(x),

      
		.y(y)

      
	);

      
	initial begin

      
		clk = 0;

      
		forever #(10) clk = ~clk;

      
	end

      
	initial begin

      
		resetb = 0;

      
		#(33) resetb = 1;

      
	end

      
	initial begin

      
		error = 0;

      
		a = 0;

      
		b = 0;

      
		repeat (64) @(posedge clk)

      
			;

      
		#(1) a = 1;

      
		b = 0;

      
		repeat (256) @(posedge clk)

      
			;

      
		#(1) a = 0;

      
		b = 1;

      
		repeat (256) @(posedge clk)

      
			;

      
		if (x !== 8'd43) begin

      
			error = 1;

      
			$display("FAILED - X Expected value 43, is %d", x);

      
		end

      
		if (y !== 8'd64) begin

      
			error = 1;

      
			$display("FAILED - Y Expected value 63, is %d", y);

      
		end

      
		if (error == 0)

      
			$display("PASSED");

      
		$finish;

      
	end

      
endmodule
