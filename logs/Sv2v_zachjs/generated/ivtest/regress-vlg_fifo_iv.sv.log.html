module fifo (
	clk,
	rstp,
	din,
	writep,
	readp,
	dout,
	emptyp,
	fullp
);
	input clk;
	input rstp;
	input [15:0] din;
	input readp;
	input writep;
	output [15:0] dout;
	output emptyp;
	output fullp;
	parameter DEPTH = 3;
	parameter MAX_COUNT = 3'b111;
	reg [DEPTH - 1:0] tail;
	reg [DEPTH - 1:0] head;
	reg [DEPTH - 1:0] count;
	reg [15:0] fifomem [0:MAX_COUNT];
	always @(posedge clk)
		if (rstp == 1)
			dout <= 16'h0000;
		else
			dout <= fifomem[tail];
	always @(posedge clk)
		if (((rstp == 1'b0) && (writep == 1'b1)) && (fullp == 1'b0))
			fifomem[head] <= din;
	always @(posedge clk)
		if (rstp == 1'b1)
			head <= 2'b00;
		else if ((writep == 1'b1) && (fullp == 1'b0))
			head <= head + 1;
	always @(posedge clk)
		if (rstp == 1'b1)
			tail <= 2'b00;
		else if ((readp == 1'b1) && (emptyp == 1'b0))
			tail <= tail + 1;
	always @(posedge clk)
		if (rstp == 1'b1)
			count <= 2'b00;
		else
			case ({readp, writep})
				2'b00: count <= count;
				2'b01:
					if (count != MAX_COUNT)
						count <= count + 1;
				2'b10:
					if (count != 2'b00)
						count <= count - 1;
				2'b11: count <= count;
			endcase
	always @(count)
		if (count == 2'b00)
			emptyp <= 1'b1;
		else
			emptyp <= 1'b0;
	always @(count)
		if (count == MAX_COUNT)
			fullp <= 1'b1;
		else
			fullp <= 1'b0;
endmodule
