	reg clock;

      
	reg red;

      
	reg amber;

      
	reg green;

      
	parameter on = 1;

      
	parameter off = 0;

      
	parameter red_tics = 350;

      
	parameter amber_tics = 30;

      
	parameter green_tics = 200;

	initial amber = off;

      
	initial green = off;

		output reg color;

			color = off;

      
		end

	end

	end
