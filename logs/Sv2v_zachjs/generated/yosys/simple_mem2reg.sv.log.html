	data

      
);

	input reset;

      
	input mode;

      
	input [2:0] addr;

      
	output wire [3:0] data;

      
	(* mem2reg *) reg [3:0] mem [0:7];

      
	assign data = mem[addr];

      
	integer i;

      
	always @(posedge clk)

      
		if (reset)

      
			for (i = 0; i < 8; i = i + 1)

      
				mem[i] <= i;

      
		else if (mode)

      
			for (i = 0; i < 8; i = i + 1)

      
				mem[i] <= mem[i] + 1;

      
		else

      
			mem[addr] <= 0;

      
endmodule
