module RippleCarryAdder (

      
	A,

      
	B,

      
	Cin,

      
	S,

      
	Cout

      
);

      
	parameter N = 4;

      
	input [N - 1:0] A;

      
	input [N - 1:0] B;

      
	input Cin;

      
	output wire [N - 1:0] S;

      
	output wire Cout;

      
	wire [N:0] CC;

      
	assign CC[0] = Cin;

      
	assign Cout = CC[N];

      
	genvar _gv_i_3;

      
	generate

      
		for (_gv_i_3 = 0; _gv_i_3 < N; _gv_i_3 = _gv_i_3 + 1) begin : addbit

      
			localparam i = _gv_i_3;

      
			FullAdder unit(

      
				.A(A[i]),

      
				.B(B[i]),

      
				.Cin(CC[i]),

      
				.S(S[i]),

      
				.Cout(CC[i + 1])

      
			);

      
		end

      
	endgenerate

      
endmodule

      
module FullAdder (

      
	A,

      
	B,

      
	Cin,

      
	S,

      
	Cout

      
);

      
	input A;

      
	input B;

      
	input Cin;
