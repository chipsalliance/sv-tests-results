module top_db_bounce (

      
	clk_100m,

      
	btn_rst,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 16;

      
	wire signed [15:0] sx;

      
	wire signed [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	wire frame_sys;

      
	xd xd_frame(

      
		.clk_i(clk_pix),

      
		.clk_o(clk_100m),

      
		.rst_i(1'b0),

      
		.rst_o(1'b0),

      
		.i(frame),

      
		.o(frame_sys)

      
	);

      
	localparam FB_WIDTH = 320;

      
	localparam FB_HEIGHT = 180;

      
	localparam FB_CIDXW = 4;

      
	localparam FB_CHANW = 4;

      
	localparam FB_SCALE = 2;

      
	localparam FB_IMAGE = "";

      
	localparam FB_PALETTE = "16_colr_4bit_palette.mem";

      
	localparam FB_BGIDX = 4'h1;

      
	reg fb_we;

      
	wire fb_busy;

      
	wire fb_wready;

      
	wire signed [15:0] fbx;

      
	wire signed [15:0] fby;

      
	reg [3:0] fb_cidx;

      
	wire [3:0] fb_red;

      
	wire [3:0] fb_green;

      
	wire [3:0] fb_blue;

      
	framebuffer_bram_db #(

      
		.WIDTH(FB_WIDTH),

      
		.HEIGHT(FB_HEIGHT),

      
		.CIDXW(FB_CIDXW),

      
		.CHANW(FB_CHANW),

      
		.SCALE(FB_SCALE),

      
		.F_IMAGE(FB_IMAGE),

      
		.F_PALETTE(FB_PALETTE)

      
	) fb_inst(

      
		.clk_sys(clk_100m),

      
		.clk_pix(clk_pix),

      
		.rst_sys(1'b0),

      
		.rst_pix(1'b0),

      
		.de(((sy >= 60) && (sy < 420)) && (sx >= 0)),

      
		.frame(frame),

      
		.line(line),

      
		.we(fb_we),

      
		.x(fbx),

      
		.y(fby),

      
		.cidx(fb_cidx),

      
		.bgidx(FB_BGIDX),

      
		.clear(1'b1),

      
		.busy(fb_busy),

      
		.wready(fb_wready),

      
		.clip(),

      
		.red(fb_red),

      
		.green(fb_green),

      
		.blue(fb_blue)

      
	);

      
	localparam Q1_SIZE = 80;

      
	reg [15:0] q1x;

      
	reg [15:0] q1y;

      
	reg q1dx;

      
	reg q1dy;

      
	reg [15:0] q1s = 1;

      
	always @(posedge clk_100m)

      
		if (frame_sys) begin

      
			if (q1x >= (FB_WIDTH - (Q1_SIZE + q1s))) begin

      
				q1dx <= 1;

      
				q1x <= q1x - q1s;

      
			end

      
			else if (q1x < q1s) begin

      
				q1dx <= 0;

      
				q1x <= q1x + q1s;

      
			end

      
			else

      
				q1x <= (q1dx ? q1x - q1s : q1x + q1s);

      
			if (q1y >= (FB_HEIGHT - (Q1_SIZE + q1s))) begin

      
				q1dy <= 1;

      
				q1y <= q1y - q1s;

      
			end

      
			else if (q1y < q1s) begin

      
				q1dy <= 0;

      
				q1y <= q1y + q1s;

      
			end

      
			else

      
				q1y <= (q1dy ? q1y - q1s : q1y + q1s);

      
		end

      
	reg [15:0] rx0;

      
	reg [15:0] ry0;

      
	reg [15:0] rx1;

      
	reg [15:0] ry1;

      
	reg draw_start;

      
	wire drawing;

      
	wire draw_done;

      
	reg [31:0] state;

      
	always @(posedge clk_100m)

      
		case (state)

      
			32'd1:

      
				if (fb_wready) begin

      
					draw_start <= 1;

      
					state <= 32'd2;

      
					rx0 <= q1x;

      
					ry0 <= q1y;

      
					rx1 <= q1x + Q1_SIZE;

      
					ry1 <= q1y + Q1_SIZE;

      
					fb_cidx <= 4'hb;

      
				end

      
			32'd2: begin

      
				draw_start <= 0;

      
				if (draw_done)

      
					state <= 32'd3;

      
			end

      
			32'd3: state <= 32'd0;

      
			default:

      
				if (frame_sys)

      
					state <= 32'd1;

      
		endcase

      
	draw_rectangle_fill #(.CORDW(CORDW)) draw_rectangle_inst(

      
		.clk(clk_100m),

      
		.rst(1'b0),

      
		.start(draw_start),

      
		.oe(!fb_busy),

      
		.x0(rx0),

      
		.y0(ry0),

      
		.x1(rx1),

      
		.y1(ry1),

      
		.x(fbx),

      
		.y(fby),

      
		.drawing(drawing),

      
		.busy(),

      
		.done(draw_done)

      
	);

      
	always @(*) fb_we = drawing;

      
	reg hsync_p1;

      
	reg vsync_p1;

      
	always @(posedge clk_pix) begin

      
		hsync_p1 <= hsync;

      
		vsync_p1 <= vsync;

      
	end

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync_p1;

      
		vga_vsync <= vsync_p1;

      
		vga_r <= fb_red;

      
		vga_g <= fb_green;

      
		vga_b <= fb_blue;

      
	end
