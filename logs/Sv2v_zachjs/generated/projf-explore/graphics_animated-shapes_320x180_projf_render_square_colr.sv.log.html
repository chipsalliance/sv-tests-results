`default_nettype none

      
module render_square_colr (

      
	clk,

      
	rst,

      
	oe,

      
	start,

      
	x,

      
	y,

      
	cidx,

      
	drawing,

      
	done

      
);

      
	parameter CORDW = 16;

      
	parameter CIDXW = 4;

      
	parameter SCALE = 1;

      
	input wire clk;

      
	input wire rst;

      
	input wire oe;

      
	input wire start;

      
	output wire signed [CORDW - 1:0] x;

      
	output wire signed [CORDW - 1:0] y;

      
	output reg [CIDXW - 1:0] cidx;

      
	output wire drawing;

      
	output reg done;

      
	localparam QS = 64;

      
	localparam SP = 1;

      
	localparam BW = 320;

      
	localparam BH = 180;

      
	localparam COLR_CHANGE = 12;

      
	reg [3:0] cnt_colr;

      
	reg signed [CORDW - 1:0] vx0;

      
	reg signed [CORDW - 1:0] vy0;

      
	reg signed [CORDW - 1:0] vx1;

      
	reg signed [CORDW - 1:0] vy1;

      
	reg dx;

      
	reg dy;

      
	reg draw_start;

      
	wire draw_done;

      
	reg [31:0] state;

      
	always @(posedge clk) begin

      
		case (state)

      
			32'd1: begin

      
				draw_start <= 1;

      
				state <= 32'd2;

      
				if (vx0 >= 255) begin

      
					dx <= 1;

      
					vx0 <= vx0 - SP;

      
				end

      
				else if (vx0 < SP) begin

      
					dx <= 0;

      
					vx0 <= vx0 + SP;

      
				end

      
				else

      
					vx0 <= (dx ? vx0 - SP : vx0 + SP);

      
				if (vy0 >= 115) begin

      
					dy <= 1;

      
					vy0 <= vy0 - SP;

      
				end

      
				else if (vy0 < SP) begin

      
					dy <= 0;

      
					vy0 <= vy0 + SP;

      
				end

      
				else

      
					vy0 <= (dy ? vy0 - SP : vy0 + SP);

      
				if (cnt_colr == 11) begin

      
					cnt_colr <= 0;

      
					cidx <= cidx + 1;

      
				end

      
				else

      
					cnt_colr <= cnt_colr + 1;

      
			end

      
			32'd2: begin

      
				draw_start <= 0;

      
				if (draw_done)

      
					state <= 32'd3;

      
			end

      
			32'd3: state <= 32'd0;

      
			default:

      
				if (start)

      
					state <= 32'd1;

      
		endcase

      
		if (rst)

      
			state <= 32'd0;

      
	end

      
	always @(*) begin

      
		vx1 = vx0 + QS;

      
		vy1 = vy0 + QS;

      
	end

      
	draw_rectangle_fill #(.CORDW(CORDW)) draw_rectangle_inst(

      
		.clk(clk),

      
		.rst(rst),

      
		.start(draw_start),

      
		.oe(oe),

      
		.x0(vx0 * SCALE),

      
		.y0(vy0 * SCALE),

      
		.x1(vx1 * SCALE),

      
		.y1(vy1 * SCALE),

      
		.x(x),

      
		.y(y),

      
		.drawing(drawing),

      
		.busy(),

      
		.done(draw_done)

      
	);

      
	always @(*) done = state == 32'd3;

      
endmodule
