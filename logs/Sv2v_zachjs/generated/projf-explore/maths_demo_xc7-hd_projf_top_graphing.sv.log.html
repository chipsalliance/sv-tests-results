module top_graphing (

      
	clk_100m,

      
	btn_rst,

      
	hdmi_tx_ch0_p,

      
	hdmi_tx_ch0_n,

      
	hdmi_tx_ch1_p,

      
	hdmi_tx_ch1_n,

      
	hdmi_tx_ch2_p,

      
	hdmi_tx_ch2_n,

      
	hdmi_tx_clk_p,

      
	hdmi_tx_clk_n

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	output wire hdmi_tx_ch0_p;

      
	output wire hdmi_tx_ch0_n;

      
	output wire hdmi_tx_ch1_p;

      
	output wire hdmi_tx_ch1_n;

      
	output wire hdmi_tx_ch2_p;

      
	output wire hdmi_tx_ch2_n;

      
	output wire hdmi_tx_clk_p;

      
	output wire hdmi_tx_clk_n;

      
	wire clk_pix;

      
	wire clk_pix_5x;

      
	wire clk_pix_locked;

      
	clock_gen_720p clock_pix_inst(

      
		.clk_100m(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(clk_pix_5x),

      
		.clk_pix_locked(clk_pix_locked)

      
	);

      
	localparam CORDW = 12;

      
	wire hsync;

      
	wire vsync;

      
	wire [11:0] sx;

      
	wire [11:0] sy;

      
	wire de;

      
	display_720p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_pix_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(),

      
		.line()

      
	);

      
	wire draw;

      
	reg axes;

      
	reg [7:0] dvi_red;

      
	reg [7:0] dvi_green;

      
	reg [7:0] dvi_blue;

      
	reg dvi_hsync;

      
	reg dvi_vsync;

      
	reg dvi_de;

      
	always @(posedge clk_pix) begin

      
		dvi_hsync <= hsync;

      
		dvi_vsync <= vsync;

      
		dvi_de <= de;

      
		dvi_red <= (!de ? 8'h00 : (axes ? 8'hcc : (draw ? 8'h33 : 8'h22)));

      
		dvi_green <= (!de ? 8'h00 : (axes ? 8'hcc : (draw ? 8'hbb : 8'h22)));

      
		dvi_blue <= (!de ? 8'h00 : (axes ? 8'hcc : (draw ? 8'hcc : 8'h22)));

      
	end

      
	wire tmds_ch0_serial;

      
	wire tmds_ch1_serial;

      
	wire tmds_ch2_serial;

      
	wire tmds_clk_serial;

      
	dvi_generator dvi_out(

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(clk_pix_5x),

      
		.rst_pix(!clk_pix_locked),

      
		.de(dvi_de),

      
		.data_in_ch0(dvi_blue),

      
		.data_in_ch1(dvi_green),

      
		.data_in_ch2(dvi_red),

      
		.ctrl_in_ch0({dvi_vsync, dvi_hsync}),

      
		.ctrl_in_ch1(2'b00),

      
		.ctrl_in_ch2(2'b00),

      
		.tmds_ch0_serial(tmds_ch0_serial),

      
		.tmds_ch1_serial(tmds_ch1_serial),

      
		.tmds_ch2_serial(tmds_ch2_serial),

      
		.tmds_clk_serial(tmds_clk_serial)

      
	);

      
	tmds_out tmds_ch0(

      
		.tmds(tmds_ch0_serial),

      
		.pin_p(hdmi_tx_ch0_p),

      
		.pin_n(hdmi_tx_ch0_n)

      
	);

      
	tmds_out tmds_ch1(

      
		.tmds(tmds_ch1_serial),

      
		.pin_p(hdmi_tx_ch1_p),

      
		.pin_n(hdmi_tx_ch1_n)

      
	);

      
	tmds_out tmds_ch2(

      
		.tmds(tmds_ch2_serial),

      
		.pin_p(hdmi_tx_ch2_p),

      
		.pin_n(hdmi_tx_ch2_n)

      
	);

      
	tmds_out tmds_clk(

      
		.tmds(tmds_clk_serial),

      
		.pin_p(hdmi_tx_clk_p),

      
		.pin_n(hdmi_tx_clk_n)

      
	);

      
	reg signed [11:0] x;

      
	reg signed [11:0] y;

      
	localparam X_OFFS = 640;

      
	localparam Y_OFFS = 359;

      
	always @(posedge clk_pix) begin

      
		x <= (sx - X_OFFS) + 4;

      
		y <= Y_OFFS - sy;

      
	end

      
	reg x_axis;

      
	reg y_axis;

      
	always @(*) begin

      
		x_axis = sy == Y_OFFS;

      
		y_axis = sx == X_OFFS;

      
		axes = x_axis || y_axis;

      
	end

      
	func_squared #(.CORDW(CORDW)) func_inst(

      
		.clk(clk_pix),

      
		.x(x),

      
		.y(y),

      
		.r(draw)

      
	);

      
endmodule
