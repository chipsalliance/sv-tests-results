	output reg [3:0] vga_g;

		.rst(!btn_rst),

		.clk_locked(clk_locked)

	wire signed [15:0] sx;

	wire vsync;

		.de(),

      
		.frame(frame),

      
		.line(line)

      
	);

	xd xd_frame(

      
		.clk_i(clk_pix),

      
		.clk_o(clk_100m),

      
		.rst_i(1'b0),

      
		.rst_o(1'b0),

      
		.i(frame),

      
		.o(frame_sys)

	localparam FB_WIDTH = 320;

      
	localparam FB_HEIGHT = 180;

      
	localparam FB_CIDXW = 4;

      
	localparam FB_CHANW = 4;

      
	localparam FB_SCALE = 2;

      
	localparam FB_IMAGE = "";

	reg fb_we;

      
	wire signed [15:0] fbx;

      
	wire signed [15:0] fby;

      
	reg [3:0] fb_cidx;

      
	wire fb_busy;

      
	wire [3:0] fb_red;

      
	wire [3:0] fb_green;

      
	wire [3:0] fb_blue;

      
	framebuffer_bram #(

      
		.WIDTH(FB_WIDTH),

      
		.HEIGHT(FB_HEIGHT),

      
		.CIDXW(FB_CIDXW),

      
		.CHANW(FB_CHANW),

      
		.SCALE(FB_SCALE),

      
		.F_IMAGE(FB_IMAGE),

      
		.F_PALETTE(FB_PALETTE)

      
	) fb_inst(

      
		.clk_sys(clk_100m),

      
		.clk_pix(clk_pix),

      
		.rst_sys(1'b0),

      
		.rst_pix(1'b0),

		.red(fb_red),

      
		.green(fb_green),

      
		.blue(fb_blue)

      
	);

      
	localparam SHAPE_CNT = 64;

      
	reg [6:0] shape_id;

      
	reg signed [15:0] vx0;

      
	reg signed [15:0] vy0;

      
	reg signed [15:0] vx1;

      
	reg signed [15:0] vy1;

      
	reg draw_start;

      
	wire drawing;

      
	wire draw_done;

      
	reg [31:0] state;

      
	always @(posedge clk_100m)

      
		case (state)

      
			32'd1: begin

      
				draw_start <= 1;

      
				state <= 32'd2;

      
				vx0 <= 60 + shape_id;

      
				vy0 <= 15 + shape_id;

      
				vx1 <= 260 - shape_id;

      
				vy1 <= 165 - shape_id;

      
				fb_cidx <= shape_id[3:0];

      
			end

      
			32'd2: begin

      
				draw_start <= 0;

      
				if (draw_done)

      
					if (shape_id == 63)

      
						state <= 32'd3;

      
					else begin

      
						shape_id <= shape_id + 1;

      
						state <= 32'd1;

      
					end

      
			end

      
			32'd3: state <= 32'd3;

      
			default:

      
				if (frame_sys)

      
					state <= 32'd1;

      
		endcase

	localparam PIX_FRAME = 200;

	reg [7:0] cnt_pix_frame;

			cnt_pix_frame <= 0;

				cnt_pix_frame <= cnt_pix_frame + 1;

      
			end

      
	end

		.rst(1'b0),

      
		.start(draw_start),

      
		.oe(draw_req && !fb_busy),

      
		.x0(vx0),

      
		.y0(vy0),

      
		.x1(vx1),

      
		.y1(vy1),

      
		.x(fbx),

      
		.y(fby),

      
		.drawing(drawing),

      
		.busy(),

      
		.done(draw_done)

      
	);

      
	always @(*) fb_we = drawing;

      
	reg hsync_p1;

      
	reg vsync_p1;

      
	always @(posedge clk_pix) begin

		vsync_p1 <= vsync;

      
	end

	end
