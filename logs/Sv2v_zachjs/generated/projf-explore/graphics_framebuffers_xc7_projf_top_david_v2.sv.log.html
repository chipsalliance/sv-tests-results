	clk_100m,

      
	btn_rst,

	vga_g,

	input wire btn_rst;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 16;

      
	wire signed [15:0] sx;

      
	wire signed [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire frame;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(),

      
		.frame(frame),

      
		.line()

      
	);

      
	localparam FB_WIDTH = 160;

      
	localparam FB_HEIGHT = 120;

      
	localparam FB_PIXELS = 19200;

      
	localparam FB_ADDRW = 15;

      
	localparam FB_DATAW = 4;

      
	localparam FB_IMAGE = "david.mem";

      
	localparam FB_PALETTE = "david_palette.mem";

      
	reg fb_we;

      
	reg [14:0] fb_addr_write;

      
	reg [14:0] fb_addr_read;

      
	reg [3:0] fb_cidx_write;

      
	wire [3:0] fb_cidx_read;

      
	bram_sdp #(

      
		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F(FB_IMAGE)

      
	) bram_inst(

      
		.clk_write(clk_pix),

      
		.clk_read(clk_pix),

      
		.we(fb_we),

      
		.addr_write(fb_addr_write),

      
		.addr_read(fb_addr_read),

      
		.data_in(fb_cidx_write),

      
		.data_out(fb_cidx_read)

      
	);

      
	reg lfsr_en;

      
	wire [14:0] lfsr;

      
	lfsr #(

      
		.LEN(15),

      
		.TAPS(15'b110000000000000)

      
	) lsfr_fz(

      
		.clk(clk_pix),

      
		.rst(!clk_locked),

      
		.en(lfsr_en),

      
		.seed(0),

      
		.sreg(lfsr)

      
	);

      
	localparam FADE_WAIT = 600;

      
	localparam FADE_RATE = 3200;

      
	reg [9:0] cnt_wait;

      
	reg [11:0] cnt_rate;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			cnt_wait <= (cnt_wait != 599 ? cnt_wait + 1 : cnt_wait);

      
		if (cnt_wait == 599)

      
			if (cnt_rate == 3199) begin

      
				lfsr_en <= 1;

      
				fb_we <= 1;

      
				fb_addr_write <= lfsr;

      
				cnt_rate <= 0;

      
			end

      
			else begin

      
				cnt_rate <= cnt_rate + 1;

      
				lfsr_en <= 0;

      
				fb_we <= 0;

      
			end

      
		fb_cidx_write <= 4'hf;

      
	end

      
	reg paint;

      
	always @(*) paint = (((sy >= 0) && (sy < FB_HEIGHT)) && (sx >= 0)) && (sx < FB_WIDTH);

      
	always @(posedge clk_pix)

      
		if (frame)

      
			fb_addr_read <= 0;

      
		else if (paint)

      
			fb_addr_read <= fb_addr_read + 1;

      
	reg [3:0] fb_cidx_read_p1;

      
	always @(posedge clk_pix) fb_cidx_read_p1 <= fb_cidx_read;

		.WIDTH(12),

      
		.DEPTH(16),

      
		.INIT_F(FB_PALETTE)

		.data(clut_colr)

      
	);

	reg [1:0] vsync_sr;

	end
