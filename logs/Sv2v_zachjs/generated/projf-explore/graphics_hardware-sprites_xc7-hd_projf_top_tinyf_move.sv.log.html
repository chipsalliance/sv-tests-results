		.clk_100m(clk_100m),

      
		.rst(!btn_rst_n),

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(clk_pix_5x),

      
		.clk_pix_locked(clk_pix_locked)

	wire [15:0] sx;

      
	wire [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

		.clk_pix(clk_pix),

		.sx(sx),

      
		.sy(sy),

		.line(line)

	localparam SPR_WIDTH = 8;

      
	localparam SPR_HEIGHT = 8;

      
	localparam SPR_SCALE = 4;

      
	localparam SPR_DATAW = 1;

      
	localparam SPR_DRAWW = 128;

      
	localparam SPR_DRAWH = 128;

      
	localparam SPR_SPX = 4;

      
	localparam SPR_FILE = "letter_f.mem";

      
	reg signed [15:0] sprx;

      
	reg signed [15:0] spry;

      
	reg dx;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			if (dx == 0) begin

      
				if ((sprx + SPR_DRAWW) >= 1536)

      
					dx <= 1;

      
				else

      
					sprx <= sprx + SPR_SPX;

      
			end

      
			else if (sprx <= (-2 * SPR_DRAWW))

      
				dx <= 0;

      
			else

      
				sprx <= sprx - SPR_SPX;

      
		if (rst_pix) begin

      
			sprx <= 576;

      
			spry <= 296;

      
			dx <= 0;

      
		end

      
	end

	wire [0:0] pix;

		.SPR_HEIGHT(SPR_HEIGHT),

      
		.SPR_SCALE(SPR_SCALE),

      
		.SPR_DATAW(SPR_DATAW)

		.rst(rst_pix),

      
		.line(line),

		.sy(sy),

      
		.sprx(sprx),

      
		.spry(spry),

		.drawing(drawing)

	reg [3:0] paint_b;

		paint_r = (drawing && pix ? 4'hf : 4'h1);

      
		paint_g = (drawing && pix ? 4'hc : 4'h3);

      
		paint_b = (drawing && pix ? 4'h0 : 4'h7);
