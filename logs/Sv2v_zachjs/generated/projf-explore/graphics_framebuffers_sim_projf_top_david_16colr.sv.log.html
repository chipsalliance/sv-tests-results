	clk_pix,

	sdl_frame,

      
	sdl_r,

      
	sdl_g,

	input wire clk_pix;

      
	input wire rst_pix;

      
	output reg signed [CORDW - 1:0] sdl_sx;

      
	output reg signed [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg sdl_frame;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	wire signed [CORDW - 1:0] sx;

      
	wire signed [CORDW - 1:0] sy;

      
	wire de;

      
	wire frame;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(rst_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de),

		.line()

	localparam BMAP_IMAGE = "../res/david/david.mem";

      
	localparam PAL_FILE = {LIB_RES, "/palettes/grey16_4b.mem"};

      
	localparam CHANW = 4;

      
	localparam COLRW = 12;

      
	localparam CIDXW = 4;

      
	localparam FB_WIDTH = 160;

      
	localparam FB_HEIGHT = 120;

      
	localparam FB_PIXELS = 19200;

      
	localparam FB_ADDRW = 15;

      
	localparam FB_DATAW = CIDXW;

      
	reg [14:0] fb_addr_read;

      
	wire [3:0] fb_colr_read;

      
	bram_sdp #(

      
		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F(BMAP_IMAGE)

		.clk_write(clk_pix),

      
		.clk_read(clk_pix),

      
		.we(),

      
		.addr_write(),

		.data_in(),

      
		.data_out(fb_colr_read)

      
	);

	always @(posedge clk_pix) begin

      
		read_fb <= (((sy >= 0) && (sy < FB_HEIGHT)) && (sx >= -LAT)) && (sx < 157);

      
		if (frame)

      
			fb_addr_read <= 0;

	);

	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

		paint_area = (((sy >= 0) && (sy < FB_HEIGHT)) && (sx >= 0)) && (sx < FB_WIDTH);

      
		{paint_r, paint_g, paint_b} = (paint_area ? fb_pix_colr : 12'h000);

      
	end
