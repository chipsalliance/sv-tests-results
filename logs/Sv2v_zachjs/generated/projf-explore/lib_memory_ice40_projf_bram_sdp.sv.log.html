	clk_write,

	data_out

	parameter WIDTH = 8;

      
	parameter DEPTH = 512;

      
	parameter INIT_F = "";

      
	localparam ADDRW = $clog2(DEPTH);

      
	input wire clk_write;

      
	input wire clk_read;

      
	input wire we;

      
	input wire [ADDRW - 1:0] addr_write;

      
	input wire [ADDRW - 1:0] addr_read;

	output reg [WIDTH - 1:0] data_out;

		$display("Creating bram from init file '%s'.", INIT_F);

      
		$readmemh(INIT_F, memory);

      
	end

      
	always @(posedge clk_write)

      
		if (we)

      
			memory[addr_write] <= data_in;

      
	always @(posedge clk_read) data_out <= memory[addr_read];
