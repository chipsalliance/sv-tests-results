		.clk_pix(clk_pix),

      
		.rst(!clk_locked),

		.sy(),

		.clk_i(clk_pix),

      
		.clk_o(clk_100m),

      
		.rst_i(1'b0),

      
		.rst_o(1'b0),

      
		.i(frame),

      
		.o(frame_sys)

      
	);

	wire life_changed;

				cnt_frames <= 0;

      
			end

      
			else

      
				cnt_frames <= cnt_frames + 1;

      
	end
