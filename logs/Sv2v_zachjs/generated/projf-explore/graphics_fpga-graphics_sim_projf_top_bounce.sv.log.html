module top_bounce (

      
	clk_pix,

      
	rst,

      
	sdl_sx,

      
	sdl_sy,

      
	sdl_de,

      
	sdl_r,

      
	sdl_g,

      
	sdl_b

      
);

      
	parameter CORDW = 10;

      
	input wire clk_pix;

      
	input wire rst;

      
	output reg [CORDW - 1:0] sdl_sx;

      
	output reg [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	wire [CORDW - 1:0] sx;

      
	wire [CORDW - 1:0] sy;

      
	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst(rst),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de)

      
	);

      
	localparam H_RES = 640;

      
	localparam V_RES = 480;

      
	reg animate;

      
	always @(*) animate = (sy == V_RES) && (sx == 0);

      
	localparam Q1_SIZE = 100;

      
	localparam Q2_SIZE = 150;

      
	localparam Q3_SIZE = 200;

      
	reg [CORDW - 1:0] q1x;

      
	reg [CORDW - 1:0] q1y;

      
	reg [CORDW - 1:0] q2x;

      
	reg [CORDW - 1:0] q2y;

      
	reg [CORDW - 1:0] q3x;

      
	reg [CORDW - 1:0] q3y;

      
	reg q1dx;

      
	reg q1dy;

      
	reg q2dx;

      
	reg q2dy;

      
	reg q3dx;

      
	reg q3dy;

      
	reg [CORDW - 1:0] q1s = 2;

      
	reg [CORDW - 1:0] q2s = 1;

      
	reg [CORDW - 1:0] q3s = 1;

      
	always @(posedge clk_pix)

      
		if (animate) begin

      
			if (q1x >= (H_RES - (Q1_SIZE + q1s))) begin

      
				q1dx <= 1;

      
				q1x <= q1x - q1s;

      
			end

      
			else if (q1x < q1s) begin

      
				q1dx <= 0;

      
				q1x <= q1x + q1s;

      
			end

      
			else

      
				q1x <= (q1dx ? q1x - q1s : q1x + q1s);

      
			if (q1y >= (V_RES - (Q1_SIZE + q1s))) begin

      
				q1dy <= 1;

      
				q1y <= q1y - q1s;

      
			end

      
			else if (q1y < q1s) begin

      
				q1dy <= 0;

      
				q1y <= q1y + q1s;

      
			end

      
			else

      
				q1y <= (q1dy ? q1y - q1s : q1y + q1s);

      
			if (q2x >= (H_RES - (Q2_SIZE + q2s))) begin

      
				q2dx <= 1;

      
				q2x <= q2x - q2s;

      
			end

      
			else if (q2x < q2s) begin

      
				q2dx <= 0;

      
				q2x <= q2x + q2s;

      
			end

      
			else

      
				q2x <= (q2dx ? q2x - q2s : q2x + q2s);

      
			if (q2y >= (V_RES - (Q2_SIZE + q2s))) begin

      
				q2dy <= 1;

      
				q2y <= q2y - q2s;

      
			end

      
			else if (q2y < q2s) begin

      
				q2dy <= 0;

      
				q2y <= q2y + q2s;

      
			end

      
			else

      
				q2y <= (q2dy ? q2y - q2s : q2y + q2s);

      
			if (q3x >= (H_RES - (Q3_SIZE + q3s))) begin

      
				q3dx <= 1;

      
				q3x <= q3x - q3s;

      
			end

      
			else if (q3x < q3s) begin

      
				q3dx <= 0;

      
				q3x <= q3x + q3s;

      
			end

      
			else

      
				q3x <= (q3dx ? q3x - q3s : q3x + q3s);

      
			if (q3y >= (V_RES - (Q3_SIZE + q3s))) begin

      
				q3dy <= 1;

      
				q3y <= q3y - q3s;

      
			end

      
			else if (q3y < q3s) begin

      
				q3dy <= 0;

      
				q3y <= q3y + q3s;

      
			end

      
			else

      
				q3y <= (q3dy ? q3y - q3s : q3y + q3s);

      
		end

      
	reg q1_draw;

      
	reg q2_draw;

      
	reg q3_draw;

      
	always @(*) begin

      
		q1_draw = (((sx >= q1x) && (sx < (q1x + Q1_SIZE))) && (sy >= q1y)) && (sy < (q1y + Q1_SIZE));

      
		q2_draw = (((sx >= q2x) && (sx < (q2x + Q2_SIZE))) && (sy >= q2y)) && (sy < (q2y + Q2_SIZE));

      
		q3_draw = (((sx >= q3x) && (sx < (q3x + Q3_SIZE))) && (sy >= q3y)) && (sy < (q3y + Q3_SIZE));

      
	end

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_r <= (de && q1_draw ? 8'hff : 8'h00);

      
		sdl_g <= (de && q2_draw ? 8'hff : 8'h00);

      
		sdl_b <= (de && q3_draw ? 8'hff : 8'h00);

      
	end
