	output wire dvi_de;

      
	output wire [3:0] dvi_r;

      
	output wire [3:0] dvi_g;

      
	output wire [3:0] dvi_b;

	wire clk_locked;

	wire hsync;

      
	wire vsync;

      
	wire de;

	localparam FB_WIDTH = 320;

      
	localparam FB_HEIGHT = 180;

      
	localparam FB_CIDXW = 4;

      
	localparam FB_CHANW = 4;

      
	localparam FB_SCALE = 2;

	wire [3:0] fb_blue;

		.WIDTH(FB_WIDTH),

      
		.HEIGHT(FB_HEIGHT),

      
		.CIDXW(FB_CIDXW),

      
		.CHANW(FB_CHANW),

      
		.SCALE(FB_SCALE),

      
		.F_IMAGE(FB_IMAGE),

      
		.F_PALETTE(FB_PALETTE)

      
	) fb_inst(

      
		.clk_sys(clk_pix),

      
		.clk_pix(clk_pix),

      
		.rst_sys(1'b0),

      
		.rst_pix(1'b0),

      
		.de(((sy >= 60) && (sy < 420)) && (sx >= 0)),

      
		.frame(frame),

      
		.line(line),

      
		.we(fb_we),

      
		.x(fbx),

      
		.y(fby),

		.busy(fb_busy),

      
		.red(fb_red),

      
		.green(fb_green),

      
		.blue(fb_blue)

      
	);

      
	localparam LINE_CNT = 9;

      
	reg [3:0] line_id;

      
	reg signed [15:0] vx0;

      
	reg signed [15:0] vy0;

      
	reg signed [15:0] vx1;

      
	reg signed [15:0] vy1;

      
	reg draw_start;

      
	wire drawing;

      
	wire draw_done;

      
	reg signed [15:0] fbx_clear;

      
	reg signed [15:0] fby_clear;

      
	reg clearing;

      
	reg [31:0] state;

      
	always @(posedge clk_pix) begin

      
		case (state)

      
			32'd1: begin

      
				fb_cidx <= 4'h0;

      
				if (!fb_busy)

      
					if ((fby_clear == 179) && (fbx_clear == 319)) begin

      
						clearing <= 0;

      
						state <= 32'd2;

      
					end

      
					else if (clearing == 1) begin

      
						if (fbx_clear == 319) begin

      
							fbx_clear <= 0;

      
							fby_clear <= fby_clear + 1;

      
						end

      
						else

      
							fbx_clear <= fbx_clear + 1;

      
					end

      
					else

      
						clearing <= 1;

      
			end

      
			32'd2: begin

      
				draw_start <= 1;

      
				state <= 32'd3;

      
				fb_cidx <= 4'h8;

      
				case (line_id)

      
					4'd0: begin

      
						vx0 <= 130;

      
						vy0 <= 60;

      
						vx1 <= 230;

      
						vy1 <= 60;

      
					end

      
					4'd1: begin

      
						vx0 <= 230;

      
						vy0 <= 60;

      
						vx1 <= 230;

      
						vy1 <= 160;

      
					end

      
					4'd2: begin

      
						vx0 <= 230;

      
						vy0 <= 160;

      
						vx1 <= 130;

      
						vy1 <= 160;

      
					end

      
					4'd3: begin

      
						vx0 <= 130;

      
						vy0 <= 160;

      
						vx1 <= 130;

      
						vy1 <= 60;

      
					end

      
					4'd4: begin

      
						vx0 <= 130;

      
						vy0 <= 160;

      
						vx1 <= 90;

      
						vy1 <= 120;

      
					end

      
					4'd5: begin

      
						vx0 <= 90;

      
						vy0 <= 120;

      
						vx1 <= 90;

      
						vy1 <= 20;

      
					end

      
					4'd6: begin

      
						vx0 <= 90;

      
						vy0 <= 20;

      
						vx1 <= 130;

      
						vy1 <= 60;

      
					end

      
					4'd7: begin

      
						vx0 <= 90;

      
						vy0 <= 20;

      
						vx1 <= 190;

      
						vy1 <= 20;

      
					end

      
					4'd8: begin

      
						vx0 <= 190;

      
						vy0 <= 20;

      
						vx1 <= 230;

      
						vy1 <= 60;

      
					end

      
					default: begin

      
						vx0 <= 0;

      
						vy0 <= 0;

      
						vx1 <= 0;

      
						vy1 <= 0;

      
					end

      
				endcase

      
			end

      
			32'd3: begin

      
				draw_start <= 0;

      
				if (draw_done)

      
					if (line_id == 8)

      
						state <= 32'd4;

      
					else begin

      
						line_id <= line_id + 1;

      
						state <= 32'd2;

      
					end

      
			end

      
			32'd4: state <= 32'd4;

      
			default:

      
				if (frame)

      
					state <= 32'd1;

      
		endcase

      
		if (!clk_locked)

      
			state <= 32'd0;

      
	end

      
	localparam FRAME_WAIT = 300;

      
	reg [8:0] cnt_frame_wait;

      
	reg draw_req;

      
	always @(posedge clk_pix) begin

      
		if (!fb_busy)

      
			draw_req <= 0;

      
		if (frame)

      
			if (cnt_frame_wait != 299)

      
				cnt_frame_wait <= cnt_frame_wait + 1;

      
			else

      
				draw_req <= 1;

      
	end

	end
