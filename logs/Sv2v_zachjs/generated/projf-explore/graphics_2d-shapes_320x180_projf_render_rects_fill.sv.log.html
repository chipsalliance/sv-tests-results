module render_rects_fill (

      
	clk,

      
	rst,

      
	oe,

      
	start,

      
	x,

      
	y,

      
	cidx,

      
	drawing,

      
	done

      
);

      
	parameter CORDW = 16;

      
	parameter CIDXW = 4;

      
	parameter SCALE = 1;

      
	input wire clk;

      
	input wire rst;

      
	input wire oe;

      
	input wire start;

      
	output wire signed [CORDW - 1:0] x;

      
	output wire signed [CORDW - 1:0] y;

      
	output reg [CIDXW - 1:0] cidx;

      
	output wire drawing;

      
	output reg done;

      
	localparam SHAPE_CNT = 15;

      
	reg [3:0] shape_id;

      
	reg signed [CORDW - 1:0] vx0;

      
	reg signed [CORDW - 1:0] vy0;

      
	reg signed [CORDW - 1:0] vx1;

      
	reg signed [CORDW - 1:0] vy1;

      
	reg draw_start;

      
	wire draw_done;

      
	reg [31:0] state;

      
	always @(posedge clk) begin

      
		case (state)

      
			32'd1: begin

      
				draw_start <= 1;

      
				state <= 32'd2;

      
				vx0 <= 80 + (4 * shape_id);

      
				vy0 <= 20 + (4 * shape_id);

      
				vx1 <= 160 + (4 * shape_id);

      
				vy1 <= 100 + (4 * shape_id);

      
				cidx <= shape_id + 1;

      
			end

      
			32'd2: begin

      
				draw_start <= 0;

      
				if (draw_done)

      
					if (shape_id == 14)

      
						state <= 32'd3;

      
					else begin

      
						shape_id <= shape_id + 1;

      
						state <= 32'd1;

      
					end

      
			end

      
			32'd3: state <= 32'd3;

      
			default:

      
				if (start)

      
					state <= 32'd1;

      
		endcase

      
		if (rst)

      
			state <= 32'd0;

      
	end

      
	draw_rectangle_fill #(.CORDW(CORDW)) draw_rectangle_inst(

      
		.clk(clk),

      
		.rst(rst),

      
		.start(draw_start),

      
		.oe(oe),

      
		.x0(vx0 * SCALE),

      
		.y0(vy0 * SCALE),

      
		.x1(vx1 * SCALE),

      
		.y1(vy1 * SCALE),

      
		.x(x),

      
		.y(y),

      
		.drawing(drawing),

      
		.busy(),

      
		.done(draw_done)

      
	);

      
	always @(*) done = state == 32'd3;

      
endmodule
