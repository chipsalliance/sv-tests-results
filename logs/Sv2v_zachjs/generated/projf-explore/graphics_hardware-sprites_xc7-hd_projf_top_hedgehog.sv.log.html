module top_hedgehog (

      
	clk_100m,

      
	btn_rst_n,

      
	hdmi_tx_ch0_p,

      
	hdmi_tx_ch0_n,

      
	hdmi_tx_ch1_p,

      
	hdmi_tx_ch1_n,

      
	hdmi_tx_ch2_p,

      
	hdmi_tx_ch2_n,

      
	hdmi_tx_clk_p,

      
	hdmi_tx_clk_n

      
);

      
	input wire clk_100m;

      
	input wire btn_rst_n;

      
	output wire hdmi_tx_ch0_p;

      
	output wire hdmi_tx_ch0_n;

      
	output wire hdmi_tx_ch1_p;

      
	output wire hdmi_tx_ch1_n;

      
	output wire hdmi_tx_ch2_p;

      
	output wire hdmi_tx_ch2_n;

      
	output wire hdmi_tx_clk_p;

      
	output wire hdmi_tx_clk_n;

      
	wire clk_pix;

      
	wire clk_pix_5x;

      
	wire clk_pix_locked;

      
	clock_720p clock_pix_inst(

      
		.clk_100m(clk_100m),

      
		.rst(!btn_rst_n),

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(clk_pix_5x),

      
		.clk_pix_locked(clk_pix_locked)

      
	);

      
	reg rst_pix;

      
	always @(*) rst_pix = !clk_pix_locked;

      
	localparam CORDW = 16;

      
	wire [15:0] sx;

      
	wire [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_720p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(rst_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam H_RES = 1280;

      
	localparam CHANW = 4;

      
	localparam COLRW = 12;

      
	localparam CIDXW = 4;

      
	localparam TRANS_INDX = 'h9;

      
	localparam PAL_FILE = "hedgehog_4b.mem";

      
	localparam SX_OFFS = 3;

      
	localparam SPR_WIDTH = 32;

      
	localparam SPR_HEIGHT = 20;

      
	localparam SPR_SCALE = 3;

      
	localparam SPR_DRAWW = 256;

      
	localparam SPR_SPX = 4;

      
	localparam SPR_FILE = "hedgehog.mem";

      
	reg signed [15:0] sprx;

      
	reg signed [15:0] spry;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			if (sprx <= -SPR_DRAWW)

      
				sprx <= H_RES;

      
			else

      
				sprx <= sprx - SPR_SPX;

      
		if (rst_pix) begin

      
			sprx <= H_RES;

      
			spry <= 320;

      
		end

      
	end

      
	wire drawing;

      
	wire [3:0] spr_pix_indx;

      
	sprite #(

      
		.CORDW(CORDW),

      
		.H_RES(H_RES),

      
		.SX_OFFS(SX_OFFS),

      
		.SPR_FILE(SPR_FILE),

      
		.SPR_WIDTH(SPR_WIDTH),

      
		.SPR_HEIGHT(SPR_HEIGHT),

      
		.SPR_SCALE(SPR_SCALE),

      
		.SPR_DATAW(CIDXW)

      
	) sprite_hedgehog(

      
		.clk(clk_pix),

      
		.rst(rst_pix),

      
		.line(line),

      
		.sx(sx),

      
		.sy(sy),

      
		.sprx(sprx),

      
		.spry(spry),

      
		.pix(spr_pix_indx),

      
		.drawing(drawing)

      
	);

      
	wire [11:0] spr_pix_colr;

      
	clut_simple #(

      
		.COLRW(COLRW),

      
		.CIDXW(CIDXW),

      
		.F_PAL(PAL_FILE)

      
	) clut_instance(

      
		.clk_write(clk_pix),

      
		.clk_read(clk_pix),

      
		.we(0),

      
		.cidx_write(0),

      
		.cidx_read(spr_pix_indx),

      
		.colr_in(0),

      
		.colr_out(spr_pix_colr)

      
	);

      
	reg drawing_t1;

      
	always @(posedge clk_pix) drawing_t1 <= drawing && (spr_pix_indx != TRANS_INDX);

      
	reg [11:0] bg_colr;

      
	always @(posedge clk_pix)

      
		if (line)

      
			if (sy == 0)

      
				bg_colr <= 12'h239;

      
			else if (sy == 120)

      
				bg_colr <= 12'h24a;

      
			else if (sy == 220)

      
				bg_colr <= 12'h25b;

      
			else if (sy == 300)

      
				bg_colr <= 12'h26c;

      
			else if (sy == 360)

      
				bg_colr <= 12'h27d;

      
			else if (sy == 410)

      
				bg_colr <= 12'h29e;

      
			else if (sy == 450)

      
				bg_colr <= 12'h2bf;

      
			else if (sy == 480)

      
				bg_colr <= 12'h260;

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) {paint_r, paint_g, paint_b} = (drawing_t1 ? spr_pix_colr : bg_colr);

      
	reg [7:0] dvi_r;

      
	reg [7:0] dvi_g;

      
	reg [7:0] dvi_b;

      
	reg dvi_hsync;

      
	reg dvi_vsync;

      
	reg dvi_de;

      
	always @(posedge clk_pix) begin

      
		dvi_hsync <= hsync;

      
		dvi_vsync <= vsync;

      
		dvi_de <= de;

      
		dvi_r <= {2 {paint_r}};

      
		dvi_g <= {2 {paint_g}};

      
		dvi_b <= {2 {paint_b}};

      
	end

      
	wire tmds_ch0_serial;

      
	wire tmds_ch1_serial;

      
	wire tmds_ch2_serial;

      
	wire tmds_clk_serial;

      
	dvi_generator dvi_out(

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(clk_pix_5x),

      
		.rst_pix(rst_pix),

      
		.de(dvi_de),

      
		.data_in_ch0(dvi_b),

      
		.data_in_ch1(dvi_g),

      
		.data_in_ch2(dvi_r),

      
		.ctrl_in_ch0({dvi_vsync, dvi_hsync}),

      
		.ctrl_in_ch1(2'b00),

      
		.ctrl_in_ch2(2'b00),

      
		.tmds_ch0_serial(tmds_ch0_serial),

      
		.tmds_ch1_serial(tmds_ch1_serial),

      
		.tmds_ch2_serial(tmds_ch2_serial),

      
		.tmds_clk_serial(tmds_clk_serial)

      
	);

      
	tmds_out tmds_ch0(

      
		.tmds(tmds_ch0_serial),

      
		.pin_p(hdmi_tx_ch0_p),

      
		.pin_n(hdmi_tx_ch0_n)

      
	);

      
	tmds_out tmds_ch1(

      
		.tmds(tmds_ch1_serial),

      
		.pin_p(hdmi_tx_ch1_p),

      
		.pin_n(hdmi_tx_ch1_n)

      
	);

      
	tmds_out tmds_ch2(

      
		.tmds(tmds_ch2_serial),

      
		.pin_p(hdmi_tx_ch2_p),

      
		.pin_n(hdmi_tx_ch2_n)

      
	);

      
	tmds_out tmds_clk(

      
		.tmds(tmds_clk_serial),

      
		.pin_p(hdmi_tx_clk_p),

      
		.pin_n(hdmi_tx_clk_n)

      
	);

      
endmodule
