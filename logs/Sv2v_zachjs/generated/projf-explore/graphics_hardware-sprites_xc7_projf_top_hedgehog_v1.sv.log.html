module top_hedgehog_v1 (

      
	clk_100m,

      
	btn_rst,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 16;

      
	wire signed [15:0] sx;

      
	wire signed [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam SPR_WIDTH = 32;

      
	localparam SPR_HEIGHT = 20;

      
	localparam SPR_SCALE_X = 4;

      
	localparam SPR_SCALE_Y = 4;

      
	localparam COLR_BITS = 4;

      
	localparam SPR_TRANS = 9;

      
	localparam SPR_FRAMES = 1;

      
	localparam SPR_FILE = "hedgehog.mem";

      
	localparam SPR_PALETTE = "hedgehog_palette.mem";

      
	localparam SPR_PIXELS = 640;

      
	localparam SPR_DEPTH = 640;

      
	localparam SPR_ADDRW = 10;

      
	reg spr_start;

      
	wire spr_drawing;

      
	wire [3:0] spr_pix;

      
	wire [3:0] spr_rom_data;

      
	wire [9:0] spr_rom_addr;

      
	rom_sync #(

      
		.WIDTH(COLR_BITS),

      
		.DEPTH(SPR_DEPTH),

      
		.INIT_F(SPR_FILE)

      
	) spr_rom(

      
		.clk(clk_pix),

      
		.addr(spr_rom_addr),

      
		.data(spr_rom_data)

      
	);

      
	localparam H_RES = 640;

      
	localparam SPR_SPEED_X = 2;

      
	reg signed [15:0] sprx;

      
	reg signed [15:0] spry;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			sprx <= (sprx > -132 ? sprx - SPR_SPEED_X : H_RES);

      
		if (!clk_locked) begin

      
			sprx <= H_RES;

      
			spry <= 200;

      
		end

      
	end

      
	always @(*) spr_start = line && (sy == spry);

      
	sprite #(

      
		.WIDTH(SPR_WIDTH),

      
		.HEIGHT(SPR_HEIGHT),

      
		.COLR_BITS(COLR_BITS),

      
		.SCALE_X(SPR_SCALE_X),

      
		.SCALE_Y(SPR_SCALE_Y),

      
		.ADDRW(SPR_ADDRW)

      
	) spr_instance(

      
		.clk(clk_pix),

      
		.rst(!clk_locked),

      
		.start(spr_start),

      
		.sx(sx),

      
		.sprx(sprx),

      
		.data_in(spr_rom_data),

      
		.pos(spr_rom_addr),

      
		.pix(spr_pix),

      
		.drawing(spr_drawing),

      
		.done()

      
	);

      
	wire [11:0] clut_colr;

      
	rom_async #(

      
		.WIDTH(12),

      
		.DEPTH(11),

      
		.INIT_F(SPR_PALETTE)

      
	) clut(

      
		.addr(spr_pix),

      
		.data(clut_colr)

      
	);

      
	reg spr_trans;

      
	reg [3:0] red_spr;

      
	reg [3:0] green_spr;

      
	reg [3:0] blue_spr;

      
	reg [3:0] red_bg;

      
	reg [3:0] green_bg;

      
	reg [3:0] blue_bg;

      
	reg [3:0] red;

      
	reg [3:0] green;

      
	reg [3:0] blue;

      
	always @(*) begin

      
		spr_trans = spr_pix == SPR_TRANS;

      
		{red_spr, green_spr, blue_spr} = clut_colr;

      
		{red_bg, green_bg, blue_bg} = 12'h260;

      
		red = (spr_drawing && !spr_trans ? red_spr : red_bg);

      
		green = (spr_drawing && !spr_trans ? green_spr : green_bg);

      
		blue = (spr_drawing && !spr_trans ? blue_spr : blue_bg);

      
	end

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync;

      
		vga_vsync <= vsync;

      
		vga_r <= (de ? red : 4'h0);

      
		vga_g <= (de ? green : 4'h0);

      
		vga_b <= (de ? blue : 4'h0);

      
	end
