	localparam FB_WIDTH = 160;

	localparam FB_DATAW = 4;

	reg [31:0] state;

		case (state)

      
			32'd1:

      
				if (cnt_draw < 159) begin

      
					fb_addr_write <= fb_addr_write + 1;

      
					cnt_draw <= cnt_draw + 1;

      
				end

					cnt_draw <= 0;

      
					state <= 32'd2;

      
				end

      
			32'd2:

      
				if (cnt_draw < 119) begin

      
					fb_addr_write <= fb_addr_write + FB_WIDTH;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					fb_addr_write <= 0;

      
					cnt_draw <= 0;

      
					state <= 32'd4;

      
				end

      
			32'd4:

      
				if (cnt_draw < 119) begin

      
					fb_addr_write <= fb_addr_write + FB_WIDTH;

      
					cnt_draw <= cnt_draw + 1;

					cnt_draw <= 0;

					fb_we <= 0;

      
					state <= 32'd5;
