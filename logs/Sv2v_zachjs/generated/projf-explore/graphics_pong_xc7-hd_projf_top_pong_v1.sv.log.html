	localparam V_RES_FULL = 750;

      
	localparam H_RES = 1280;

      
	localparam V_RES = 720;

      
	reg animate;

      
	always @(*) animate = (sy == V_RES) && (sx == 0);

      
	localparam B_SIZE = 16;

      
	reg [11:0] bx;

      
	reg [11:0] by;

      
	reg dx;

      
	reg dy;

      
	reg [11:0] spx = 2;

      
	reg [11:0] spy = 2;

      
	reg b_draw;

      
	always @(posedge clk_pix)

      
		if (animate) begin

      
			if (bx >= (H_RES - (spx + B_SIZE))) begin

      
				dx <= 1;

      
				bx <= bx - spx;

      
			end

      
			else if (bx < spx) begin

      
				dx <= 0;

      
				bx <= bx + spx;

      
			end

      
			else

      
				bx <= (dx ? bx - spx : bx + spx);

      
			if (by >= (V_RES - (spy + B_SIZE))) begin

      
				dy <= 1;

      
				by <= by - spy;

      
			end

      
			else if (by < spy) begin

      
				dy <= 0;

      
				by <= by + spy;

      
			end

      
			else

      
				by <= (dy ? by - spy : by + spy);

      
		end

      
	always @(*) b_draw = (((sx >= bx) && (sx < (bx + B_SIZE))) && (sy >= by)) && (sy < (by + B_SIZE));

	reg [7:0] dvi_green;

      
	reg [7:0] dvi_blue;

      
	reg dvi_hsync;

      
	reg dvi_vsync;

		dvi_vsync <= vsync;

      
		dvi_de <= de;

      
		dvi_red <= (!de ? 8'h00 : (b_draw ? 8'hff : 8'h00));

      
		dvi_green <= (!de ? 8'h00 : (b_draw ? 8'hff : 8'h00));

      
		dvi_blue <= (!de ? 8'h00 : (b_draw ? 8'hff : 8'h00));

      
	end
