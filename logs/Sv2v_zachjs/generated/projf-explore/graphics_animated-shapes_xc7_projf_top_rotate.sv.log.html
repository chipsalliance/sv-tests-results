	clk_100m,

	input wire btn_rst;

      
	input wire btn_inc;

      
	input wire btn_dec;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

		.clk_locked(clk_locked)

	wire vsync;

      
	wire frame;

		.de(),

	xd xd_frame(

		.in(btn_inc),

      
		.out(),

      
		.ondn(),

      
		.onup(sig_inc)

      
	);

      
	debounce deb_dec(

      
		.clk(clk_100m),

      
		.in(btn_dec),

      
		.out(),

      
		.ondn(),

      
		.onup(sig_dec)

      
	);

      
	localparam FB_WIDTH = 320;

      
	localparam FB_HEIGHT = 180;

      
	localparam FB_CIDXW = 4;

      
	localparam FB_CHANW = 4;

      
	localparam FB_SCALE = 2;

      
	localparam FB_IMAGE = "";

      
	localparam FB_PALETTE = "16_colr_4bit_palette.mem";

      
	localparam FB_BGIDX = 4'h1;

      
	reg fb_we;

      
	wire fb_busy;

      
	wire fb_wready;

      
	reg signed [15:0] fbx;

      
	reg signed [15:0] fby;

      
	reg [3:0] fb_cidx;

      
	wire [3:0] fb_red;

      
	wire [3:0] fb_green;

      
	wire [3:0] fb_blue;

      
	framebuffer_bram_db #(

      
		.WIDTH(FB_WIDTH),

      
		.HEIGHT(FB_HEIGHT),

      
		.CIDXW(FB_CIDXW),

      
		.CHANW(FB_CHANW),

      
		.SCALE(FB_SCALE),

      
		.F_IMAGE(FB_IMAGE),

      
		.F_PALETTE(FB_PALETTE)

      
	) fb_inst(

      
		.clk_sys(clk_100m),

      
		.clk_pix(clk_pix),

      
		.rst_sys(1'b0),

      
		.rst_pix(1'b0),

      
		.de(((sy >= 60) && (sy < 420)) && (sx >= 0)),

      
		.frame(frame),

      
		.line(line),

      
		.we(fb_we),

      
		.x(fbx),

      
		.y(fby),

      
		.cidx(fb_cidx),

      
		.bgidx(FB_BGIDX),

      
		.clear(1'b1),

      
		.busy(fb_busy),

      
		.wready(fb_wready),

      
		.clip(),

      
		.red(fb_red),

      
		.green(fb_green),

      
		.blue(fb_blue)

      
	);

      
	localparam ANGLEW = 8;

      
	reg [7:0] angle;

      
	reg [15:0] rot_xi;

      
	reg [15:0] rot_yi;

      
	wire [15:0] rot_x;

      
	wire [15:0] rot_y;

      
	reg rot_start;

      
	wire rot_done;

      
	always @(posedge clk_100m) begin

      
		if (sig_inc)

      
			angle <= angle + 2;

      
		if (sig_dec)

      
			angle <= angle - 2;

      
	end

      
	rotate_xy #(

      
		.CORDW(CORDW),

      
		.ANGLEW(ANGLEW)

      
	) rotate_xy_inst(

      
		.clk(clk_100m),

      
		.rst(1'b0),

      
		.start(rot_start),

      
		.angle(angle),

      
		.xi(rot_xi),

      
		.yi(rot_yi),

      
		.x(rot_x),

      
		.y(rot_y),

      
		.done(rot_done)

      
	);

      
	reg signed [15:0] vx0;

      
	reg signed [15:0] vy0;

      
	reg signed [15:0] vx1;

      
	reg signed [15:0] vy1;

      
	reg signed [15:0] vx2;

      
	reg signed [15:0] vy2;

      
	reg signed [15:0] offs_x;

      
	reg signed [15:0] offs_y;

      
	wire signed [15:0] fbx_fill;

      
	wire signed [15:0] fby_fill;

      
	wire signed [15:0] fbx_outline;

      
	wire signed [15:0] fby_outline;

      
	reg drawing;

      
	reg draw_start_fill;

      
	wire drawing_fill;

      
	wire draw_done_fill;

      
	reg draw_start_outline;

      
	wire drawing_outline;

      
	wire draw_done_outline;

      
	reg [31:0] state;

      
	always @(posedge clk_100m) begin

      
		rot_start <= 0;

      
		case (state)

      
			32'd1:

      
				if (fb_wready) begin

      
					state <= 32'd2;

      
					vx0 <= 0;

      
					vy0 <= -40;

      
					vx1 <= -30;

      
					vy1 <= 20;

      
					vx2 <= 45;

      
					vy2 <= 70;

      
					offs_x <= 160;

      
					offs_y <= 90;

      
					fb_cidx <= 4'h9;

      
				end

      
			32'd2: begin

      
				rot_xi <= vx0;

      
				rot_yi <= vy0;

      
				rot_start <= 1;

      
				state <= 32'd3;

      
			end

      
			32'd3:

      
				if (rot_done) begin

      
					vx0 <= rot_x + offs_x;

      
					vy0 <= rot_y + offs_y;

      
					rot_xi <= vx1;

      
					rot_yi <= vy1;

      
					rot_start <= 1;

      
					state <= 32'd4;

      
				end

      
			32'd4:

      
				if (rot_done) begin

      
					vx1 <= rot_x + offs_x;

      
					vy1 <= rot_y + offs_y;

      
					rot_xi <= vx2;

      
					rot_yi <= vy2;

      
					rot_start <= 1;

      
					state <= 32'd5;

      
				end

      
			32'd5:

      
				if (rot_done) begin

      
					vx2 <= rot_x + offs_x;

      
					vy2 <= rot_y + offs_y;

      
					draw_start_fill <= 1;

      
					state <= 32'd6;

      
				end

      
			32'd6: begin

      
				draw_start_fill <= 0;

      
				if (draw_done_fill) begin

      
					state <= 32'd7;

      
					draw_start_outline <= 1;

      
					fb_cidx <= 4'h8;

      
				end

      
			end

      
			32'd7: begin

      
				draw_start_outline <= 0;

      
				if (draw_done_outline)

      
					state <= 32'd8;

      
			end

      
			32'd8: state <= 32'd0;

      
			default:

      
				if (frame_sys)

      
					state <= 32'd1;

      
		endcase

      
	end

      
	always @(*) drawing = drawing_fill || drawing_outline;

      
	draw_triangle_fill #(.CORDW(CORDW)) draw_triangle_fill_inst(

      
		.clk(clk_100m),

      
		.rst(1'b0),

		.busy(),

      
		.done(draw_done_fill)

      
	);

		.rst(1'b0),

		.y0(vy0),

		.y1(vy1),

      
		.x2(vx2),

      
		.y2(vy2),

      
		.x(fbx_outline),

      
		.y(fby_outline),

      
		.drawing(drawing_outline),

      
		.busy(),

      
		.done(draw_done_outline)

      
	);

      
	always @(posedge clk_100m) begin

      
		fb_we <= drawing;

      
		fbx <= (drawing_fill ? fbx_fill : fbx_outline);

      
		fby <= (drawing_fill ? fby_fill : fby_outline);

      
	end

		hsync_p1 <= hsync;

      
		vsync_p1 <= vsync;

      
	end

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync_p1;

      
		vga_vsync <= vsync_p1;

      
		vga_r <= fb_red;

      
		vga_g <= fb_green;

      
		vga_b <= fb_blue;

      
	end
