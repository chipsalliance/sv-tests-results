module top_rasterbars (

      
	clk_pix,

      
	sim_rst,

      
	sdl_sx,

      
	sdl_sy,

      
	sdl_de,

      
	sdl_r,

      
	sdl_g,

      
	sdl_b

      
);

      
	parameter CORDW = 10;

      
	input wire clk_pix;

      
	input wire sim_rst;

      
	output reg [CORDW - 1:0] sdl_sx;

      
	output reg [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	wire [CORDW - 1:0] sx;

      
	wire [CORDW - 1:0] sy;

      
	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(sim_rst),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de)

      
	);

      
	localparam V_RES_FULL = 525;

      
	localparam H_RES = 640;

      
	localparam START_COLR = 12'h126;

      
	localparam COLR_NUM = 10;

      
	localparam LINE_NUM = 2;

      
	reg [11:0] bar_colr;

      
	reg bar_inc;

      
	reg [4:0] cnt_colr;

      
	reg [1:0] cnt_line;

      
	always @(posedge clk_pix)

      
		if (sx == H_RES)

      
			if (sy == 524) begin

      
				bar_colr <= START_COLR;

      
				bar_inc <= 1;

      
				cnt_colr <= 0;

      
				cnt_line <= 0;

      
			end

      
			else if (cnt_line == 1) begin

      
				cnt_line <= 0;

      
				if (cnt_colr == 9) begin

      
					bar_inc <= ~bar_inc;

      
					cnt_colr <= 0;

      
				end

      
				else begin

      
					bar_colr <= (bar_inc ? bar_colr + 12'h111 : bar_colr - 12'h111);

      
					cnt_colr <= cnt_colr + 1;

      
				end

      
			end

      
			else

      
				cnt_line <= cnt_line + 1;

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) {paint_r, paint_g, paint_b} = bar_colr;

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_r <= {2 {paint_r}};

      
		sdl_g <= {2 {paint_g}};

      
		sdl_b <= {2 {paint_b}};

      
	end

      
endmodule
