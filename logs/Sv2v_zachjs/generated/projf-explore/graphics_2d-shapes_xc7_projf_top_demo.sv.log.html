		.addr_read(fb_addr_read),

      
		.data_in(fb_colr_write),

      
		.data_out(fb_colr_read)

	always @(*) fb_we = fb_we_sr[0] && !clip;

		if (line0_sys)

      
			cnt_lb_line <= 0;

      
		else if (line_sys)

	always @(posedge clk_sys) begin

			lb_line <= 1;

			lb_line <= 0;
