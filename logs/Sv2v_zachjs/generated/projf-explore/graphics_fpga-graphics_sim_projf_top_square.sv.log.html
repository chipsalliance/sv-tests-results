module top_square (

      
	clk_pix,

      
	sim_rst,

	sdl_r,

      
	sdl_g,

      
	sdl_b

	parameter CORDW = 10;

      
	input wire clk_pix;

      
	input wire sim_rst;

      
	output reg [CORDW - 1:0] sdl_sx;

      
	output reg [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg [7:0] sdl_r;

	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(sim_rst),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de)

      
	);

      
	reg square;

      
	always @(*) square = ((sx > 220) && (sx < 420)) && ((sy > 140) && (sy < 340));

	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) begin

      
		paint_r = (square ? 4'hf : 4'h1);

      
		paint_g = (square ? 4'hf : 4'h3);

      
		paint_b = (square ? 4'hf : 4'h7);

      
	end

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;
