module top_pong (

      
	clk_100m,

      
	btn_rst,

      
	btn_up,

      
	btn_ctrl,

      
	btn_dn,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	input wire btn_up;

      
	input wire btn_ctrl;

      
	input wire btn_dn;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 10;

      
	wire [9:0] sx;

      
	wire [9:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de)

      
	);

      
	localparam H_RES_FULL = 800;

      
	localparam V_RES_FULL = 525;

      
	localparam H_RES = 640;

      
	localparam V_RES = 480;

      
	reg animate;

      
	always @(*) animate = (sy == V_RES) && (sx == 0);

      
	wire sig_ctrl;

      
	wire move_up;

      
	wire move_dn;

      
	debounce deb_ctrl(

      
		.clk(clk_pix),

      
		.in(btn_ctrl),

      
		.out(),

      
		.ondn(),

      
		.onup(sig_ctrl)

      
	);

      
	debounce deb_up(

      
		.clk(clk_pix),

      
		.in(btn_up),

      
		.out(move_up),

      
		.ondn(),

      
		.onup()

      
	);

      
	debounce deb_dn(

      
		.clk(clk_pix),

      
		.in(btn_dn),

      
		.out(move_dn),

      
		.ondn(),

      
		.onup()

      
	);

      
	localparam B_SIZE = 8;

      
	reg [9:0] bx;

      
	reg [9:0] by;

      
	reg dx;

      
	reg dy;

      
	reg [9:0] spx;

      
	reg [9:0] spy;

      
	reg lft_col;

      
	reg rgt_col;

      
	reg b_draw;

      
	localparam P_H = 40;

      
	localparam P_W = 10;

      
	localparam P_SP = 4;

      
	localparam P_OFFS = 32;

      
	reg [9:0] p1y;

      
	reg [9:0] p2y;

      
	reg p1_draw;

      
	reg p2_draw;

      
	reg p1_col;

      
	reg p2_col;

      
	reg [31:0] state;

      
	reg [31:0] state_next;

      
	always @(*)

      
		case (state)

      
			32'd0: state_next = 32'd1;

      
			32'd1: state_next = (sig_ctrl ? 32'd2 : 32'd1);

      
			32'd2: state_next = (sig_ctrl ? 32'd3 : 32'd2);

      
			32'd3: state_next = (lft_col || rgt_col ? 32'd4 : 32'd3);

      
			32'd4: state_next = (sig_ctrl ? 32'd2 : 32'd4);

      
			default: state_next = 32'd1;

      
		endcase

      
	always @(posedge clk_pix) state <= state_next;

      
	always @(posedge clk_pix)

      
		if ((state == 32'd0) || (state == 32'd2)) begin

      
			p1y <= 220;

      
			p2y <= 220;

      
		end

      
		else if (animate && (state != 32'd4)) begin

      
			if (state == 32'd3) begin

      
				if (move_up)

      
					if (p1y > P_SP)

      
						p1y <= p1y - P_SP;

      
				if (move_dn)

      
					if (p1y < 436)

      
						p1y <= p1y + P_SP;

      
			end

      
			else if ((p1y + 22) < (by + 4)) begin

      
				if (p1y < 438)

      
					p1y <= p1y + P_SP;

      
			end

      
			else if ((p1y + 20) > (by + 6))

      
				if (p1y > P_SP)

      
					p1y <= p1y - P_SP;

      
			if ((p2y + 22) < (by + 4)) begin

      
				if (p2y < 438)

      
					p2y <= p2y + P_SP;

      
			end

      
			else if ((p2y + 20) > (by + 6))

      
				if (p2y > P_SP)

      
					p2y <= p2y - P_SP;

      
		end

      
	always @(*) begin

      
		p1_draw = (((sx >= P_OFFS) && (sx < 42)) && (sy >= p1y)) && (sy < (p1y + P_H));

      
		p2_draw = (((sx >= 598) && (sx < 608)) && (sy >= p2y)) && (sy < (p2y + P_H));

      
	end

      
	always @(posedge clk_pix)

      
		if (animate) begin

      
			p1_col <= 0;

      
			p2_col <= 0;

      
		end

      
		else if (b_draw) begin

      
			if (p1_draw)

      
				p1_col <= 1;

      
			if (p2_draw)

      
				p2_col <= 1;

      
		end

      
	localparam SPEED_STEP = 5;

      
	reg [2:0] cnt_sp;

      
	always @(posedge clk_pix)

      
		if (state == 32'd0) begin

      
			spx <= 6;

      
			spy <= 4;

      
		end

      
		else if (state == 32'd2) begin

      
			spx <= 4;

      
			spy <= 2;

      
		end

      
		else if (((state == 32'd3) && animate) && (p1_col || p2_col))

      
			if (cnt_sp == 4) begin

      
				spx <= spx + 1;

      
				spy <= spy + 1;

      
				cnt_sp <= 0;

      
			end

      
			else

      
				cnt_sp <= cnt_sp + 1;

      
	always @(posedge clk_pix)

      
		if ((state == 32'd0) || (state == 32'd2)) begin

      
			bx <= 316;

      
			by <= 236;

      
			dx <= 0;

      
			dy <= ~dy;

      
			lft_col <= 0;

      
			rgt_col <= 0;

      
		end

      
		else if (animate && (state != 32'd4)) begin

      
			if (p1_col) begin

      
				dx <= 0;

      
				bx <= bx + spx;

      
				dy <= ((by + 4) < (p1y + 20) ? 1 : 0);

      
			end

      
			else if (p2_col) begin

      
				dx <= 1;

      
				bx <= bx - spx;

      
				dy <= ((by + 4) < (p1y + 20) ? 1 : 0);

      
			end

      
			else if (bx >= (H_RES - (spx + B_SIZE)))

      
				rgt_col <= 1;

      
			else if (bx < spx)

      
				lft_col <= 1;

      
			else

      
				bx <= (dx ? bx - spx : bx + spx);

      
			if (by >= (V_RES - (spy + B_SIZE))) begin

      
				dy <= 1;

      
				by <= by - spy;

      
			end

      
			else if (by < spy) begin

      
				dy <= 0;

      
				by <= by + spy;

      
			end

      
			else

      
				by <= (dy ? by - spy : by + spy);

      
		end

      
	always @(*) b_draw = (((sx >= bx) && (sx < (bx + B_SIZE))) && (sy >= by)) && (sy < (by + B_SIZE));

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync;

      
		vga_vsync <= vsync;

      
		vga_r <= (de && ((b_draw | p1_draw) | p2_draw) ? 4'hf : 4'h0);

      
		vga_g <= (de && ((b_draw | p1_draw) | p2_draw) ? 4'hf : 4'h0);

      
		vga_b <= (de && ((b_draw | p1_draw) | p2_draw) ? 4'hf : 4'h0);

      
	end
