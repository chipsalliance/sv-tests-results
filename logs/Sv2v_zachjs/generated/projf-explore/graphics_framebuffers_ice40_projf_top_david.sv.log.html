module top_david (

      
	clk_12m,

      
	btn_rst,

      
	dvi_clk,

      
	dvi_hsync,

      
	dvi_vsync,

      
	dvi_de,

      
	dvi_r,

      
	dvi_g,

      
	dvi_b

      
);

      
	input wire clk_12m;

      
	input wire btn_rst;

      
	output wire dvi_clk;

      
	output wire dvi_hsync;

      
	output wire dvi_vsync;

      
	output wire dvi_de;

      
	output wire [3:0] dvi_r;

      
	output wire [3:0] dvi_g;

      
	output wire [3:0] dvi_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_12m),

      
		.rst(btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 16;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_locked),

      
		.sx(),

      
		.sy(),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam FB_WIDTH = 160;

      
	localparam FB_HEIGHT = 120;

      
	localparam FB_CIDXW = 4;

      
	localparam FB_CHANW = 4;

      
	localparam FB_SCALE = 4;

      
	localparam FB_IMAGE = "../res/david/david.mem";

      
	localparam FB_PALETTE = "../res/david/david_palette.mem";

      
	reg fb_we;

      
	reg signed [15:0] fbx;

      
	reg signed [15:0] fby;

      
	reg [3:0] fb_cidx;

      
	wire [3:0] fb_red;

      
	wire [3:0] fb_green;

      
	wire [3:0] fb_blue;

      
	framebuffer_bram #(

      
		.WIDTH(FB_WIDTH),

      
		.HEIGHT(FB_HEIGHT),

      
		.CIDXW(FB_CIDXW),

      
		.CHANW(FB_CHANW),

      
		.SCALE(FB_SCALE),

      
		.F_IMAGE(FB_IMAGE),

      
		.F_PALETTE(FB_PALETTE)

      
	) fb_inst(

      
		.clk_sys(clk_pix),

      
		.clk_pix(clk_pix),

      
		.rst_sys(1'b0),

      
		.rst_pix(1'b0),

      
		.de(de),

      
		.frame(frame),

      
		.line(line),

      
		.we(fb_we),

      
		.x(fbx),

      
		.y(fby),

      
		.cidx(fb_cidx),

      
		.clip(),

      
		.busy(),

      
		.red(fb_red),

      
		.green(fb_green),

      
		.blue(fb_blue)

      
	);

      
	reg [31:0] state;

      
	always @(posedge clk_pix) begin

      
		case (state)

      
			32'd1:

      
				if (fbx < 159)

      
					fbx <= fbx + 1;

      
				else

      
					state <= 32'd2;

      
			32'd2:

      
				if (fby < 119)

      
					fby <= fby + 1;

      
				else begin

      
					fbx <= 0;

      
					fby <= 0;

      
					state <= 32'd4;

      
				end

      
			32'd4:

      
				if (fby < 119)

      
					fby <= fby + 1;

      
				else

      
					state <= 32'd3;

      
			32'd3:

      
				if (fbx < 159)

      
					fbx <= fbx + 1;

      
				else begin

      
					fb_we <= 0;

      
					state <= 32'd5;

      
				end

      
			32'd0:

      
				if (frame) begin

      
					fbx <= 0;

      
					fby <= 0;

      
					fb_cidx <= 4'h0;

      
					fb_we <= 1;

      
					state <= 32'd1;

      
				end

      
			default: state <= 32'd5;

      
		endcase

      
		if (!clk_locked)

      
			state <= 32'd0;

      
	end

      
	reg hsync_p1;

      
	reg vsync_p1;

      
	reg de_p1;

      
	always @(posedge clk_pix) begin

      
		hsync_p1 <= hsync;

      
		vsync_p1 <= vsync;

      
		de_p1 <= de;

      
	end

      
	SB_IO #(.PIN_TYPE(6'b010000)) dvi_clk_io(

      
		.PACKAGE_PIN(dvi_clk),

      
		.OUTPUT_CLK(clk_pix),

      
		.D_OUT_0(1'b0),

      
		.D_OUT_1(1'b1)

      
	);

      
	SB_IO #(.PIN_TYPE(6'b010100)) dvi_signal_io[14:0](

      
		.PACKAGE_PIN({dvi_hsync, dvi_vsync, dvi_de, dvi_r, dvi_g, dvi_b}),

      
		.OUTPUT_CLK(clk_pix),

      
		.D_OUT_0({hsync_p1, vsync_p1, de_p1, fb_red, fb_green, fb_blue}),

      
		.D_OUT_1()

      
	);
