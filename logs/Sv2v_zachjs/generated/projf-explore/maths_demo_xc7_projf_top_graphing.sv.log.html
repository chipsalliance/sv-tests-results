module top_graphing (

      
	clk_100m,

      
	btn_rst,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 12;

      
	wire hsync;

      
	wire vsync;

      
	wire [11:0] sx;

      
	wire [11:0] sy;

      
	wire de;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(),

      
		.line()

      
	);

      
	wire draw;

      
	reg axes;

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync;

      
		vga_vsync <= vsync;

      
		vga_r <= (!de ? 4'h0 : (axes ? 4'hc : (draw ? 4'h3 : 4'h2)));

      
		vga_g <= (!de ? 4'h0 : (axes ? 4'hc : (draw ? 4'hb : 4'h2)));

      
		vga_b <= (!de ? 4'h0 : (axes ? 4'hc : (draw ? 4'hc : 4'h2)));

      
	end

      
	reg signed [11:0] x;

      
	reg signed [11:0] y;

      
	localparam X_OFFS = 320;

      
	localparam Y_OFFS = 239;

      
	always @(posedge clk_pix) begin

      
		x <= (sx - X_OFFS) + 4;

      
		y <= Y_OFFS - sy;

      
	end

      
	reg x_axis;

      
	reg y_axis;

      
	always @(*) begin

      
		x_axis = sy == Y_OFFS;

      
		y_axis = sx == X_OFFS;

      
		axes = x_axis || y_axis;

      
	end

      
	func_squared #(.CORDW(CORDW)) func_inst(

      
		.clk(clk_pix),

      
		.x(x),

      
		.y(y),

      
		.r(draw)

      
	);

      
endmodule
