`default_nettype none

      
module top_tinyf_move (

      
	clk_pix,

      
	rst_pix,

      
	sdl_sx,

      
	sdl_sy,

      
	sdl_de,

      
	sdl_frame,

      
	sdl_r,

      
	sdl_g,

      
	sdl_b

      
);

      
	parameter CORDW = 16;

      
	input wire clk_pix;

      
	input wire rst_pix;

      
	output reg signed [CORDW - 1:0] sdl_sx;

      
	output reg signed [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg sdl_frame;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	wire signed [CORDW - 1:0] sx;

      
	wire signed [CORDW - 1:0] sy;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(rst_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam H_RES = 640;

      
	localparam V_RES = 480;

      
	localparam SPR_WIDTH = 8;

      
	localparam SPR_HEIGHT = 8;

      
	localparam SPR_SCALE = 3;

      
	localparam SPR_DATAW = 1;

      
	localparam SPR_DRAWW = 64;

      
	localparam SPR_DRAWH = 64;

      
	localparam SPR_SPX = 4;

      
	localparam SPR_FILE = "../res/sprites/letter_f.mem";

      
	reg signed [CORDW - 1:0] sprx;

      
	reg signed [CORDW - 1:0] spry;

      
	reg dx;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			if (dx == 0) begin

      
				if ((sprx + SPR_DRAWW) >= 768)

      
					dx <= 1;

      
				else

      
					sprx <= sprx + SPR_SPX;

      
			end

      
			else if (sprx <= (-2 * SPR_DRAWW))

      
				dx <= 0;

      
			else

      
				sprx <= sprx - SPR_SPX;

      
		if (rst_pix) begin

      
			sprx <= 288;

      
			spry <= 208;

      
			dx <= 0;

      
		end

      
	end

      
	wire drawing;

      
	wire [0:0] pix;

      
	sprite #(

      
		.CORDW(CORDW),

      
		.H_RES(H_RES),

      
		.SPR_FILE(SPR_FILE),

      
		.SPR_WIDTH(SPR_WIDTH),

      
		.SPR_HEIGHT(SPR_HEIGHT),

      
		.SPR_SCALE(SPR_SCALE),

      
		.SPR_DATAW(SPR_DATAW)

      
	) sprite_f(

      
		.clk(clk_pix),

      
		.rst(rst_pix),

      
		.line(line),

      
		.sx(sx),

      
		.sy(sy),

      
		.sprx(sprx),

      
		.spry(spry),

      
		.pix(pix),

      
		.drawing(drawing)

      
	);

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) begin

      
		paint_r = (drawing && pix ? 4'hf : 4'h1);

      
		paint_g = (drawing && pix ? 4'hc : 4'h3);

      
		paint_b = (drawing && pix ? 4'h0 : 4'h7);

      
	end

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_frame <= frame;

      
		sdl_r <= {2 {paint_r}};

      
		sdl_g <= {2 {paint_g}};

      
		sdl_b <= {2 {paint_b}};

      
	end

      
endmodule
