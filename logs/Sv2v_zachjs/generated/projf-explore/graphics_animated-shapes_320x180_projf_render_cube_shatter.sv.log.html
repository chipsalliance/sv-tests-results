module render_cube_shatter (

      
	clk,

      
	rst,

      
	oe,

	cidx,

      
	drawing,

      
	done

      
);

	parameter CIDXW = 4;

      
	parameter SCALE = 1;

      
	input wire clk;

      
	input wire rst;

      
	input wire oe;

      
	input wire start;

      
	output wire signed [CORDW - 1:0] x;

      
	output wire signed [CORDW - 1:0] y;

      
	output reg [CIDXW - 1:0] cidx;

      
	output wire drawing;

	reg signed [CORDW - 1:0] vx2;

      
	reg signed [CORDW - 1:0] vy2;

      
	reg draw_start;

	localparam WAIT_STEPS = 120;

      
	reg [CORDW - 1:0] offs;

      
	reg [6:0] cnt_wait;

      
	reg dir;

      
	reg [31:0] anim_state;

      
	always @(posedge clk) begin

      
		if (start)

      
			case (anim_state)

      
				32'd0: begin

      
					if (offs < 20)

      
						offs <= (!dir ? offs + 1 : offs - 1);

      
					else

      
						offs <= (!dir ? offs + 3 : offs - 3);

      
					if ((!dir && (offs >= 159)) || (dir && (offs <= 1))) begin

      
						anim_state <= 32'd1;

      
						cnt_wait <= 0;

      
					end

      
				end

      
				default:

      
					if (cnt_wait == 119) begin

      
						anim_state <= 32'd0;

      
						dir <= ~dir;

      
					end

      
					else

      
						cnt_wait <= cnt_wait + 1;

      
			endcase

      
		if (rst) begin

      
			anim_state <= 32'd1;

      
			cnt_wait <= 0;

      
			dir <= 1;

      
		end

      
	end

      
	reg [31:0] state;

      
	always @(posedge clk) begin

      
		case (state)

      
			32'd1: begin

      
				draw_start <= 1;

      
				state <= 32'd2;

      
				case (shape_id)

      
					'd0: begin

      
						vx0 <= 160 + offs;

      
						vy0 <= 90;

      
						vx1 <= 240 + offs;

      
						vy1 <= 90;

      
						vx2 <= 240 + offs;

					end

						vx0 <= 160 + offs;

      
						vy0 <= 90 + offs;

      
						vx1 <= 240 + offs;

      
						vy1 <= 170 + offs;

      
						vx2 <= 160 + offs;

      
						vy2 <= 170 + offs;

      
						cidx <= (offs == 0 ? 'h4 : 'h5);

      
					end

      
					'd2: begin

      
						vx0 <= 160 - offs;

      
						vy0 <= 90 + offs;

      
						vx1 <= 120 - offs;

      
						vy1 <= 130 + offs;

      
						vx2 <= 160 - offs;

      
						vy2 <= 170 + offs;

      
						cidx <= (offs == 0 ? 'ha : 'h9);

      
					end

      
					'd3: begin

      
						vx0 <= 120 - offs;

      
						vy0 <= 50;

      
						vx1 <= 160 - offs;

      
						vy1 <= 90;

      
						vx2 <= 120 - offs;

      
						vy2 <= 130;

      
						cidx <= (offs == 0 ? 'ha : 'h9);

      
					end

						vx0 <= 120;

						vx1 <= 200;
