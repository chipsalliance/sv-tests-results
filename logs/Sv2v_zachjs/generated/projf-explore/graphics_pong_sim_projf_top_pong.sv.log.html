module top_pong (

      
	clk_pix,

	input wire clk_pix;

      
	input wire sim_rst;

      
	input wire btn_fire;

      
	input wire btn_up;

      
	input wire btn_dn;

      
	output reg [CORDW - 1:0] sdl_sx;

      
	output reg [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	localparam WIN = 4;

      
	localparam SPEEDUP = 5;

      
	localparam BALL_SIZE = 8;

      
	localparam BALL_ISPX = 5;

      
	localparam BALL_ISPY = 3;

      
	localparam PAD_HEIGHT = 48;

      
	localparam PAD_WIDTH = 10;

      
	localparam PAD_OFFS = 32;

      
	localparam PAD_SPY = 3;

      
	wire [CORDW - 1:0] sx;

      
	wire [CORDW - 1:0] sy;

      
	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(sim_rst),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de)

      
	);

      
	localparam H_RES = 640;

      
	localparam V_RES = 480;

      
	reg frame;

      
	always @(*) frame = (sy == V_RES) && (sx == 0);

      
	reg [3:0] score_l;

      
	reg [3:0] score_r;

      
	reg ball;

      
	reg padl;

      
	reg padr;

      
	reg [CORDW - 1:0] ball_x;

      
	reg [CORDW - 1:0] ball_y;

      
	reg [CORDW - 1:0] ball_spx;

      
	reg [CORDW - 1:0] ball_spy;

      
	reg [3:0] shot_cnt;

      
	reg ball_dx;

      
	reg ball_dy;

      
	reg ball_dx_prev;

      
	reg coll_r;

      
	reg coll_l;

      
	reg [CORDW - 1:0] padl_y;

      
	reg [CORDW - 1:0] padr_y;

      
	reg [CORDW - 1:0] ai_y;

      
	reg [CORDW - 1:0] play_y;

      
	always @(*) begin

      
		padl_y = play_y;

      
		padr_y = ai_y;

      
	end

      
	wire sig_fire;

      
	wire sig_up;

      
	wire sig_dn;

      
	debounce deb_fire(

      
		.clk(clk_pix),

      
		.in(btn_fire),

      
		.out(),

      
		.ondn(),

      
		.onup(sig_fire)

      
	);

      
	debounce deb_up(

      
		.clk(clk_pix),

      
		.in(btn_up),

      
		.out(sig_up),

      
		.ondn(),

      
		.onup()

      
	);

      
	debounce deb_dn(

      
		.clk(clk_pix),

      
		.in(btn_dn),

      
		.out(sig_dn),

      
		.ondn(),

      
		.onup()

      
	);

      
	reg [31:0] state;

      
	reg [31:0] state_next;

      
	always @(*) begin

      
		case (state)

      
			32'd0: state_next = 32'd1;

      
			32'd1: state_next = 32'd2;

      
			32'd2: state_next = (sig_fire ? 32'd5 : 32'd2);

      
			32'd3: state_next = (sig_fire ? 32'd1 : 32'd3);

      
			32'd4: state_next = (sig_fire ? 32'd0 : 32'd4);

      
			32'd5:

      
				if (coll_l || coll_r) begin

      
					if ((score_l == WIN) || (score_r == WIN))

      
						state_next = 32'd4;

      
					else

      
						state_next = 32'd3;

      
				end

      
				else

      
					state_next = 32'd5;

      
			default: state_next = 32'd0;

      
		endcase

      
		if (sim_rst)

      
			state_next = 32'd0;

      
	end

      
	always @(posedge clk_pix) state <= state_next;

      
	always @(posedge clk_pix)

      
		if (state == 32'd1)

      
			ai_y <= 216;

      
		else if (frame && (state == 32'd5))

      
			if ((ai_y + 24) < ball_y) begin

      
				if (((ai_y + PAD_HEIGHT) + PAD_SPY) >= 479)

      
					ai_y <= 431;

      
				else

      
					ai_y <= ai_y + PAD_SPY;

      
			end

      
			else if ((ai_y + 24) > (ball_y + BALL_SIZE))

      
				if (ai_y < PAD_SPY)

      
					ai_y <= 0;

      
				else

      
					ai_y <= ai_y - PAD_SPY;

      
	always @(posedge clk_pix)

      
		if (state == 32'd1)

      
			play_y <= 216;

      
		else if (frame && (state == 32'd5))

      
			if (sig_dn) begin

      
				if (((play_y + PAD_HEIGHT) + PAD_SPY) >= 479)

      
					play_y <= 431;

      
				else

      
					play_y <= play_y + PAD_SPY;

      
			end

      
			else if (sig_up)

      
				if (play_y < PAD_SPY)

      
					play_y <= 0;

      
				else

      
					play_y <= play_y - PAD_SPY;

      
	always @(posedge clk_pix) begin

      
		case (state)

      
			32'd0: begin

      
				score_l <= 0;

      
				score_r <= 0;

      
			end

      
			32'd1: begin

      
				coll_l <= 0;

      
				coll_r <= 0;

      
				ball_spx <= BALL_ISPX;

      
				ball_spy <= BALL_ISPY;

      
				shot_cnt <= 0;

      
				ball_y <= 236;

      
				if (coll_r) begin

      
					ball_x <= 590;

      
					ball_dx <= 1;

      
				end

      
				else begin

      
					ball_x <= 42;

      
					ball_dx <= 0;

      
				end

      
			end

      
			32'd5:

      
				if (frame) begin

      
					if (ball_dx == 0) begin

      
						if (((ball_x + BALL_SIZE) + ball_spx) >= 639) begin

      
							ball_x <= 632;

      
							score_l <= score_l + 1;

      
							coll_r <= 1;

      
						end

      
						else

      
							ball_x <= ball_x + ball_spx;

      
					end

      
					else if (ball_x < ball_spx) begin

      
						ball_x <= 0;

      
						score_r <= score_r + 1;

      
						coll_l <= 1;

      
					end

      
					else

      
						ball_x <= ball_x - ball_spx;

      
					if (ball_dy == 0) begin

      
						if (((ball_y + BALL_SIZE) + ball_spy) >= 479)

      
							ball_dy <= 1;

      
						else

      
							ball_y <= ball_y + ball_spy;

      
					end

      
					else if (ball_y < ball_spy)

      
						ball_dy <= 0;

      
					else

      
						ball_y <= ball_y - ball_spy;

      
					if (ball_dx_prev != ball_dx)

      
						shot_cnt <= shot_cnt + 1;

      
					if (shot_cnt == SPEEDUP) begin

      
						ball_spx <= (ball_spx < PAD_WIDTH ? ball_spx + 1 : ball_spx);

      
						ball_spy <= ball_spy + 1;

      
						shot_cnt <= 0;

      
					end

      
				end

      
		endcase

      
		if ((ball && padl) && (ball_dx == 1))

      
			ball_dx <= 0;

      
		if ((ball && padr) && (ball_dx == 0))

      
			ball_dx <= 1;

      
		if (frame)

      
			ball_dx_prev <= ball_dx;

      
	end

      
	always @(*) begin

      
		ball = (((sx >= ball_x) && (sx < (ball_x + BALL_SIZE))) && (sy >= ball_y)) && (sy < (ball_y + BALL_SIZE));

      
		padl = (((sx >= PAD_OFFS) && (sx < 42)) && (sy >= padl_y)) && (sy < (padl_y + PAD_HEIGHT));

      
		padr = (((sx >= 597) && (sx < 607)) && (sy >= padr_y)) && (sy < (padr_y + PAD_HEIGHT));

      
	end

		.clk_pix(clk_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.score_l(score_l),

      
		.score_r(score_r),

      
		.pix(pix_score)

      
	);

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*)

      
		if (pix_score)

      
			{paint_r, paint_g, paint_b} = 12'hf30;

      
		else if (ball)

      
			{paint_r, paint_g, paint_b} = 12'hfc0;

      
		else if (padl || padr)

      
			{paint_r, paint_g, paint_b} = 12'hfff;

      
		else

      
			{paint_r, paint_g, paint_b} = 12'h137;

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_r <= {2 {paint_r}};

      
		sdl_g <= {2 {paint_g}};

      
		sdl_b <= {2 {paint_b}};

      
	end
