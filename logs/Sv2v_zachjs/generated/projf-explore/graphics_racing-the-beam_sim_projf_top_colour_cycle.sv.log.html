`default_nettype none

      
module top_colour_cycle (

      
	clk_pix,

      
	sim_rst,

      
	sdl_sx,

      
	sdl_sy,

      
	sdl_de,

      
	sdl_r,

      
	sdl_g,

      
	sdl_b

      
);

      
	parameter CORDW = 10;

      
	input wire clk_pix;

      
	input wire sim_rst;

      
	output reg [CORDW - 1:0] sdl_sx;

      
	output reg [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	wire [CORDW - 1:0] sx;

      
	wire [CORDW - 1:0] sy;

      
	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(sim_rst),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de)

      
	);

      
	localparam V_RES = 480;

      
	reg frame;

      
	always @(*) frame = (sy == V_RES) && (sx == 0);

      
	localparam FRAME_NUM = 30;

      
	reg [5:0] cnt_frame;

      
	reg [3:0] colr_level;

      
	always @(posedge clk_pix)

      
		if (frame)

      
			if (cnt_frame == 29) begin

      
				cnt_frame <= 0;

      
				colr_level <= colr_level + 1;

      
			end

      
			else

      
				cnt_frame <= cnt_frame + 1;

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) begin

      
		paint_r = sx[7:4];

      
		paint_g = sy[7:4];

      
		paint_b = colr_level;

      
	end

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_r <= {2 {paint_r}};

      
		sdl_g <= {2 {paint_g}};

      
		sdl_b <= {2 {paint_b}};

      
	end

      
endmodule
