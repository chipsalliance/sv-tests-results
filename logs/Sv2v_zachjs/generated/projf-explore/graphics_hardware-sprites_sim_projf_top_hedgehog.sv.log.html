`default_nettype none

      
module top_hedgehog (

      
	clk_pix,

      
	rst_pix,

      
	sdl_sx,

      
	sdl_sy,

      
	sdl_de,

      
	sdl_frame,

      
	sdl_r,

      
	sdl_g,

      
	sdl_b

      
);

      
	parameter CORDW = 16;

      
	input wire clk_pix;

      
	input wire rst_pix;

      
	output reg signed [CORDW - 1:0] sdl_sx;

      
	output reg signed [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg sdl_frame;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	wire signed [CORDW - 1:0] sx;

      
	wire signed [CORDW - 1:0] sy;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(rst_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam H_RES = 640;

      
	localparam CHANW = 4;

      
	localparam COLRW = 12;

      
	localparam CIDXW = 4;

      
	localparam TRANS_INDX = 'h9;

      
	localparam PAL_FILE = "../res/palettes/hedgehog_4b.mem";

      
	localparam SX_OFFS = 3;

      
	localparam SPR_WIDTH = 32;

      
	localparam SPR_HEIGHT = 20;

      
	localparam SPR_SCALE = 2;

      
	localparam SPR_DRAWW = 128;

      
	localparam SPR_SPX = 2;

      
	localparam SPR_FILE = "../res/sprites/hedgehog.mem";

      
	reg signed [CORDW - 1:0] sprx;

      
	reg signed [CORDW - 1:0] spry;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			if (sprx <= -SPR_DRAWW)

      
				sprx <= H_RES;

      
			else

      
				sprx <= sprx - SPR_SPX;

      
		if (rst_pix) begin

      
			sprx <= H_RES;

      
			spry <= 240;

      
		end

      
	end

      
	wire drawing;

      
	wire [3:0] spr_pix_indx;

      
	sprite #(

      
		.CORDW(CORDW),

      
		.H_RES(H_RES),

      
		.SX_OFFS(SX_OFFS),

      
		.SPR_FILE(SPR_FILE),

      
		.SPR_WIDTH(SPR_WIDTH),

      
		.SPR_HEIGHT(SPR_HEIGHT),

      
		.SPR_SCALE(SPR_SCALE),

      
		.SPR_DATAW(CIDXW)

      
	) sprite_hedgehog(

      
		.clk(clk_pix),

      
		.rst(rst_pix),

      
		.line(line),

      
		.sx(sx),

      
		.sy(sy),

      
		.sprx(sprx),

      
		.spry(spry),

      
		.pix(spr_pix_indx),

      
		.drawing(drawing)

      
	);

      
	wire [11:0] spr_pix_colr;

      
	clut_simple #(

      
		.COLRW(COLRW),

      
		.CIDXW(CIDXW),

      
		.F_PAL(PAL_FILE)

      
	) clut_instance(

      
		.clk_write(clk_pix),

      
		.clk_read(clk_pix),

      
		.we(0),

      
		.cidx_write(0),

      
		.cidx_read(spr_pix_indx),

      
		.colr_in(0),

      
		.colr_out(spr_pix_colr)

      
	);

      
	reg drawing_t1;

      
	always @(posedge clk_pix) drawing_t1 <= drawing && (spr_pix_indx != TRANS_INDX);

      
	reg [11:0] bg_colr;

      
	always @(posedge clk_pix)

      
		if (line)

      
			if (sy == 0)

      
				bg_colr <= 12'h239;

      
			else if (sy == 80)

      
				bg_colr <= 12'h24a;

      
			else if (sy == 140)

      
				bg_colr <= 12'h25b;

      
			else if (sy == 190)

      
				bg_colr <= 12'h26c;

      
			else if (sy == 230)

      
				bg_colr <= 12'h27d;

      
			else if (sy == 265)

      
				bg_colr <= 12'h29e;

      
			else if (sy == 295)

      
				bg_colr <= 12'h2bf;

      
			else if (sy == 320)

      
				bg_colr <= 12'h260;

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) {paint_r, paint_g, paint_b} = (drawing_t1 ? spr_pix_colr : bg_colr);

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_frame <= frame;

      
		sdl_r <= {2 {paint_r}};

      
		sdl_g <= {2 {paint_g}};

      
		sdl_b <= {2 {paint_b}};

      
	end

      
endmodule
