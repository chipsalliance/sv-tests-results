		.clk_100m(clk_100m),

		.clk_pix(clk_pix),

      
		.clk_pix_5x(),

      
		.clk_pix_locked(clk_pix_locked)

      
	);

	localparam CORDW = 16;

	wire signed [15:0] sy;

	localparam FB_WIDTH = 160;

      
	localparam FB_HEIGHT = 120;

      
	localparam FB_PIXELS = 19200;

		vga_vsync <= vsync;
