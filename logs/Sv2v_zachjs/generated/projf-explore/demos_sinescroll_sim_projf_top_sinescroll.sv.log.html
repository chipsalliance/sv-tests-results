	localparam CIDXW = 4;

      
	localparam PAL_FILE = {LIB_RES, "/palettes/sweetie16_4b.mem"};

	localparam FB_HEIGHT = 180;

      
	localparam FB_SCALE = 2;

      
	localparam FB_OFFX = 0;

      
	localparam FB_OFFY = 60;

      
	localparam FB_PIXELS = 57600;

      
	localparam FB_ADDRW = 16;

      
	localparam FB_DATAW = CIDXW;

      
	reg [15:0] fb_addr_write;

      
	reg [15:0] fb_addr_clear;

      
	wire [15:0] fb_addr_render;

      
	reg [15:0] fb_addr_read;

      
	reg [3:0] fb_colr_write;

      
	reg [3:0] fb_colr_clear;

      
	reg [3:0] fb_colr_render;

      
	reg [3:0] fb_colr_read;

      
	wire [3:0] fb_colr_read_0;

      
	wire [3:0] fb_colr_read_1;

      
	reg fb_we;

      
	reg fb_front;

      
	bram_sdp #(

      
		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F("")

      
	) bram_inst_0(

      
		.clk_write(clk_sys),

      
		.clk_read(clk_sys),

      
		.we(fb_we && fb_front),

      
		.addr_write(fb_addr_write),

      
		.addr_read(fb_addr_read),

      
		.data_in(fb_colr_write),

      
		.data_out(fb_colr_read_0)

		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F("")

		.clk_write(clk_sys),

      
		.clk_read(clk_sys),

      
		.we(fb_we && !fb_front),

      
		.addr_write(fb_addr_write),

      
		.addr_read(fb_addr_read),

      
		.data_in(fb_colr_write),

      
		.data_out(fb_colr_read_1)

      
	);

	xd xd_frame(

      
		.clk_src(clk_pix),

      
		.clk_dst(clk_sys),

      
		.flag_src(frame),

      
		.flag_dst(frame_sys)

		.clk_src(clk_pix),

      
		.clk_dst(clk_sys),

      
		.flag_src(line),

      
		.flag_dst(line_sys)

		.clk_src(clk_pix),

      
		.clk_dst(clk_sys),

      
		.flag_src(line && (sy == FB_OFFY)),

      
		.flag_dst(line0_sys)

      
	);

      
	reg render_start;

      
	wire render_done;

      
	reg [31:0] state;

      
	always @(posedge clk_sys) begin

      
		case (state)

      
			32'd1: begin

      
				state <= 32'd2;

      
				fb_front <= ~fb_front;

      
				fb_addr_clear <= 0;

      
				fb_colr_clear <= 'h0;

      
			end

      
			32'd2: begin

      
				fb_addr_clear <= fb_addr_clear + 1;

      
				if (fb_addr_clear == 57599) begin

      
					state <= 32'd3;

      
					render_start <= 1;

      
				end

      
			end

      
			32'd3: begin

      
				state <= (render_done ? 32'd4 : 32'd3);

      
				render_start <= 0;

      
			end

      
			32'd4: state <= 32'd0;

      
			default:

      
				if (frame_sys)

      
					state <= 32'd1;

      
		endcase

      
		if (rst_sys)

      
			state <= 32'd0;

      
	end

      
	always @(posedge clk_sys) begin

      
		fb_addr_write <= (state == 32'd2 ? fb_addr_clear : fb_addr_render);

      
		fb_colr_write <= (state == 32'd2 ? fb_colr_clear : fb_colr_render);

      
	end

	wire drawing;

      
	wire clip;

      
	wire signed [CORDW - 1:0] drx;

      
	wire signed [CORDW - 1:0] dry;

		.CORDW(CORDW),

      
		.GREET_FILE(GREET_FILE),

      
		.FONT_FILE(FONT_FILE),

      
		.SIN_FILE(SIN_FILE),

      
		.SIN_SHIFT(2)

		.clk(clk_sys),

		.oe(1'b1),

		.CORDW(CORDW),

      
		.ADDRW(FB_ADDRW)

		.bmpw(FB_WIDTH),

      
		.bmph(FB_HEIGHT),

      
		.x(drx),

      
		.y(dry),

      
		.offx(0),

      
		.offy(0),

      
		.addr(fb_addr_render),

      
		.clip(clip)

      
	);

      
	localparam LAT_ADDR = 3;

      
	reg [2:0] fb_we_sr;

      
	always @(posedge clk_sys) begin

      
		fb_we_sr <= {drawing, fb_we_sr[2:1]};

      
		if (rst_sys)

      
			fb_we_sr <= 0;

      
		fb_we <= (state == 32'd2) || (fb_we_sr[0] && !clip);

      
	end

      
	reg [2:0] char_pix_sr;

      
	always @(posedge clk_sys) begin

      
		char_pix_sr <= {char_pix, char_pix_sr[2:1]};

      
		if (rst_sys)

      
			char_pix_sr <= 0;

      
		fb_colr_render <= (char_pix_sr[0] ? 'h3 : 'h0);

      
	end

      
	always @(posedge clk_sys) fb_colr_read <= (fb_front ? fb_colr_read_1 : fb_colr_read_0);

      
	reg [1:0] cnt_lb_line;

      
	always @(posedge clk_sys)

      
		if (line0_sys)

      
			cnt_lb_line <= 0;

      
		else if (line_sys)

      
			cnt_lb_line <= (cnt_lb_line == 1 ? 0 : cnt_lb_line + 1);

      
	reg lb_line;

      
	always @(posedge clk_sys) begin

      
		if (line0_sys)

      
			lb_line <= 1;

      
		if (frame_sys)

      
			lb_line <= 0;

      
	end

      
	reg lb_en_in;

      
	reg [8:0] cnt_lbx;

      
	always @(*) lb_en_in = (lb_line && (cnt_lb_line == 0)) && (cnt_lbx < FB_WIDTH);

      
	always @(posedge clk_sys) begin

      
		if (line_sys)

      
			cnt_lbx <= 0;

      
		else if (lb_en_in) begin

      
			fb_addr_read <= fb_addr_read + 1;

      
			cnt_lbx <= cnt_lbx + 1;

      
		end

      
		if (frame_sys)

      
			fb_addr_read <= 0;

      
	end

      
	reg lb_en_out;

      
	localparam LAT_LB = 4;

      
	always @(posedge clk_pix) lb_en_out <= (((sy >= FB_OFFY) && (sy < 420)) && (sx >= (FB_OFFX - LAT_LB))) && (sx < 636);

      
	wire [3:0] lb_colr_out;

      
	linebuffer_simple #(

      
		.DATAW(FB_DATAW),

      
		.LEN(FB_WIDTH)

      
	) linebuffer_instance(

      
		.clk_sys(clk_sys),

      
		.clk_pix(clk_pix),

      
		.line(line),

      
		.line_sys(line_sys),

      
		.en_in(lb_en_in),

		.colr_in(0),

      
		.colr_out(fb_pix_colr)

      
	);

	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

		paint_area = (((sy >= FB_OFFY) && (sy < 420)) && (sx >= FB_OFFX)) && (sx < 640);

      
		{paint_r, paint_g, paint_b} = (de && paint_area ? fb_pix_colr : 12'h000);

      
	end

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_frame <= frame;

      
		sdl_r <= {2 {paint_r}};

      
		sdl_g <= {2 {paint_g}};

      
		sdl_b <= {2 {paint_b}};

      
	end
