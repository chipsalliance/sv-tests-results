		.sx(),

		.hsync(hsync),

		.V(750),

		.V(750),

      
		.INC(-4),

		.en(1'b1),

      
		.rst(!clk_pix_locked),

		.sf_star(sf3_star)

      
	);

	reg [3:0] green;

      
	reg [3:0] blue;
