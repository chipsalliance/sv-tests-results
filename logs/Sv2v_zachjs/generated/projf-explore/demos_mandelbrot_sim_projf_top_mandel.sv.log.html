module top_mandel (

      
	clk_pix,

      
	rst_pix,

      
	btn_fire,

      
	btn_up,

      
	btn_dn,

      
	sdl_sx,

      
	sdl_sy,

      
	sdl_de,

      
	sdl_frame,

      
	sdl_r,

      
	sdl_g,

      
	sdl_b

      
);

      
	parameter CORDW = 16;

      
	input wire clk_pix;

      
	input wire rst_pix;

      
	input wire btn_fire;

      
	input wire btn_up;

      
	input wire btn_dn;

      
	output reg signed [CORDW - 1:0] sdl_sx;

      
	output reg signed [CORDW - 1:0] sdl_sy;

      
	output reg sdl_de;

      
	output reg sdl_frame;

      
	output reg [7:0] sdl_r;

      
	output reg [7:0] sdl_g;

      
	output reg [7:0] sdl_b;

      
	localparam FP_WIDTH = 25;

      
	localparam FP_INT = 4;

      
	localparam ITER_MAX = 255;

      
	localparam SUPERSAMPLE = 1;

      
	localparam X_START = 25'b1100100000000000000000000;

      
	localparam Y_START = 25'b1110100000000000000000000;

      
	localparam STEP = 25'b0000000001000000000000000;

      
	reg clk_sys;

      
	reg rst_sys;

      
	always @(*) begin

      
		clk_sys = clk_pix;

      
		rst_sys = rst_pix;

      
	end

      
	wire signed [CORDW - 1:0] sx;

      
	wire signed [CORDW - 1:0] sy;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(rst_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(),

      
		.vsync(),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	wire sig_fire;

      
	wire sig_up;

      
	wire sig_dn;

      
	debounce deb_fire(

      
		.clk(clk_sys),

      
		.in(btn_fire),

      
		.out(),

      
		.ondn(),

      
		.onup(sig_fire)

      
	);

      
	debounce deb_up(

      
		.clk(clk_sys),

      
		.in(btn_up),

      
		.out(sig_up),

      
		.ondn(),

      
		.onup()

      
	);

      
	debounce deb_dn(

      
		.clk(clk_sys),

      
		.in(btn_dn),

      
		.out(sig_dn),

      
		.ondn(),

      
		.onup()

      
	);

      
	localparam CHANW = 8;

      
	localparam COLRW = 24;

      
	localparam CIDXW = 8;

      
	localparam FB_WIDTH = 320;

      
	localparam FB_HEIGHT = 180;

      
	localparam FB_SCALE = 2;

      
	localparam FB_OFFX = 0;

      
	localparam FB_OFFY = 60;

      
	localparam FB_PIXELS = 57600;

      
	localparam FB_ADDRW = 16;

      
	localparam FB_DATAW = CIDXW;

      
	wire [15:0] fb_addr_write;

      
	reg [15:0] fb_addr_read;

      
	wire [7:0] fb_colr_write;

      
	wire [7:0] fb_colr_read;

      
	reg fb_we;

      
	bram_sdp #(

      
		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F("")

      
	) bram_inst(

      
		.clk_write(clk_sys),

      
		.clk_read(clk_sys),

      
		.we(fb_we),

      
		.addr_write(fb_addr_write),

      
		.addr_read(fb_addr_read),

      
		.data_in(fb_colr_write),

      
		.data_out(fb_colr_read)

      
	);

      
	wire frame_sys;

      
	wire line_sys;

      
	wire line0_sys;

      
	xd xd_frame(

      
		.clk_src(clk_pix),

      
		.clk_dst(clk_sys),

      
		.flag_src(frame),

      
		.flag_dst(frame_sys)

      
	);

      
	xd xd_line(

      
		.clk_src(clk_pix),

      
		.clk_dst(clk_sys),

      
		.flag_src(line),

      
		.flag_dst(line_sys)

      
	);

      
	xd xd_line0(

      
		.clk_src(clk_pix),

      
		.clk_dst(clk_sys),

      
		.flag_src(line && (sy == FB_OFFY)),

      
		.flag_dst(line0_sys)

      
	);

      
	reg signed [24:0] x_start;

      
	reg signed [24:0] x_start_p;

      
	reg signed [24:0] y_start;

      
	reg signed [24:0] y_start_p;

      
	reg signed [24:0] step;

      
	reg signed [24:0] step_p;

      
	reg start_func;

      
	reg render_required;

      
	wire drawing;

      
	wire render_busy;

      
	reg changed_params;

      
	reg [31:0] state;

      
	always @(posedge clk_sys) begin

      
		case (state)

      
			32'd0:

      
				if (sig_fire)

      
					state <= 32'd1;

      
			32'd1:

      
				if (sig_fire)

      
					state <= 32'd2;

      
			32'd2:

      
				if (sig_fire)

      
					state <= 32'd0;

      
		endcase

      
		x_start_p <= x_start;

      
		y_start_p <= y_start;

      
		step_p <= step;

      
		if ((frame_sys && !changed_params) && !render_busy)

      
			case (state)

      
				32'd0:

      
					if (sig_up) begin

      
						x_start_p <= x_start - (step <<< 4);

      
						changed_params <= 1;

      
						$display(">> Move left");

      
					end

      
					else if (sig_dn) begin

      
						x_start_p <= x_start + (step <<< 4);

      
						changed_params <= 1;

      
						$display(">> Move right");

      
					end

      
				32'd1:

      
					if (sig_up) begin

      
						y_start_p <= y_start - (step <<< 4);

      
						changed_params <= 1;

      
						$display(">> Move up");

      
					end

      
					else if (sig_dn) begin

      
						y_start_p <= y_start + (step <<< 4);

      
						changed_params <= 1;

      
						$display(">> Move down");

      
					end

      
				32'd2:

      
					if (sig_up) begin

      
						x_start_p <= x_start - (step <<< 7);

      
						y_start_p <= (y_start - (step <<< 6)) - (step <<< 5);

      
						step_p <= 2 * step;

      
						changed_params <= 1;

      
						$display(">> Zoom out");

      
					end

      
					else if (sig_dn) begin

      
						x_start_p <= x_start + (step <<< 6);

      
						y_start_p <= (y_start + (step <<< 5)) + (step <<< 4);

      
						step_p <= step / 2;

      
						changed_params <= 1;

      
						$display(">> Zoom in");

      
					end

      
			endcase

      
		start_func <= 0;

      
		if (changed_params && !render_busy) begin

      
			changed_params <= 0;

      
			if ((step_p != 0) && (step_p <= STEP)) begin

      
				render_required <= 1;

      
				x_start <= x_start_p;

      
				y_start <= y_start_p;

      
				step <= step_p;

      
			end

      
		end

      
		else if (render_required) begin

      
			start_func <= 1;

      
			render_required <= 0;

      
		end

      
		if (rst_sys) begin

      
			state <= 32'd0;

      
			x_start <= X_START;

      
			y_start <= Y_START;

      
			step <= STEP;

      
			render_required <= 1;

      
			changed_params <= 0;

      
		end

      
	end

      
	wire clip;

      
	wire signed [CORDW - 1:0] drx;

      
	wire signed [CORDW - 1:0] dry;

      
	render_mandel #(

      
		.CORDW(CORDW),

      
		.FB_WIDTH(FB_WIDTH),

      
		.FB_HEIGHT(FB_HEIGHT),

      
		.CIDXW(CIDXW),

      
		.FP_WIDTH(FP_WIDTH),

      
		.FP_INT(FP_INT),

      
		.ITER_MAX(ITER_MAX),

      
		.SUPERSAMPLE(SUPERSAMPLE)

      
	) render_mandel_instance(

      
		.clk(clk_sys),

      
		.rst(rst_sys),

      
		.start(start_func),

      
		.x_start(x_start),

      
		.y_start(y_start),

      
		.step(step),

      
		.x(drx),

      
		.y(dry),

      
		.cidx(fb_colr_write),

      
		.drawing(drawing),

      
		.busy(render_busy),

      
		.done()

      
	);

      
	bitmap_addr #(

      
		.CORDW(CORDW),

      
		.ADDRW(FB_ADDRW)

      
	) bitmap_addr_instance(

      
		.clk(clk_sys),

      
		.bmpw(FB_WIDTH),

      
		.bmph(FB_HEIGHT),

      
		.x(drx),

      
		.y(dry),

      
		.offx(0),

      
		.offy(0),

      
		.addr(fb_addr_write),

      
		.clip(clip)

      
	);

      
	localparam LAT_ADDR = 3;

      
	reg [2:0] fb_we_sr;

      
	always @(posedge clk_sys) begin

      
		fb_we_sr <= {drawing, fb_we_sr[2:1]};

      
		if (rst_sys)

      
			fb_we_sr <= 0;

      
	end

      
	always @(*) fb_we = fb_we_sr[0] && !clip;

      
	reg [1:0] cnt_lb_line;

      
	always @(posedge clk_sys)

      
		if (line0_sys)

      
			cnt_lb_line <= 0;

      
		else if (line_sys)

      
			cnt_lb_line <= (cnt_lb_line == 1 ? 0 : cnt_lb_line + 1);

      
	reg lb_line;

      
	always @(posedge clk_sys) begin

      
		if (line0_sys)

      
			lb_line <= 1;

      
		if (frame_sys)

      
			lb_line <= 0;

      
	end

      
	reg lb_en_in;

      
	reg [8:0] cnt_lbx;

      
	always @(*) lb_en_in = (lb_line && (cnt_lb_line == 0)) && (cnt_lbx < FB_WIDTH);

      
	always @(posedge clk_sys) begin

      
		if (line_sys)

      
			cnt_lbx <= 0;

      
		else if (lb_en_in) begin

      
			fb_addr_read <= fb_addr_read + 1;

      
			cnt_lbx <= cnt_lbx + 1;

      
		end

      
		if (frame_sys)

      
			fb_addr_read <= 0;

      
	end

      
	reg lb_en_out;

      
	localparam LAT_LB = 2;

      
	always @(posedge clk_pix) lb_en_out <= (((sy >= FB_OFFY) && (sy < 420)) && (sx >= (FB_OFFX - LAT_LB))) && (sx < 638);

      
	wire [7:0] lb_colr_out;

      
	linebuffer_simple #(

      
		.DATAW(FB_DATAW),

      
		.LEN(FB_WIDTH)

      
	) linebuffer_instance(

      
		.clk_sys(clk_sys),

      
		.clk_pix(clk_pix),

      
		.line(line),

      
		.line_sys(line_sys),

      
		.en_in(lb_en_in),

      
		.en_out(lb_en_out),

      
		.scale(FB_SCALE),

      
		.data_in(fb_colr_read),

      
		.data_out(lb_colr_out)

      
	);

      
	reg [7:0] lb_colr_out_2;

      
	always @(*) lb_colr_out_2 = lb_colr_out / 2;

      
	reg paint_area;

      
	reg [23:0] paint_colr;

      
	reg [7:0] paint_r;

      
	reg [7:0] paint_g;

      
	reg [7:0] paint_b;

      
	always @(*) begin

      
		paint_colr = {lb_colr_out_2, lb_colr_out, lb_colr_out};

      
		paint_area = (((sy >= FB_OFFY) && (sy < 420)) && (sx >= FB_OFFX)) && (sx < 640);

      
		{paint_r, paint_g, paint_b} = (de && paint_area ? paint_colr : 24'h001020);

      
	end

      
	always @(posedge clk_pix) begin

      
		sdl_sx <= sx;

      
		sdl_sy <= sy;

      
		sdl_de <= de;

      
		sdl_frame <= frame;

      
		sdl_r <= paint_r;

      
		sdl_g <= paint_g;

      
		sdl_b <= paint_b;

      
	end

      
endmodule
