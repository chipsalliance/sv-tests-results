module top_sprite_v2a (

      
	clk_100m,

      
	btn_rst,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 16;

      
	wire signed [15:0] sx;

      
	wire signed [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam SPR_WIDTH = 8;

      
	localparam SPR_HEIGHT = 8;

      
	localparam SPR_SCALE_X = 6;

      
	localparam SPR_SCALE_Y = 6;

      
	localparam SPR_FILE = "saucer.mem";

      
	reg spr_start;

      
	wire spr_pix;

      
	localparam H_RES = 640;

      
	localparam SPR_SPEED_X = 4;

      
	reg signed [15:0] sprx;

      
	reg signed [15:0] spry;

      
	reg dx;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			if (sprx >= 588) begin

      
				dx <= 1;

      
				sprx <= sprx - SPR_SPEED_X;

      
			end

      
			else if (sprx < SPR_SPEED_X) begin

      
				dx <= 0;

      
				sprx <= sprx + SPR_SPEED_X;

      
			end

      
			else

      
				sprx <= (dx ? sprx - SPR_SPEED_X : sprx + SPR_SPEED_X);

      
		if (!clk_locked) begin

      
			sprx <= 296;

      
			spry <= 216;

      
			dx <= 0;

      
		end

      
	end

      
	always @(*) spr_start = line && (sy == spry);

      
	sprite_v2 #(

      
		.WIDTH(SPR_WIDTH),

      
		.HEIGHT(SPR_HEIGHT),

      
		.SCALE_X(SPR_SCALE_X),

      
		.SCALE_Y(SPR_SCALE_Y),

      
		.SPR_FILE(SPR_FILE)

      
	) spr_instance(

      
		.clk(clk_pix),

      
		.rst(!clk_locked),

      
		.start(spr_start),

      
		.sx(sx),

      
		.sprx(sprx),

      
		.pix(spr_pix)

      
	);

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync;

      
		vga_vsync <= vsync;

      
		vga_r <= (de && spr_pix ? 4'hf : 4'h0);

      
		vga_g <= (de && spr_pix ? 4'hc : 4'h0);

      
		vga_b <= (de && spr_pix ? 4'h0 : 4'h0);

      
	end

      
endmodule
