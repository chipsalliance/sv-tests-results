`default_nettype none

      
module framebuffer_spram (

      
	clk_sys,

      
	clk_pix,

      
	rst_sys,

      
	rst_pix,

      
	de,

      
	frame,

      
	line,

      
	we,

      
	x,

      
	y,

      
	cidx,

      
	busy,

      
	clip,

      
	red,

      
	green,

      
	blue

      
);

      
	parameter CORDW = 16;

      
	parameter WIDTH = 320;

      
	parameter HEIGHT = 180;

      
	parameter CIDXW = 4;

      
	parameter CHANW = 4;

      
	parameter SCALE = 4;

      
	parameter F_IMAGE = "";

      
	parameter F_PALETTE = "";

      
	input wire clk_sys;

      
	input wire clk_pix;

      
	input wire rst_sys;

      
	input wire rst_pix;

      
	input wire de;

      
	input wire frame;

      
	input wire line;

      
	input wire we;

      
	input wire signed [CORDW - 1:0] x;

      
	input wire signed [CORDW - 1:0] y;

      
	input wire [CIDXW - 1:0] cidx;

      
	output reg busy;

      
	output reg clip;

      
	output reg [CHANW - 1:0] red;

      
	output reg [CHANW - 1:0] green;

      
	output reg [CHANW - 1:0] blue;

      
	wire frame_sys;

      
	xd xd_frame(

      
		.clk_i(clk_pix),

      
		.clk_o(clk_sys),

      
		.rst_i(rst_pix),

      
		.rst_o(rst_sys),

      
		.i(frame),

      
		.o(frame_sys)

      
	);

      
	localparam FB_PIXELS = WIDTH * HEIGHT;

      
	localparam FB_ADDRW = $clog2(FB_PIXELS);

      
	localparam FB_DATAW = CIDXW;

      
	localparam FB_DUALPORT = 0;

      
	reg [FB_ADDRW - 1:0] fb_addr_read;

      
	reg [FB_ADDRW - 1:0] fb_addr_write;

      
	wire [FB_DATAW - 1:0] fb_cidx_read;

      
	reg [FB_DATAW - 1:0] fb_cidx_read_p1;

      
	reg signed [CORDW - 1:0] x_add;

      
	reg [FB_ADDRW - 1:0] fb_addr_line;

      
	always @(posedge clk_sys) begin

      
		fb_addr_line <= WIDTH * y;

      
		x_add <= x;

      
		fb_addr_write <= fb_addr_line + x_add;

      
	end

      
	reg fb_we;

      
	reg we_in_p1;

      
	reg [FB_DATAW - 1:0] fb_cidx_write;

      
	reg [FB_DATAW - 1:0] cidx_in_p1;

      
	always @(posedge clk_sys) begin

      
		we_in_p1 <= we;

      
		cidx_in_p1 <= cidx;

      
		clip <= (((y < 0) || (y >= HEIGHT)) || (x < 0)) || (x >= WIDTH);

      
		fb_we <= (busy || clip ? 0 : we_in_p1);

      
		fb_cidx_write <= cidx_in_p1;

      
	end

      
	spram_nibble spram_inst(

      
		.clk(clk_sys),

      
		.we(fb_we),

      
		.addr((fb_we ? fb_addr_write : fb_addr_read)),

      
		.data_in(fb_cidx_write),

      
		.data_out(fb_cidx_read)

      
	);

      
	localparam LB_SCALE = SCALE;

      
	localparam LB_LEN = WIDTH;

      
	localparam LB_BPC = CHANW;

      
	wire lb_data_req;

      
	reg [$clog2(LB_LEN + 1) - 1:0] cnt_h;

      
	reg lb_en_in;

      
	reg lb_en_out;

      
	always @(*) lb_en_in = cnt_h < LB_LEN;

      
	always @(*) lb_en_out = de;

      
	localparam LAT = 5;

      
	reg [4:0] lb_en_in_sr;

      
	always @(posedge clk_sys) begin

      
		lb_en_in_sr <= {lb_en_in, lb_en_in_sr[4:1]};

      
		if (rst_sys)

      
			lb_en_in_sr <= 0;

      
	end

      
	always @(posedge clk_sys) begin

      
		if (fb_addr_read < (FB_PIXELS - 1)) begin

      
			if (lb_data_req) begin

      
				cnt_h <= 0;

      
				if (!FB_DUALPORT)

      
					busy <= 1;

      
			end

      
			else if (cnt_h < LB_LEN) begin

      
				cnt_h <= cnt_h + 1;

      
				fb_addr_read <= fb_addr_read + 1;

      
			end

      
		end

      
		else

      
			cnt_h <= LB_LEN;

      
		if (frame_sys) begin

      
			fb_addr_read <= 0;

      
			busy <= 0;

      
		end

      
		if (rst_sys) begin

      
			fb_addr_read <= 0;

      
			busy <= 0;

      
			cnt_h <= LB_LEN;

      
		end

      
		if (lb_en_in_sr == 5'b10000)

      
			busy <= 0;

      
	end

      
	reg [LB_BPC - 1:0] lb_in_0;

      
	reg [LB_BPC - 1:0] lb_in_1;

      
	reg [LB_BPC - 1:0] lb_in_2;

      
	wire [LB_BPC - 1:0] lb_out_0;

      
	wire [LB_BPC - 1:0] lb_out_1;

      
	wire [LB_BPC - 1:0] lb_out_2;

      
	linebuffer #(

      
		.WIDTH(LB_BPC),

      
		.LEN(LB_LEN),

      
		.SCALE(LB_SCALE)

      
	) lb_inst(

      
		.clk_in(clk_sys),

      
		.clk_out(clk_pix),

      
		.rst_in(rst_sys),

      
		.rst_out(rst_pix),

      
		.data_req(lb_data_req),

      
		.en_in(lb_en_in_sr[0]),

      
		.en_out(lb_en_out),

      
		.frame(frame),

      
		.line(line),

      
		.din_0(lb_in_0),

      
		.din_1(lb_in_1),

      
		.din_2(lb_in_2),

      
		.dout_0(lb_out_0),

      
		.dout_1(lb_out_1),

      
		.dout_2(lb_out_2)

      
	);

      
	always @(posedge clk_sys) fb_cidx_read_p1 <= fb_cidx_read;

      
	localparam CLUTW = 3 * CHANW;

      
	wire [CLUTW - 1:0] clut_colr;

      
	rom_async #(

      
		.WIDTH(CLUTW),

      
		.DEPTH(2 ** CIDXW),

      
		.INIT_F(F_PALETTE)

      
	) clut(

      
		.addr(fb_cidx_read_p1),

      
		.data(clut_colr)

      
	);

      
	always @(posedge clk_sys) {lb_in_2, lb_in_1, lb_in_0} <= clut_colr;

      
	reg lb_en_out_p1;

      
	always @(posedge clk_pix) lb_en_out_p1 <= lb_en_out;

      
	always @(*) begin

      
		red = (lb_en_out_p1 ? lb_out_2 : 0);

      
		green = (lb_en_out_p1 ? lb_out_1 : 0);

      
		blue = (lb_en_out_p1 ? lb_out_0 : 0);

      
	end

      
endmodule
