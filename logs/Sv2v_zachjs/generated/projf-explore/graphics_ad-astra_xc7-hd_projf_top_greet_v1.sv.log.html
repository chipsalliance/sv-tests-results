		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(frame),

	localparam GREET_MSGS = 32;

      
	localparam GREET_LENGTH = 16;

      
	localparam G_ROM_WIDTH = $clog2('h5f);

      
	localparam G_ROM_DEPTH = 512;

      
	localparam GREET_FILE = "greet.mem";

      
	reg [8:0] greet_rom_addr;

      
	wire [G_ROM_WIDTH - 1:0] greet_rom_data;

      
	rom_sync #(

      
		.WIDTH(G_ROM_WIDTH),

      
		.DEPTH(G_ROM_DEPTH),

      
		.INIT_F(GREET_FILE)

      
	) greet_rom(

      
		.clk(clk_pix),

      
		.addr(greet_rom_addr),

      
		.data(greet_rom_data)

      
	);

      
	localparam MSG_CHG = 80;

      
	reg [6:0] cnt_frm;

      
	reg [4:0] greeting;

      
	always @(posedge clk_pix)

      
		if (frame) begin

      
			cnt_frm <= cnt_frm + 1;

      
			if (cnt_frm == MSG_CHG) begin

      
				greeting <= greeting + 1;

      
				cnt_frm <= 0;

      
			end

      
		end

      
	localparam FONT_WIDTH = 8;

      
	localparam FONT_HEIGHT = 8;

      
	localparam FONT_GLYPHS = 64;

      
	localparam F_ROM_DEPTH = 512;

      
	localparam CP_START = 'h20;

      
	localparam FONT_FILE = "font_unscii_8x8_latin_uc.mem";

      
	reg [8:0] font_rom_addr;

      
	wire [7:0] font_rom_data;

		.WIDTH(FONT_WIDTH),

      
		.DEPTH(F_ROM_DEPTH),

      
		.INIT_F(FONT_FILE)

		.addr(font_rom_addr),

      
		.data(font_rom_data)

	localparam SPR_CNT = 8;

      
	localparam LINE2 = 360;

      
	localparam SPR_SCALE_X = 16;

      
	localparam SPR_SCALE_Y = 16;

      
	localparam SPR_DMA = -16;

	reg signed [15:0] spr_y [0:1];

		spr_x[0] = 128;

      
		spr_x[1] = 256;

      
		spr_x[2] = 384;

      
		spr_x[3] = 512;

      
		spr_x[4] = 640;

      
		spr_x[5] = 768;

      
		spr_x[6] = 896;

      
		spr_x[7] = 1024;

      
		spr_y[0] = 225;

      
		spr_y[1] = 375;

      
	end

      
	reg spr_start;

      
	always @(*) spr_start = (sy < LINE2 ? line && (sy == spr_y[0]) : line && (sy == spr_y[1]));

      
	integer i;

      
	reg [8:0] msg_start;

      
	always @(*) begin

      
		greet_rom_addr = 0;

      
		msg_start = greeting * GREET_LENGTH;

      
		for (i = 0; i < SPR_CNT; i = i + 1)

      
			if (sx == (SPR_DMA + i))

      
				greet_rom_addr = (sy < LINE2 ? msg_start + i : (msg_start + i) + 8);

      
	end

      
	reg [G_ROM_WIDTH - 1:0] spr_cp [0:7];

      
	always @(posedge clk_pix)

      
		for (i = 0; i < SPR_CNT; i = i + 1)

      
			if (sx == ((SPR_DMA + i) + 1))

      
				spr_cp[i] <= greet_rom_data;

      
	reg [8:0] spr_glyph_addr [0:7];

      
	wire [2:0] spr_glyph_line [0:7];

      
	reg [7:0] spr_fdma;

      
	always @(*) begin

      
		font_rom_addr = 0;

      
		for (i = 0; i < SPR_CNT; i = i + 1)

      
			begin

      
				spr_fdma[i] = sx == ((SPR_DMA + i) + 2);

      
				spr_glyph_addr[i] = (spr_cp[i] - CP_START) * FONT_HEIGHT;

      
				if (spr_fdma[i])

      
					font_rom_addr = spr_glyph_addr[i] + spr_glyph_line[i];

      
			end

      
	end

      
	wire [7:0] spr_pix;

      
	genvar m;

      
	generate

      
		for (m = 0; m < SPR_CNT; m = m + 1) begin : sprite_gen

      
			sprite #(

      
				.WIDTH(FONT_WIDTH),

      
				.HEIGHT(FONT_HEIGHT),

      
				.SCALE_X(SPR_SCALE_X),

      
				.SCALE_Y(SPR_SCALE_Y),

      
				.LSB(0),

      
				.CORDW(CORDW),

      
				.ADDRW(3)

      
			) spr0(

      
				.clk(clk_pix),

      
				.rst(!clk_pix_locked),

      
				.start(spr_start),

      
				.dma_avail(spr_fdma[m]),

      
				.sx(sx),

      
				.sprx(spr_x[m]),

      
				.data_in(font_rom_data),

      
				.pos(spr_glyph_line[m]),

      
				.pix(spr_pix[m]),

      
				.drawing(),

      
				.done()

      
			);

      
		end

      
	endgenerate

      
	wire sf1_on;

      
	wire sf2_on;

      
	wire sf3_on;

		.V(750),

      
		.INC(-2),

		.sf_star(sf2_star)

		.H(1650),

      
		.V(750),

      
		.INC(-4),

      
		.MASK(21'h0007ff)

		.en(1'b1),

      
		.rst(!clk_pix_locked),

		.sf_star(sf3_star)

	always @(*) begin

      
		{red_spr, green_spr, blue_spr} = (spr_pix != 0 ? 12'hfc0 : 12'h000);

      
		starlight = (sf1_on ? sf1_star[7:4] : (sf2_on ? sf2_star[7:4] : (sf3_on ? sf3_star[7:4] : 4'h0)));

	reg [3:0] green;

      
	reg [3:0] blue;

		red = (de ? (spr_pix != 0 ? red_spr : starlight) : 4'h0);

      
		green = (de ? (spr_pix != 0 ? green_spr : starlight) : 4'h0);

      
		blue = (de ? (spr_pix != 0 ? blue_spr : starlight) : 4'h0);
