module top_square (

      
	clk_100m,

      
	btn_rst_n,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst_n;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_pix_locked;

      
	clock_480p clock_pix_inst(

      
		.clk_100m(clk_100m),

      
		.rst(!btn_rst_n),

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(),

      
		.clk_pix_locked(clk_pix_locked)

      
	);

      
	localparam CORDW = 10;

      
	wire [9:0] sx;

      
	wire [9:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	simple_480p display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_pix_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de)

      
	);

      
	reg square;

      
	always @(*) square = ((sx > 220) && (sx < 420)) && ((sy > 140) && (sy < 340));

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) begin

      
		paint_r = (square ? 4'hf : 4'h1);

      
		paint_g = (square ? 4'hf : 4'h3);

      
		paint_b = (square ? 4'hf : 4'h7);

      
	end

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync;

      
		vga_vsync <= vsync;

      
		if (de) begin

      
			vga_r <= paint_r;

      
			vga_g <= paint_g;

      
			vga_b <= paint_b;

      
		end

      
		else begin

      
			vga_r <= 4'h0;

      
			vga_g <= 4'h0;

      
			vga_b <= 4'h0;

      
		end

      
	end

      
endmodule
