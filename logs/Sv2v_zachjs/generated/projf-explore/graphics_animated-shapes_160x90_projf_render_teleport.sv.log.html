module render_teleport (

      
	clk,

      
	rst,

      
	oe,

      
	start,

      
	x,

      
	y,

      
	cidx,

      
	drawing,

      
	done

	parameter CORDW = 16;

      
	parameter CIDXW = 2;

      
	parameter SCALE = 1;

      
	input wire clk;

      
	input wire rst;

      
	input wire oe;

      
	input wire start;

      
	output wire signed [CORDW - 1:0] x;

      
	output wire signed [CORDW - 1:0] y;

      
	output reg [CIDXW - 1:0] cidx;

      
	output wire drawing;

      
	output reg done;

      
	localparam ANIM_CNT = 5;

      
	localparam ANIM_SPEED = 3;

      
	reg [2:0] cnt_anim;

      
	reg [1:0] cnt_anim_speed;

      
	reg [CIDXW - 1:0] colr_offs;

      
	always @(posedge clk)

      
		if (start)

      
			if (cnt_anim_speed == 2) begin

      
				if (cnt_anim == 4) begin

      
					cnt_anim <= 0;

      
					colr_offs <= colr_offs + 1;

      
				end

      
				else

      
					cnt_anim <= cnt_anim + 1;

      
				cnt_anim_speed <= 0;

      
			end

      
			else

      
				cnt_anim_speed <= cnt_anim_speed + 1;

      
	localparam SHAPE_CNT = 8;

      
	reg [3:0] shape_id;

      
	reg signed [CORDW - 1:0] dx0;

      
	reg signed [CORDW - 1:0] dy0;

      
	reg signed [CORDW - 1:0] dx1;

      
	reg signed [CORDW - 1:0] dy1;

      
	reg draw_start;

      
	wire draw_done;

      
	reg [31:0] state;

      
	always @(posedge clk) begin

      
		case (state)

      
			32'd1: begin

      
				draw_start <= 1;

      
				state <= 32'd2;

      
				cidx <= colr_offs + shape_id[1:0];

      
				case (shape_id)

      
					'd0: begin

      
						dx0 <= 0;

      
						dy0 <= 0;

      
						dx1 <= 159;

      
						dy1 <= 89;

      
					end

      
					'd1: begin

      
						dx0 <= 40 - (cnt_anim * 12);

      
						dy0 <= -(cnt_anim * 12);

      
						dx1 <= 279 + (cnt_anim * 12);

      
						dy1 <= 249 + (cnt_anim * 12);

      
					end

      
					'd2: begin

      
						dx0 <= 80 - (cnt_anim * 8);

      
						dy0 <= 10 - (cnt_anim * 8);

      
						dx1 <= 239 + (cnt_anim * 8);

      
						dy1 <= 169 + (cnt_anim * 8);

      
					end

      
					'd3: begin

      
						dx0 <= 105 - (cnt_anim * 5);

      
						dy0 <= 35 - (cnt_anim * 5);

      
						dx1 <= 214 + (cnt_anim * 5);

      
						dy1 <= 144 + (cnt_anim * 5);

      
					end

      
					'd4: begin

      
						dx0 <= 125 - (cnt_anim * 4);

      
						dy0 <= 55 - (cnt_anim * 4);

      
						dx1 <= 194 + (cnt_anim * 4);

      
						dy1 <= 124 + (cnt_anim * 4);

      
					end

						dx0 <= 140 - (cnt_anim * 3);

      
						dy0 <= 70 - (cnt_anim * 3);

      
						dx1 <= 179 + (cnt_anim * 3);

      
						dy1 <= 109 + (cnt_anim * 3);

      
					end

						dx0 <= 150 - (cnt_anim * 2);

      
						dy0 <= 80 - (cnt_anim * 2);

      
						dx1 <= 169 + (cnt_anim * 2);

      
						dy1 <= 99 + (cnt_anim * 2);

      
					end

      
					default: begin

      
						dx0 <= 155 - (cnt_anim * 1);

      
						dy0 <= 85 - (cnt_anim * 1);

      
						dx1 <= 164 + (cnt_anim * 1);

      
						dy1 <= 94 + (cnt_anim * 1);

      
					end

      
				endcase

      
			end

      
			32'd2: begin

      
				draw_start <= 0;

      
				if (draw_done)
