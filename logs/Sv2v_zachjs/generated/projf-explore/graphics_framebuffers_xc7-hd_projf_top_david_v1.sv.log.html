	localparam FB_WIDTH = 160;

      
	localparam FB_HEIGHT = 120;

      
	localparam FB_PIXELS = 19200;

      
	localparam FB_ADDRW = 15;

      
	localparam FB_DATAW = 1;

      
	localparam FB_IMAGE = "david_1bit.mem";

      
	reg fb_we;

      
	reg [14:0] fb_addr_write;

      
	reg [14:0] fb_addr_read;

      
	reg [0:0] fb_colr_write;

      
	wire [0:0] fb_colr_read;

      
	bram_sdp #(

      
		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F(FB_IMAGE)

		.clk_write(clk_pix),

      
		.clk_read(clk_pix),

		.addr_write(fb_addr_write),

      
		.addr_read(fb_addr_read),

      
		.data_in(fb_colr_write),

      
		.data_out(fb_colr_read)

      
	);

      
	reg [7:0] cnt_draw;

      
	reg [31:0] state;

      
	always @(posedge clk_pix)

      
		case (state)

      
			32'd1:

      
				if (cnt_draw < 159) begin

      
					fb_addr_write <= fb_addr_write + 1;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					cnt_draw <= 0;

      
					state <= 32'd2;

      
				end

      
			32'd2:

      
				if (cnt_draw < 119) begin

      
					fb_addr_write <= fb_addr_write + FB_WIDTH;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					fb_addr_write <= 0;

      
					cnt_draw <= 0;

      
					state <= 32'd4;

      
				end

      
			32'd4:

      
				if (cnt_draw < 119) begin

      
					fb_addr_write <= fb_addr_write + FB_WIDTH;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					cnt_draw <= 0;

      
					state <= 32'd3;

      
				end

      
			32'd3:

      
				if (cnt_draw < 159) begin

      
					fb_addr_write <= fb_addr_write + 1;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					fb_we <= 0;

      
					state <= 32'd5;

      
				end

      
			32'd0:

      
				if (frame) begin

      
					fb_colr_write <= 1;

      
					fb_we <= 1;

      
					cnt_draw <= 0;

      
					state <= 32'd1;

      
				end

      
			default: state <= 32'd5;

      
		endcase

		dvi_blue <= (paint_p1 && fb_colr_read ? 8'hff : 8'h00);

      
	end
