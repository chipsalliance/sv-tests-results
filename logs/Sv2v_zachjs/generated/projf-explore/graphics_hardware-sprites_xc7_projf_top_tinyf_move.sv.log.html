module top_tinyf_move (

      
	clk_100m,

      
	btn_rst_n,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst_n;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_pix_locked;

      
	clock_480p clock_pix_inst(

      
		.clk_100m(clk_100m),

      
		.rst(!btn_rst_n),

      
		.clk_pix(clk_pix),

      
		.clk_pix_5x(),

      
		.clk_pix_locked(clk_pix_locked)

      
	);

      
	reg rst_pix;

      
	always @(*) rst_pix = !clk_pix_locked;

      
	localparam CORDW = 16;

      
	wire signed [15:0] sx;

      
	wire signed [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire de;

      
	wire frame;

      
	wire line;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(rst_pix),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(de),

      
		.frame(frame),

      
		.line(line)

      
	);

      
	localparam H_RES = 640;

      
	localparam V_RES = 480;

      
	localparam SPR_WIDTH = 8;

      
	localparam SPR_HEIGHT = 8;

      
	localparam SPR_SCALE = 3;

      
	localparam SPR_DATAW = 1;

      
	localparam SPR_DRAWW = 64;

      
	localparam SPR_DRAWH = 64;

      
	localparam SPR_SPX = 4;

      
	localparam SPR_FILE = "letter_f.mem";

      
	reg signed [15:0] sprx;

      
	reg signed [15:0] spry;

      
	reg dx;

      
	always @(posedge clk_pix) begin

      
		if (frame)

      
			if (dx == 0) begin

      
				if ((sprx + SPR_DRAWW) >= 768)

      
					dx <= 1;

      
				else

      
					sprx <= sprx + SPR_SPX;

      
			end

      
			else if (sprx <= (-2 * SPR_DRAWW))

      
				dx <= 0;

      
			else

      
				sprx <= sprx - SPR_SPX;

      
		if (rst_pix) begin

      
			sprx <= 288;

      
			spry <= 208;

      
			dx <= 0;

      
		end

      
	end

      
	wire drawing;

      
	wire [0:0] pix;

      
	sprite #(

      
		.CORDW(CORDW),

      
		.H_RES(H_RES),

      
		.SPR_FILE(SPR_FILE),

      
		.SPR_WIDTH(SPR_WIDTH),

      
		.SPR_HEIGHT(SPR_HEIGHT),

      
		.SPR_SCALE(SPR_SCALE),

      
		.SPR_DATAW(SPR_DATAW)

      
	) sprite_f(

      
		.clk(clk_pix),

      
		.rst(rst_pix),

      
		.line(line),

      
		.sx(sx),

      
		.sy(sy),

      
		.sprx(sprx),

      
		.spry(spry),

      
		.pix(pix),

      
		.drawing(drawing)

      
	);

      
	reg [3:0] paint_r;

      
	reg [3:0] paint_g;

      
	reg [3:0] paint_b;

      
	always @(*) begin

      
		paint_r = (drawing && pix ? 4'hf : 4'h1);

      
		paint_g = (drawing && pix ? 4'hc : 4'h3);

      
		paint_b = (drawing && pix ? 4'h0 : 4'h7);

      
	end

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync;

      
		vga_vsync <= vsync;

      
		if (de) begin

      
			vga_r <= paint_r;

      
			vga_g <= paint_g;

      
			vga_b <= paint_b;

      
		end

      
		else begin

      
			vga_r <= 4'h0;

      
			vga_g <= 4'h0;

      
			vga_b <= 4'h0;

      
		end

      
	end
