module top_david_v1 (

      
	clk_100m,

      
	btn_rst,

      
	vga_hsync,

      
	vga_vsync,

      
	vga_r,

      
	vga_g,

      
	vga_b

      
);

      
	input wire clk_100m;

      
	input wire btn_rst;

      
	output reg vga_hsync;

      
	output reg vga_vsync;

      
	output reg [3:0] vga_r;

      
	output reg [3:0] vga_g;

      
	output reg [3:0] vga_b;

      
	wire clk_pix;

      
	wire clk_locked;

      
	clock_gen_480p clock_pix_inst(

      
		.clk(clk_100m),

      
		.rst(!btn_rst),

      
		.clk_pix(clk_pix),

      
		.clk_locked(clk_locked)

      
	);

      
	localparam CORDW = 16;

      
	wire signed [15:0] sx;

      
	wire signed [15:0] sy;

      
	wire hsync;

      
	wire vsync;

      
	wire frame;

      
	display_480p #(.CORDW(CORDW)) display_inst(

      
		.clk_pix(clk_pix),

      
		.rst_pix(!clk_locked),

      
		.sx(sx),

      
		.sy(sy),

      
		.hsync(hsync),

      
		.vsync(vsync),

      
		.de(),

      
		.frame(frame),

      
		.line()

      
	);

      
	localparam FB_WIDTH = 160;

      
	localparam FB_HEIGHT = 120;

      
	localparam FB_PIXELS = 19200;

      
	localparam FB_ADDRW = 15;

      
	localparam FB_DATAW = 1;

      
	localparam FB_IMAGE = "david_1bit.mem";

      
	reg fb_we;

      
	reg [14:0] fb_addr_write;

      
	reg [14:0] fb_addr_read;

      
	reg [0:0] fb_colr_write;

      
	wire [0:0] fb_colr_read;

      
	bram_sdp #(

      
		.WIDTH(FB_DATAW),

      
		.DEPTH(FB_PIXELS),

      
		.INIT_F(FB_IMAGE)

      
	) bram_inst(

      
		.clk_write(clk_pix),

      
		.clk_read(clk_pix),

      
		.we(fb_we),

      
		.addr_write(fb_addr_write),

      
		.addr_read(fb_addr_read),

      
		.data_in(fb_colr_write),

      
		.data_out(fb_colr_read)

      
	);

      
	reg [7:0] cnt_draw;

      
	reg [31:0] state;

      
	always @(posedge clk_pix)

      
		case (state)

      
			32'd1:

      
				if (cnt_draw < 159) begin

      
					fb_addr_write <= fb_addr_write + 1;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					cnt_draw <= 0;

      
					state <= 32'd2;

      
				end

      
			32'd2:

      
				if (cnt_draw < 119) begin

      
					fb_addr_write <= fb_addr_write + FB_WIDTH;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					fb_addr_write <= 0;

      
					cnt_draw <= 0;

      
					state <= 32'd4;

      
				end

      
			32'd4:

      
				if (cnt_draw < 119) begin

      
					fb_addr_write <= fb_addr_write + FB_WIDTH;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					cnt_draw <= 0;

      
					state <= 32'd3;

      
				end

      
			32'd3:

      
				if (cnt_draw < 159) begin

      
					fb_addr_write <= fb_addr_write + 1;

      
					cnt_draw <= cnt_draw + 1;

      
				end

      
				else begin

      
					fb_we <= 0;

      
					state <= 32'd5;

      
				end

      
			32'd0:

      
				if (frame) begin

      
					fb_colr_write <= 1;

      
					fb_we <= 1;

      
					cnt_draw <= 0;

      
					state <= 32'd1;

      
				end

      
			default: state <= 32'd5;

      
		endcase

      
	reg paint;

      
	always @(*) paint = (((sy >= 0) && (sy < FB_HEIGHT)) && (sx >= 0)) && (sx < FB_WIDTH);

      
	always @(posedge clk_pix)

      
		if (frame)

      
			fb_addr_read <= 0;

      
		else if (paint)

      
			fb_addr_read <= fb_addr_read + 1;

      
	reg paint_p1;

      
	reg hsync_p1;

      
	reg vsync_p1;

      
	always @(posedge clk_pix) begin

      
		paint_p1 <= paint;

      
		hsync_p1 <= hsync;

      
		vsync_p1 <= vsync;

      
	end

      
	always @(posedge clk_pix) begin

      
		vga_hsync <= hsync_p1;

      
		vga_vsync <= vsync_p1;

      
		vga_r <= (paint_p1 && fb_colr_read ? 4'hf : 4'h0);

      
		vga_g <= (paint_p1 && fb_colr_read ? 4'hf : 4'h0);

      
		vga_b <= (paint_p1 && fb_colr_read ? 4'hf : 4'h0);

      
	end
